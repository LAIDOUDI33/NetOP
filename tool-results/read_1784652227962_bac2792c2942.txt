   107→   107→function severityBadgeVariant(s: IncidentSeverity): 'default' | 'secondary' | 'destructive' | 'outline' {
   108→   108→  if (s === 'critical' || s === 'high') return 'destructive';
   109→   109→  if (s === 'medium') return 'secondary';
   110→   110→  return 'default';
   111→   111→}
   112→   112→
   113→   113→function statusBadgeVariant(s: IncidentStatus): 'default' | 'secondary' | 'destructive' | 'outline' {
   114→   114→  if (s === 'open') return 'destructive';
   115→   115→  if (s === 'investigating') return 'secondary';
   116→   116→  if (s === 'resolved') return 'outline';
   117→   117→  return 'secondary';
   118→   118→}
   119→   119→
   120→   120→function computeMTTR(incident: Incident): string {
   121→   121→  if (!incident.resolvedAt) return '—';
   122→   122→  const created = new Date(incident.createdAt).getTime();
   123→   123→  const resolved = new Date(incident.resolvedAt).getTime();
   124→   124→  const minutes = (resolved - created) / 60000;
   125→   125→  return `${Math.round(minutes)} min`;
   126→   126→}
   127→   127→
   128→   128→function formatDate(iso: string): string {
   129→   129→  const d = new Date(iso);
   130→   130→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
   131→   131→}

   203→   203→const RADIAN = Math.PI / 180;
   204→   204→function renderPieLabel({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) {
   205→   205→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   206→   206→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   207→   207→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   208→   208→  if (percent < 0.06) return null;
   209→   209→  return (
   210→   210→    <text x={x} y={y} fill="white" textAnchor="middle" dominantBaseline="central" fontSize={11} fontWeight={600}>
   211→   211→      {`${(percent * 100).toFixed(0)}%`}
   212→   212→    </text>
   213→   213→  );
   214→   214→}
