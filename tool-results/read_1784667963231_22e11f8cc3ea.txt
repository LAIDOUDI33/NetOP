    99→// ─── Helper Functions ──────────────────────────────────────────────────
   100→
   101→function mosColor(score: number | null): string {
   102→  if (score == null) return 'text-muted-foreground';
   103→  if (score >= 4) return 'text-emerald-600 dark:text-emerald-400';
   104→  if (score >= 3) return 'text-amber-600 dark:text-amber-400';
   105→  return 'text-red-600 dark:text-red-400';
   106→}
   107→
   108→function satisfactionColor(index: number | null): string {
   109→  if (index == null) return 'text-muted-foreground';
   110→  if (index >= 70) return 'text-emerald-600 dark:text-emerald-400';
   111→  if (index >= 50) return 'text-amber-600 dark:text-amber-400';
   112→  return 'text-red-600 dark:text-red-400';
   113→}
   114→
   115→function satisfactionBg(index: number | null): string {
   116→  if (index == null) return 'bg-muted';
   117→  if (index >= 70) return 'bg-emerald-500';
   118→  if (index >= 50) return 'bg-amber-500';
   119→  return 'bg-red-500';
   120→}
   121→
   122→function computeOverallAvgMos(sites: QoESiteSummary[]): number {
   123→  const valid = sites.filter((s) => s.mosScore != null);
   124→  if (valid.length === 0) return 0;
   125→  return Number((valid.reduce((sum, s) => sum + (s.mosScore ?? 0), 0) / valid.length).toFixed(2));
   126→}
   127→
   128→function computeOverallAvgSatisfaction(sites: QoESiteSummary[]): number {
   129→  const valid = sites.filter((s) => s.satisfactionIndex != null);
   130→  if (valid.length === 0) return 0;
   131→  return Number((valid.reduce((sum, s) => sum + (s.satisfactionIndex ?? 0), 0) / valid.length).toFixed(1));
   132→}
   133→
   134→function formatBucketTime(bucket: string): string {
   135→  const d = new Date(bucket);
   136→  return `${String(d.getHours()).padStart(2, '0')}:00`;
   137→}
