    73→function stdDev(arr: number[]) {
    74→  if (arr.length === 0) return 0;
    75→  const mean = arr.reduce((a, b) => a + b, 0) / arr.length;
    76→  const squaredDiffs = arr.map(v => Math.pow(v - mean, 2));
    77→  return Math.sqrt(squaredDiffs.reduce((a, b) => a + b, 0) / arr.length);
    78→}
    79→
    80→function formatValue(value: number, metric: string): string {
    81→  return PERCENT_METRICS.has(metric) ? `${value.toFixed(2)}%` : value.toFixed(2);
    82→}
    83→
    84→function isBreach(value: number, metric: string, tech: string): boolean {
    85→  for (const threshold of Object.values(BREACH_THRESHOLDS)) {
    86→    if (threshold.metric !== metric) continue;
    87→    if (threshold.tech && threshold.tech !== tech) continue;
    88→    if (threshold.condition === 'lt' && value < threshold.value) return true;
    89→    if (threshold.condition === 'gt' && value > threshold.value) return true;
    90→  }
    91→  return false;
    92→}
    93→
    94→function countBreaches(sites: { technology: Technology; value: number }[], metric: string): number {
    95→  let count = 0;
    96→  for (const site of sites) {
    97→    if (isBreach(site.value, metric, site.technology)) count++;
    98→  }
    99→  return count;
   100→}

   102→function formatTime(iso: string): string {
   103→  try {
   104→    return new Date(iso).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
   105→  } catch {
   106→    return iso;
   107→  }
   108→}
   109→
   110→function formatDateTime(iso: string | null): string {
   111→  if (!iso) return '—';
   112→  try {
   113→    return new Date(iso).toLocaleString([], { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' });
   114→  } catch {
   115→    return iso;
   116→  }
   117→}

   173→function KpiReportTab() {
   174→  const [metric, setMetric] = useState('downloadThroughput');
   175→
   176→  const { data, isLoading } = useQuery<KpiResponse>({
   177→    queryKey: ['kpi-report', metric],
   178→    queryFn: () => fetch(`/api/kpi?technology=all&metric=${metric}`).then(r => r.json()),
   179→    refetchInterval: 60000,
   180→  });
   181→
   182→  const chartData = useMemo(() => {
   183→    if (!data) return [];
   184→    return data.timestamps.map((ts, i) => {
   185→      const point: Record<string, string | number> = { time: ts };
   186→      if (data.data) {
   187→        Object.entries(data.data).forEach(([tech, techData]) => {
   188→          point[tech] = techData.values[i];
   189→        });
   190→      }
   191→      return point;
   192→    });
   193→  }, [data]);
   194→
   195→  const rankedSites = useMemo(() => {
   196→    if (!data?.data) return [];
   197→    return Object.values(data.data)
   198→      .flatMap(d => d.sites)
   199→      .map(s => ({
   200→        ...s,
   201→        displayValue: formatValue(s.value, metric),
   202→        breached: isBreach(s.value, metric, s.technology),
   203→      }))
   204→      .sort((a, b) => b.value - a.value);
   205→  }, [data, metric]);
   206→
   207→  const summaryStats = useMemo(() => {
   208→    if (!data?.data) return { min: '0', max: '0', avg: '0', stddev: '0', breaches: 0 };
   209→    const allValues = Object.values(data.data).flatMap(d => d.values);
   210→    const allSites = Object.values(data.data).flatMap(d => d.sites);
   211→    if (allValues.length === 0) return { min: '0', max: '0', avg: '0', stddev: '0', breaches: 0 };
   212→
   213→    const min = Math.min(...allValues);
   214→    const max = Math.max(...allValues);
   215→    const avg = allValues.reduce((a, b) => a + b, 0) / allValues.length;
   216→    const sd = stdDev(allValues);
   217→    const breaches = countBreaches(allSites, metric);
   218→
   219→    return {
   220→      min: formatValue(min, metric),
   221→      max: formatValue(max, metric),
   222→      avg: formatValue(avg, metric),
   223→      stddev: formatValue(sd, metric),
   224→      breaches,
   225→    };
   226→  }, [data, metric]);
   227→
   228→  if (isLoading || !data) {
   229→    return (
   230→      <div className="space-y-4">
   231→        <div className="no-print flex flex-wrap gap-4 items-center">
   232→          <Skeleton className="h-10 w-56" />
   233→        </div>
   234→        <ChartSkeleton />
   235→        <SummarySkeleton />
   236→        <TableSkeleton />
   237→      </div>
   238→    );
   239→  }
   240→
   241→  const techs = data.technologies || [];
   242→
   243→  return (
   244→    <div className="space-y-4">
   245→      {/* Metric Filter */}
   246→      <div className="no-print flex flex-wrap gap-4 items-center">
   247→        <Select value={metric} onValueChange={setMetric}>
   248→          <SelectTrigger className="w-[280px]">
   249→            <SelectValue placeholder="Select Metric" />
   250→          </SelectTrigger>
   251→          <SelectContent>
   252→            {METRICS.map((m) => (
   253→              <SelectItem key={m.value} value={m.value}>{m.label}</SelectItem>
   254→            ))}
   255→          </SelectContent>
   256→        </Select>
   257→      </div>
   258→
   259→      {/* Combined Trend Chart */}
   260→      <Card>
   261→        <CardHeader className="pb-2">
   262→          <CardTitle className="text-base font-semibold">
   263→            {METRICS.find(m => m.value === metric)?.label} — All Technologies
   264→          </CardTitle>
   265→        </CardHeader>
   266→        <CardContent className="p-4">
   267→          <div className="h-72">
   268→            <ResponsiveContainer width="100%" height="100%">
   269→              <LineChart data={chartData} margin={{ top: 5, right: 10, left: -10, bottom: 5 }}>
   270→                <CartesianGrid strokeDasharray="3 3" className="opacity-30" />
   271→                <XAxis dataKey="time" tick={{ fontSize: 11 }} />
   272→                <YAxis tick={{ fontSize: 11 }} />
   273→                <Tooltip
   274→                  contentStyle={{ borderRadius: '8px', fontSize: '12px', border: '1px solid hsl(var(--border))' }}
   275→                  labelStyle={{ fontWeight: 600 }}
   276→                />

   278→                {techs.map((tech) => (
   279→                  <Line
   280→                    key={tech}
   281→                    type="monotone"
   282→                    dataKey={tech}
   283→                    stroke={TECH_COLORS[tech]}
   284→                    name={tech}
   285→                    strokeWidth={2}

   287→                  />
   288→                ))}
   289→              </LineChart>
   290→            </ResponsiveContainer>
   291→          </div>
   292→        </CardContent>
   293→      </Card>
   294→
   295→      {/* Summary Statistics */}
   296→      <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-5 gap-4">
   297→        <StatCard icon={ArrowDownRight} label="Minimum" value={summaryStats.min} />
   298→        <StatCard icon={ArrowUpRight} label="Maximum" value={summaryStats.max} />
   299→        <StatCard icon={TrendingUp} label="Average" value={summaryStats.avg} />
   300→        <StatCard icon={TrendingDown} label="Std Dev" value={summaryStats.stddev} />
   301→        <StatCard icon={AlertTriangle} label="Breach Count" value={summaryStats.breaches} variant={summaryStats.breaches > 0 ? 'warning' : 'default'} />
   302→      </div>
   303→
   304→      {/* Site Performance Ranking */}
   305→      <Card>
   306→        <CardHeader className="pb-2">
   307→          <CardTitle className="text-base font-semibold">
   308→            Site Performance Ranking — {METRICS.find(m => m.value === metric)?.label}
   309→          </CardTitle>
   310→        </CardHeader>
   311→        <CardContent className="p-4">
   312→          <ScrollArea className="max-h-96">
   313→            <div className="min-w-[540px]">

   321→                    <TableHead className="text-xs text-right">Value</TableHead>
   322→                  </TableRow>
   323→                </TableHeader>
   324→                <TableBody>
   325→                  {rankedSites.map((site, i) => (
   326→                    <TableRow key={site.siteId}>
   327→                      <TableCell className="text-xs text-muted-foreground">
   328→                        {i < 3 ? (

   372→function SonActivityTab() {
   373→  const { data, isLoading } = useQuery<SonResponse>({
   374→    queryKey: ['son-report'],
   375→    queryFn: () => fetch('/api/son').then(r => r.json()),
   376→    refetchInterval: 60000,
   377→  });
   378→
   379→  const modules = data?.modules || [];
   380→
   381→  const summaryCards = useMemo(() => {
   382→    const totalModules = modules.length;
   383→    let actions24h = 0;
   384→    let successCount = 0;
   385→    let totalImpact = 0;
   386→    let impactCount = 0;
   387→
   388→    const now = new Date();
   389→    const oneDayAgo = new Date(now.getTime() - 24 * 60 * 60 * 1000);
   390→
   391→    for (const mod of modules) {
   392→      for (const action of mod.recentActions) {
   393→        const actionDate = new Date(action.createdAt);
   394→        if (actionDate >= oneDayAgo) {
   395→          actions24h++;
   396→          if (action.status === 'applied') successCount++;
   397→          if (action.impactScore !== null && action.impactScore !== undefined) {
   398→            totalImpact += action.impactScore;
   399→            impactCount++;
   400→          }
   401→        }
   402→      }
   403→    }
   404→
   405→    const successRate = actions24h > 0 ? ((successCount / actions24h) * 100).toFixed(1) : '—';
   406→    const avgImpact = impactCount > 0 ? (totalImpact / impactCount).toFixed(2) : '—';
   407→
   408→    return { totalModules, actions24h, successRate, avgImpact };
   409→  }, [modules]);
   410→
   411→  const allActions = useMemo(() => {
   412→    return modules
   413→      .flatMap(m => m.recentActions.map(a => ({ ...a, moduleName: m.displayName, moduleTechnology: m.technology })))
   414→      .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
   415→  }, [modules]);
   416→
   417→  if (isLoading) {
   418→    return (
   419→      <div className="space-y-4">
   420→        <SummarySkeleton />
   421→        <TableSkeleton />
   422→      </div>
   423→    );
   424→  }
   425→
   426→  if (modules.length === 0) {
   427→    return (
   428→      <Card>
   429→        <CardContent className="p-12 text-center text-muted-foreground">
   430→          <Cpu className="h-10 w-10 mx-auto mb-3 opacity-40" />
   431→          <p className="text-sm">No SON modules found.</p>
   432→        </CardContent>
   433→      </Card>
   434→    );
   435→  }
   436→
   437→  return (
   438→    <div className="space-y-4">
   439→      {/* Summary Cards */}
   440→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   441→        <StatCard icon={Cpu} label="Total Modules" value={summaryCards.totalModules} />
   442→        <StatCard icon={Activity} label="Actions (24h)" value={summaryCards.actions24h} />
   443→        <StatCard icon={CheckCircle} label="Success Rate" value={`${summaryCards.successRate}%`} />
   444→        <StatCard icon={TrendingUp} label="Avg Impact" value={summaryCards.avgImpact} />
   445→      </div>
   446→
   447→      <Separator />
   448→
   449→      {/* Recent Actions Table */}
   450→      <Card>
   451→        <CardHeader className="pb-2">
   452→          <CardTitle className="text-base font-semibold">Recent SON Actions</CardTitle>
   453→        </CardHeader>
   454→        <CardContent className="p-4">
   455→          <ScrollArea className="max-h-96">
   456→            <div className="min-w-[800px]">
   457→              <Table>
   458→                <TableHeader>
   459→                  <TableRow>
   460→                    <TableHead className="text-xs">Time</TableHead>
   461→                    <TableHead className="text-xs">Module</TableHead>
   462→                    <TableHead className="text-xs">Site</TableHead>
   463→                    <TableHead className="text-xs">Action Type</TableHead>
   464→                    <TableHead className="text-xs">Parameter</TableHead>
   465→                    <TableHead className="text-xs">Before → After</TableHead>
   466→                    <TableHead className="text-xs text-right">Impact</TableHead>
   467→                    <TableHead className="text-xs">Status</TableHead>
   468→                  </TableRow>
   469→                </TableHeader>
   470→                <TableBody>
   471→                  {allActions.length === 0 ? (
   472→                    <TableRow>
   473→                      <TableCell colSpan={8} className="text-center text-muted-foreground text-xs py-8">
   474→                        No actions recorded yet.
   475→                      </TableCell>
   476→                    </TableRow>
   477→                  ) : (
   478→                    allActions.map((action) => (
   479→                      <TableRow key={action.id}>
   480→                        <TableCell className="text-xs text-muted-foreground whitespace-nowrap">
   481→                          <span className="flex items-center gap-1"><Clock className="h-3 w-3" />{formatTime(action.createdAt)}</span>
   482→                        </TableCell>
   483→                        <TableCell className="text-xs font-medium">{action.moduleName}</TableCell>
   484→                        <TableCell className="text-xs">{action.siteName || '—'}{action.siteCode ? ` (${action.siteCode})` : ''}</TableCell>
   485→                        <TableCell className="text-xs">
   486→                          <Badge variant="outline" className="text-xs font-mono">{action.actionType}</Badge>
   487→                        </TableCell>
   488→                        <TableCell className="text-xs font-mono">{action.parameter}</TableCell>
   489→                        <TableCell className="text-xs whitespace-nowrap">
   490→                          <span className="text-muted-foreground">{action.previousValue}</span>
   491→                          <span className="mx-1.5 text-muted-foreground">→</span>
   492→                          <span className="font-medium">{action.newValue}</span>
   493→                        </TableCell>
   494→                        <TableCell className="text-xs text-right font-medium">
   495→                          {action.impactScore !== null && action.impactScore !== undefined

   505→                            <Badge variant="default" className="text-xs gap-1"><CheckCircle className="h-3 w-3" />Applied</Badge>
   506→                          )}
   507→                          {action.status === 'pending' && (

   509→                          )}
   510→                          {action.status === 'rolled_back' && (
   511→                            <Badge variant="outline" className="text-xs gap-1">Rolled Back</Badge>
   512→                          )}
   513→                          {action.status === 'failed' && (
   514→                            <Badge variant="destructive" className="text-xs gap-1"><XCircle className="h-3 w-3" />Failed</Badge>
   515→                          )}
   516→                        </TableCell>
   517→                      </TableRow>
   518→                    ))
   519→                  )}
   520→                </TableBody>
   521→              </Table>
   522→            </div>
   523→          </ScrollArea>
   524→        </CardContent>
   525→      </Card>
   526→    </div>
   527→  );
   528→}

   532→function PolicyReportTab() {
   533→  const { data, isLoading } = useQuery<PolicyResponse>({
   534→    queryKey: ['policy-report'],
   535→    queryFn: () => fetch('/api/policies').then(r => r.json()),
   536→    refetchInterval: 60000,
   537→  });
   538→
   539→  const policies = data?.policies || [];
   540→
   541→  const summaryCards = useMemo(() => {
   542→    const totalPolicies = policies.length;
   543→    const activePolicies = policies.filter(p => p.enabled).length;
   544→    const totalExecutions = policies.reduce((sum, p) => sum + p.executionStats.totalRuns, 0);
   545→    const rates = policies.filter(p => p.executionStats.totalRuns > 0).map(p => p.executionStats.successRate);
   546→    const avgSuccessRate = rates.length > 0
   547→      ? (rates.reduce((a, b) => a + b, 0) / rates.length).toFixed(1)
   548→      : '—';
   549→
   550→    return { totalPolicies, activePolicies, totalExecutions, avgSuccessRate };
   551→  }, [policies]);
   552→
   553→  if (isLoading) {
   554→    return (
   555→      <div className="space-y-4">
   556→        <SummarySkeleton />
   557→        <TableSkeleton />
   558→      </div>
   559→    );
   560→  }
   561→
   562→  if (policies.length === 0) {
   563→    return (
   564→      <Card>
   565→        <CardContent className="p-12 text-center text-muted-foreground">
   566→          <Shield className="h-10 w-10 mx-auto mb-3 opacity-40" />
   567→          <p className="text-sm">No policies found.</p>
   568→        </CardContent>
   569→      </Card>
   570→    );
   571→  }
   572→
   573→  return (
   574→    <div className="space-y-4">
   575→      {/* Summary Cards */}
   576→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   577→        <StatCard icon={Shield} label="Total Policies" value={summaryCards.totalPolicies} />
   578→        <StatCard icon={Activity} label="Active Policies" value={summaryCards.activePolicies} />
   579→        <StatCard icon={Cpu} label="Total Executions" value={summaryCards.totalExecutions} />
   580→        <StatCard icon={TrendingUp} label="Avg Success Rate" value={`${summaryCards.avgSuccessRate}%`} />
   581→      </div>
   582→
   583→      <Separator />
   584→
   585→      {/* Policies Table */}
   586→      <Card>
   587→        <CardHeader className="pb-2">
   588→          <CardTitle className="text-base font-semibold">Policy Execution Summary</CardTitle>
   589→        </CardHeader>
   590→        <CardContent className="p-4">
   591→          <ScrollArea className="max-h-96">
   592→            <div className="min-w-[700px]">
   593→              <Table>
   594→                <TableHeader>
   595→                  <TableRow>
   596→                    <TableHead className="text-xs">Name</TableHead>
   597→                    <TableHead className="text-xs">Tech</TableHead>
   598→                    <TableHead className="text-xs">Trigger</TableHead>
   599→                    <TableHead className="text-xs">Scope</TableHead>
   600→                    <TableHead className="text-xs text-center">Priority</TableHead>
   601→                    <TableHead className="text-xs text-center">Enabled</TableHead>
   602→                    <TableHead className="text-xs text-right">Total Runs</TableHead>
   603→                    <TableHead className="text-xs text-right">Success Rate</TableHead>
   604→                    <TableHead className="text-xs">Last Run</TableHead>
   605→                  </TableRow>
   606→                </TableHeader>
   607→                <TableBody>
   608→                  {policies.map((policy) => (
   609→                    <TableRow key={policy.id}>
   610→                      <TableCell className="text-xs font-medium">{policy.name}</TableCell>
   611→                      <TableCell>
   612→                        <Badge
   613→                          className="text-xs"
   614→                          style={{ backgroundColor: TECH_COLORS[policy.technology as Technology] || '#94A3B8', color: '#fff' }}
   615→                        >
   616→                          {policy.technology}
   617→                        </Badge>
   618→                      </TableCell>
   619→                      <TableCell className="text-xs">
   620→                        <Badge variant="outline" className="text-xs font-mono">{policy.triggerType}</Badge>
   621→                      </TableCell>
   622→                      <TableCell className="text-xs">{policy.scope}{policy.scopeValue ? ` (${policy.scopeValue})` : ''}</TableCell>
   623→                      <TableCell className="text-xs text-center">
   624→                        <Badge variant={policy.priority <= 3 ? 'destructive' : policy.priority <= 7 ? 'secondary' : 'outline'} className="text-xs">
   625→                          {policy.priority}
   626→                        </Badge>
   627→                      </TableCell>
   628→                      <TableCell className="text-xs text-center">
   629→                        {policy.enabled ? (
   630→                          <CheckCircle className="h-4 w-4 text-emerald-500 mx-auto" />
   631→                        ) : (
   632→                          <XCircle className="h-4 w-4 text-muted-foreground mx-auto" />
   633→                        )}
   634→                      </TableCell>
   635→                      <TableCell className="text-xs text-right font-medium">{policy.executionStats.totalRuns}</TableCell>
   636→                      <TableCell className="text-xs text-right font-medium">
   637→                        {policy.executionStats.totalRuns > 0 ? (
   638→                          <span className={policy.executionStats.successRate >= 80 ? 'text-emerald-600' : policy.executionStats.successRate >= 50 ? 'text-amber-600' : 'text-red-600'}>
   639→                            {policy.executionStats.successRate}%
   640→                          </span>
   641→                        ) : '—'}
   642→                      </TableCell>
   643→                      <TableCell className="text-xs text-muted-foreground whitespace-nowrap">
   644→                        {formatDateTime(policy.executionStats.lastRun)}
   645→                      </TableCell>
   646→                    </TableRow>
   647→                  ))}
   648→                </TableBody>
   649→              </Table>
   650→            </div>
   651→          </ScrollArea>
   652→        </CardContent>
   653→      </Card>
   654→    </div>
   655→  );
   656→}
