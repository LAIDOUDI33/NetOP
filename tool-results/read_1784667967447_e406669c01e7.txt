    94→function formatTime(ts: string | null | undefined): string {
    95→  if (!ts) return '—';
    96→  try {
    97→    const d = new Date(ts);
    98→    return d.toLocaleString(undefined, {
    99→      month: 'short',
   100→      day: 'numeric',
   101→      hour: '2-digit',
   102→      minute: '2-digit',
   103→    });
   104→  } catch {
   105→    return '—';
   106→  }
   107→}
   108→
   109→function formatDuration(ms: number | null): string {
   110→  if (!ms) return '—';
   111→  if (ms < 1000) return `${ms}ms`;
   112→  if (ms < 60000) return `${(ms / 1000).toFixed(1)}s`;
   113→  return `${(ms / 60000).toFixed(1)}m`;
   114→}

   116→function getTechBadgeClass(tech: string): string {
   117→  const map: Record<string, string> = {
   118→    '2G': 'bg-slate-500/10 text-slate-700 dark:text-slate-300 border-slate-500/20',
   119→    '3G': 'bg-cyan-500/10 text-cyan-700 dark:text-cyan-300 border-cyan-500/20',
   120→    '4G': 'bg-emerald-500/10 text-emerald-700 dark:text-emerald-300 border-emerald-500/20',
   121→    '5G': 'bg-amber-500/10 text-amber-700 dark:text-amber-300 border-amber-500/20',
   122→  };
   123→  return map[tech] ?? 'bg-slate-500/10 text-slate-700 dark:text-slate-300 border-slate-500/20';
   124→}

   549→                        : 'bg-slate-500/10 text-slate-700 dark:text-slate-300 border-slate-500/20'
   550→                  }`}
   551→                >
   552→                  <Activity className="h-3 w-3" />
   553→                  {kpiImpactEntries.length} KPI{kpiImpactEntries.length !== 1 ? 's' : ''}
   554→                </Badge>
   555→              </TooltipTrigger>
   556→              <TooltipContent>
   557→                <div className="space-y-1">
   558→                  {kpiImpactEntries.map(([k, v]) => (
   559→                    <p key={k} className="text-xs">
   560→                      {k}: <span className={Number(v) > 0 ? 'text-emerald-500' : Number(v) < 0 ? 'text-red-500' : ''}>{String(v)}</span>
   561→                    </p>
   562→                  ))}
   563→                </div>
   564→              </TooltipContent>
   565→            </Tooltip>
   566→          </TooltipProvider>
   567→        ) : (
   568→          <span className="text-xs text-muted-foreground">—</span>
   569→        )}
   570→      </TableCell>
   571→
   572→      {/* Rollback */}
   573→      <TableCell className="max-w-[150px]">
   574→        {execution.status === 'rolled_back' && execution.rollbackReason ? (
   575→          <TooltipProvider>
   576→            <Tooltip>
   577→              <TooltipTrigger asChild>
   578→                <Badge variant="outline" className="text-xs gap-1 bg-slate-500/10 text-slate-600 dark:text-slate-400 border-slate-500/20 cursor-default">
   579→                  <RotateCcw className="h-3 w-3" />
   580→                  {t('status.rolledBack')}
   581→                </Badge>
   582→              </TooltipTrigger>
   583→              <TooltipContent className="max-w-[250px]">
   584→                <p className="text-xs">{execution.rollbackReason}</p>
   585→              </TooltipContent>
   586→            </Tooltip>
   587→          </TooltipProvider>
   588→        ) : (
   589→          <span className="text-xs text-muted-foreground">—</span>
   590→        )}
   591→      </TableCell>
   592→    </TableRow>
   593→  );
   594→}
