     3→ * Phase 11: Unified Security Integration Hub

     5→ * This module provides a unified interface to all security tool integrations,

     7→ *

    17→ * ┌─────────────────────────────────────────────────────┐

   164→  private openctiClient?: OpenctiClient;

   166→  private soarClient?: TheHiveCortexClient;
   167→  private vulnClient?: OpenvasDefectDojoClient;
   168→  
   169→  // State
   170→  private isInitialized = false;
   171→  private startTime: Date = new Date();
   172→  private eventBuffer: CorrelatedEvent[] = [];
   173→  private bufferFlushInterval: NodeJS.Timeout | null = null;
   174→  
   175→  // Statistics
   176→  public stats = {
   177→    eventsProcessed: 0,
   178→    correlationsPerformed: 0,
   179→    autoResponsesExecuted: 0,
   180→    errorsEncountered: 0,

   185→  constructor(config: PlatformIntegrationConfig) {
   186→    super();
   187→    this.config = config;
   188→
   189→    // Set up internal handlers
   190→    this.on('error', (err) => {
   191→      console.error('[COORDINATOR] Error:', err.message);

   193→    });
   194→
   195→    this.on('correlationComplete', (event: CorrelatedEvent) => {
   196→      console.log(`[COORDINATOR] Correlation complete: ${event.title} (${event.sourceEvents.length} sources)`);
   197→      
   198→      // Auto-escalation if configured
   199→      if (this.shouldAutoEscalate(event)) {
   200→        this.executeAutoEscalation(event);
   201→      }
   202→    });
   203→
   204→    // Start buffer flush interval

   212→  /**
   213→   * Initialize all configured integrations
   214→   */
   215→  async initialize(): Promise<void> {
   216→    if (this.isInitialized) {
   217→      console.warn('[COORDINATOR] Already initialized');
   218→      return;
   219→    }
   220→
   221→    console.log('[COORDINATOR] Initializing all integrations...');
   222→
   223→    try {
   224→      // Initialize clients in order of dependency
   225→      const initPromises: Promise<any>[] = [];
   226→
   227→      // EDR (GRR/Osquery)
   228→      if (this.config.edr) {
   229→        this.edrClient = new GrrOsqueryEdrClient(this.config.edr);
   230→        initPromises.push(
   231→          this.edrClient.connect()
   232→            .then(() => console.log('[COORDINATOR] ✓ EDR connected'))
   233→            .catch(err => { console.error('[COORDINATOR] ✗ EDR failed:', err.message); throw err; })
   234→        );
   235→      }
   236→
   237→      // NSM (Suricata/Zeek/Arkime)
   238→      if (this.config.nsm) {
   239→        this.nsmClient = new SuricataZeekArkimeClient(this.config.nsm);
   240→        initPromises.push(
   241→          this.nsmClient.connect()
   242→            .then(() => console.log('[COORDINATOR] ✓ NSM connected'))
   243→            .catch(err => { console.error('[COORDINATOR] ✗ NSM failed:', err.message); throw err; })
   244→        );
   245→      }
   246→
   247→      // SIEM (Wazuh/Elasticsearch)
   248→      if (this.config.siem) {
   249→        this.siemClient = new WazuhElasticsearchClient(this.config.siem);
   250→        initPromises.push(
   251→          this.siemClient.connect()
   252→            .then(() => console.log('[COORDINATOR] ✓ SIEM connected'))
   253→            .catch(err => { console.error('[COORDINATOR] ✗ SIEM failed:', err.message); throw err; })
   254→        );
   255→      }
   256→
   257→      // Threat Intelligence (OpenCTI)
   258→      if (this.config.opencti) {
   259→        this.openctiClient = new OpenctiClient(this.config.opencti);
   260→        initPromises.push(
   261→          this.openctiClient.connect()
   262→            .then(() => console.log('[COORDINATOR] ✓ OpenCTI connected'))
   263→            .catch(err => { console.error('[COORDINATOR] ✗ OpenCTI failed:', err.message); throw err; })
   264→        );
   265→      }
   266→
   267→      // Threat Intelligence (MISP)
   268→      if (this.config.misp) {
   269→        this.mispClient = new MispClient(this.config.misp.config);
   270→        initPromises.push(
   271→          this.mispClient.connect()
   272→            .then(() => console.log('[COORDINATOR] ✓ MISP connected'))
   273→            .catch(err => { console.error('[COORDINATOR] ✗ MISP failed:', err.message); throw err; })
   274→        );
   275→      }
   276→
   277→      // SOAR (TheHive/Cortex)
   278→      if (this.config.soar) {
   279→        this.soarClient = new TheHiveCortexClient(this.config.soar);
   280→        initPromises.push(
   281→          this.soarClient.connect()
   282→            .then(() => console.log('[COORDINATOR] ✓ SOAR connected'))
   283→            .catch(err => { console.error('[COORDINATOR] ✗ SOAR failed:', err.message); throw err; })
   284→        );
   285→      }
   286→
   287→      // Vulnerability Management (OpenVAS/DefectDojo)
   288→      if (this.config.vulnerability) {
   289→        this.vulnClient = new OpenvasDefectDojoClient(this.config.vulnerability);
   290→        initPromises.push(
   291→          this.vulnClient.connect()
   292→            .then(() => console.log('[COORDINATOR] ✓ Vulnerability Management connected'))
   293→            .catch(err => { console.error('[COORDINATOR] ✗ Vulnerability Management failed:', err.message); throw err; })
   294→        );
   295→      }
   296→
   297→      // Wait for all initializations (with partial failure tolerance)
   298→      const results = await Promise.allSettled(initPromises);
   299→
   300→      const succeeded = results.filter(r => r.status === 'fulfilled').length;
   301→      const failed = results.filter(r => r.status === 'rejected').length;
   302→
   303→      console.log(`[COORDINATOR] Initialization complete: ${succeeded} succeeded, ${failed} failed`);
   304→
   305→      this.isInitialized = true;
   306→      this.emit('initialized', { succeeded, failed, timestamp: new Date() });
   307→
   308→    } catch (error) {
   309→      this.isInitialized = false;
   310→      this.emit('error', error);
   311→      throw error;
   312→    }
   313→  }

   315→  /**
   316→   * Gracefully shutdown all connections
   317→   */
   318→  async shutdown(): Promise<void> {
   319→    console.log('[COORDINATOR] Shutting down...');
   320→
   321→    // Clear intervals
   322→    if (this.bufferFlushInterval) {
   323→      clearInterval(this.bufferFlushInterval);
   324→    }
   325→
   326→    // Flush remaining events
   327→    await this.flushEventBuffer();
   328→
   329→    // Disconnect all clients
   330→    const disconnectPromises: Promise<any>[] = [
   331→      this.edrClient?.disconnect(),
   332→      this.nsmClient?.disconnect(),
   333→      this.siemClient?.disconnect(),
   334→      this.openctiClient?.disconnect(),
   335→      this.mispClient?.disconnect(),
   336→      this.soarClient?.disconnect(),
   337→      this.vulnClient?.disconnect(),
   338→    ].filter(Boolean);
   339→
   340→    await Promise.allSettled(disconnectPromises);
   341→
   342→    this.isInitialized = false;
   343→    this.emit('shutdown', { timestamp: new Date() });
   344→    console.log('[COORDINATOR] Shutdown complete');
   345→  }

   355→  /**
   356→   * Process an incoming event through correlation and enrichment pipeline
   357→   */
   358→  async processEvent(params: {
   359→    source: 'edr' | 'nsm' | 'siem' | 'ti' | 'vuln' | 'manual';
   360→    rawEvent: any;
   361→    severity?: string;
   362→    context?: Record<string, any>;
   363→  }): Promise<CorrelatedEvent> {
   364→    const { source, rawEvent, severity, context = {} } = params;
   365→
   366→    try {
   367→      console.log(`[COORDINATOR] Processing ${source} event`);
   368→
   369→      // Step 1: Normalize event format
   370→      const normalizedEvent = await this.normalizeEvent(source, rawEvent);
   371→
   372→      // Step 2: Enrich with additional context
   373→      const enrichedEvent = await this.enrichEvent(normalizedEvent);
   374→
   375→      // Step 3: Correlate with other sources
   376→      const correlatedEvent = await this.correlateEvent(enrichedEvent);
   377→
   378→      // Step 4: Determine if automated response should be triggered
   379→      if (this.config.autoResponseEnabled && this.shouldAutoRespond(correlatedEvent)) {
   380→        const responseResult = await this.executeAutomatedResponse(correlatedEvent);
   381→        correlatedEvent.actionsTaken = correlatedEvent.actionsTaken || [];
   382→        correlatedEvent.actionsTaken.push(responseResult);
   383→      }
   384→
   385→      // Step 5: Publish to Kafka for downstream consumers
   386→      await this.publishToKafka('alerts', correlatedEvent);
   387→
   388→      // Step 6: Buffer for batch processing
   389→      this.eventBuffer.push(correlatedEvent);
   390→
   391→      // Update statistics
   392→      this.stats.eventsProcessed++;
   393→      this.stats.lastEventTimestamp = new Date();
   394→
   395→      this.emit('eventProcessed', correlatedEvent);
   396→      return correlatedEvent;
   397→
   398→    } catch (error) {
   399→      this.emit('error', error);
   400→      throw error;
   401→    }
   402→  }
   403→
   404→  /**
   405→   * Batch process multiple events
   406→   */
   407→  async processEvents(events: Array<{
   408→    source: 'edr' | 'nsm' | 'siem' | 'ti' | 'vuln' | 'manual';
   409→    rawEvent: any;
   410→    severity?: string;
   411→  }>): Promise<CorrelatedEvent[]> {
   412→    console.log(`[COORDINATOR] Batch processing ${events.length} events`);
   413→
   414→    // Process in parallel batches for performance
   415→    const batchSize = 10;
   416→    const results: CorrelatedEvent[] = [];
   417→
   418→    for (let i = 0; i < events.length; i += batchSize) {
   419→      const batch = events.slice(i, i + batchSize);
   420→      const batchResults = await Promise.allSettled(
   421→        batch.map(event => this.processEvent(event))
   422→      );
   423→
   424→      batchResults.forEach(result => {
   425→        if (result.status === 'fulfilled') {
   426→          results.push(result.value);
   427→        } else {
   428→          console.error(`[COORDINATOR] Event processing failed:`, result.reason);
   429→        }
   430→      });
   431→    }
   432→
   433→    return results;
   434→  }

   440→  /**
   441→   * Correlate an event across all available data sources
   442→   */
   443→  private async correlateEvent(event: any): Promise<CorrelatedEvent> {
   444→    if (!this.config.autoCorrelationEnabled) {
   445→      return this.createMinimalCorrelation(event);
   446→    }
   447→
   448→    const sourceEvents: CorrelatedEvent['sourceEvents'] = [];
   449→    let correlationScore = 0;
   450→    let correlationType = '';
   451→
   452→    // Extract key identifiers for correlation
   453→    const ipAddresses = this.extractIPs(event);
   454→    const hashes = this.extractHashes(event);
   455→    const domains = this.extractDomains(event);
   456→    const msisdns = this.extractMSISDNs(event);
   457→
   458→    // Search each integration for matching events
   459→    const searchPromises: Promise<any>[] = [];
   460→
   461→    // NSM correlation (Suricata/Zeek)
   462→    if (this.nsmClient && ipAddresses.length > 0) {
   463→      searchPromises.push(
   464→        this.nsmClient.correlateEvents({ sourceIp: ipAddresses[0], timeRangeMinutes: 60 })
   465→          .then(nsmEvents => {
   466→            nsmEvents.forEach((e: any) => {
   467→              sourceEvents.push({
   468→                source: 'nsm',
   469→                originalId: e.eventId,
   470→                summary: e.title,
   471→                confidence: e.severity === 'critical' ? 90 : e.severity === 'high' ? 70 : 50,
   472→              });
   473→            });
   474→            if (nsmEvents.length > 0) correlationScore += 30;
   475→          })
   476→          .catch(() => {})
   477→      );
   478→    }
   479→
   480→    // Threat Intel correlation (OpenCTI/MISP)
   481→    if ((this.openctiClient || this.mispClient) && (ipAddresses.length > 0 || hashes.length > 0)) {
   482→      const observableValue = ipAddresses[0] || hashes[0];
   483→      const observableType = ipAddresses[0] ? 'IPv4-Addr' : 'File-Sha256';
   484→
   485→      if (this.openctiClient) {
   486→        searchPromises.push(
   487→          this.openctiClient.matchObservable({
   488→            observableValue,
   489→            observableType,
   490→            includeContext: true,
   491→          }).then(match => {
   492→            if (match.matched) {
   493→              match.indicators.forEach(indicator => {
   494→                sourceEvents.push({
   495→                  source: 'ti',
   496→                  originalId: indicator.id,
   497→                  summary: indicator.name || indicator.indicator_pattern,
   498→                  confidence: indicator.x_opencti_score || 70,
   499→                });
   500→              });
   501→              correlationScore += 40;
   502→              correlationType = 'ioc_match';
   503→              
   504→              // Attach threat intel context
   505→              event.threatIntelContext = match.context;
   506→            }
   507→          })
   508→          .catch(() => {})
   509→        );
   510→      }
   511→    }
   512→
   513→    // EDR correlation (if we have host info)
   514→    if (this.edrClient && event.hostname) {
   515→      searchPromises.push(
   516→        this.edrClient.searchClients({ query: event.hostname, limit: 5 })
   517→          .then(clients => {
   518→            if (clients.clients.length > 0) {
   519→              sourceEvents.push({
   520→                source: 'edr',
   521→                originalId: clients.clients[0].clientId,
   522→                summary: `Endpoint found: ${clients.clients[0].hostname}`,
   523→                confidence: 85,
   524→              });
   525→              correlationScore += 25;
   526→              correlationType = 'asset_correlation';
   527→              
   528→              // Attach asset context
   529→              event.assetContext = clients.clients[0];
   530→            }
   531→          })
   532→          .catch(() => {})
   533→      );
   534→    }
   535→
   536→    // Vulnerability correlation
   537→    if (this.vulnClient && event.host) {
   538→      searchPromises.push(
   539→        this.vulnClient.getUnifiedVulnerabilities({ host: event.host, limit: 10 })
   540→          .then(vulns => {
   541→            vulns.vulnerabilities.filter(v => v.status === 'open').forEach(vuln => {
   542→              sourceEvents.push({
   543→                source: 'vuln',
   544→                originalId: vuln.vulnId,
   545→                summary: `${vuln.severity}: ${vuln.title}`,
   546→                confidence: vuln.cvssScore > 7 ? 90 : 70,
   547→              });
   548→            });
   549→            if (vulns.vulnerabilities.some(v => v.status === 'open')) {
   550→              correlationScore += 20;
   551→            }
   552→          })
   553→          .catch(() => {})
   554→      );
   555→    }
   556→
   557→    // Execute all searches in parallel
   558→    await Promise.allSettled(searchPromises);
   559→
   560→    // Create correlated event
   561→    const correlatedEvent: CorrelatedEvent = {
   562→      eventId: `corr-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
   563→      timestamp: new Date().toISOString(),
   564→      severity: this.calculateCorrelatedSeverity(event, sourceEvents),
   565→      title: this.generateCorrelationTitle(event, sourceEvents),
   566→      description: this.generateCorrelationDescription(event, sourceEvents),
   567→      sourceEvents,
   568→      correlationScore: Math.min(100, correlationScore),
   569→      correlationType,
   570→      ...event,
   571→    };
   572→
   573→    // Generate recommendations based on findings
   574→    correlatedEvent.recommendedActions = this.generateRecommendations(correlatedEvent);
   575→
   576→    this.stats.correlationsPerformed++;
   577→    this.emit('correlationComplete', correlatedEvent);
   578→
   579→    return correlatedEvent;
   580→  }

   611→  /**
   612→   * Execute automated response actions
   613→   */
   614→  private async executeAutomatedResponse(event: CorrelatedEvent): Promise<AutomatedResponseResult> {
   615→    const action = this.determineBestAction(event);
   616→    
   617→    try {
   618→      let result: any;
   619→
   620→      switch (action.type) {
   621→        case 'isolate_endpoint':
   622→          if (this.edrClient && event.assetContext?.clientId) {
   623→            result = await this.edrClient.isolateEndpoint(
   624→              event.assetContext.clientId,
   625→              `Automated isolation due to ${event.title}`,
   626→              'SOC Platform'
   627→            );
   628→          }
   629→          break;
   630→
   631→        case 'block_ip':
   632→          // Would integrate with firewall/network device API
   633→          result = { success: true, blockedIp: action.target };
   634→          break;
   635→
   636→        case 'create_incident':
   637→          if (this.soarClient) {
   638→            result = await this.soarClient.createCase({
   639→              title: `[AUTO] ${event.title}`,
   640→              description: event.description,
   641→              severity: event.severity.toUpperCase(),
   642→              tags: ['auto-generated', 'automated-response'],
   643→              tlp: 3, // AMBER by default for auto-generated
   644→            });
   645→          }
   646→          break;
   647→
   648→        case 'alert_analyst':
   649→          // Just emit alert, no automatic action
   650→          result = { success: true, alerted: true };
   651→          break;
   652→
   653→        default:
   654→          result = { success: true, action: 'none' };
   655→      }
   656→
   657→      const responseResult: AutomatedResponseResult = {
   658→        success: result.success ?? true,
   659→        action: action.type,
   660→        target: action.target || 'N/A',
   661→        result,
   662→        timestamp: new Date().toISOString(),
   663→        triggeredBy: {
   664→          source: event.sourceEvents[0]?.source || 'unknown',
   665→          eventId: event.eventId,
   666→          severity: event.severity,
   667→        },
   668→      };
   669→
   670→      this.stats.autoResponsesExecuted++;
   671→      this.emit('automatedResponse', responseResult);
   672→
   673→      return responseResult;
   674→
   675→    } catch (error) {
   676→      this.emit('error', error);
   677→      return {
   678→        success: false,
   679→        action: action.type,
   680→        target: action.target || 'N/A',
   681→        result: { error: error.message },
   682→        timestamp: new Date().toISOString(),
   683→        triggeredBy: {
   684→          source: event.sourceEvents[0]?.source || 'unknown',
   685→          eventId: event.eventId,
   686→          severity: event.severity,
   687→        },
   688→      };
   689→    }
   690→  }

   854→  private async normalizeEvent(source: string, rawEvent: any): Promise<any> {

   869→          severity: rawEvent.severity,

   875→        return {
   876→          ...baseEvent,

   891→          severity: rawEvent.rule?.level || rawEvent.severity,

  1014→  private calculateCorrelatedSeverity(event: any, sourceEvents: any[]): CorrelatedEvent['severity'] {
  1015→    // Start with base severity
  1016→    let baseSeverity = (event.severity || 'info').toLowerCase();
  1017→    
  1018→    // Upgrade severity based on correlation strength
  1019→    if (sourceEvents.length >= 3 && baseSeverity !== 'critical') {
  1020→      baseSeverity = 'high';
  1021→    } else if (sourceEvents.length === 2 && ['low', 'info'].includes(baseSeverity)) {
  1022→      baseSeverity = 'medium';
  1023→    }
  1024→
  1025→    // Check for critical indicators in source events
  1026→    const hasCriticalSource = sourceEvents.some(s => s.confidence > 85);
  1027→    if (hasCriticalSource && baseSeverity !== 'critical') {
  1028→      baseSeverity = 'high';
  1029→    }
  1030→
  1031→    return baseSeverity as CorrelatedEvent['severity'];
  1032→  }

  1098→  private determineBestAction(event: CorrelatedEvent): { type: string; target: string } {
  1099→    // Critical severity with endpoint -> isolate
  1100→    if (event.severity === 'critical' && event.assetContext?.clientId) {
  1101→      return { type: 'isolate_endpoint', target: event.assetContext.clientId };
  1102→    }
  1103→
  1104→    // Critical/high with IP -> block
  1105→    if ((event.severity === 'critical' || event.severity === 'high') && event.sourceIp) {
  1106→      return { type: 'block_ip', target: event.sourceIp };
  1107→    }
  1108→
  1109→    // High correlation score -> create incident
  1110→    if (event.correlationScore > 60) {
  1111→      return { type: 'create_incident', target: 'SOAR' };
  1112→    }
  1113→
  1114→    // Default -> just alert
  1115→    return { type: 'alert_analyst', target: 'SOC Analyst' };
  1116→  }
