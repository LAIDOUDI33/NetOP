   115→// ─── Helper Functions ──────────────────────────────────────────────────
   116→
   117→function probabilityBarColor(pct: number): string {
   118→  if (pct >= 80) return 'bg-red-500';
   119→  if (pct >= 60) return 'bg-amber-500';
   120→  if (pct >= 40) return 'bg-amber-400';
   121→  return 'bg-emerald-500';
   122→}
   123→
   124→function probabilityProgressClass(pct: number): string {
   125→  if (pct >= 80) return '[&>div]:bg-red-500';
   126→  if (pct >= 60) return '[&>div]:bg-amber-500';
   127→  if (pct >= 40) return '[&>div]:bg-amber-400';
   128→  return '[&>div]:bg-emerald-500';
   129→}
   130→
   131→function formatFaultType(ft: string): string {
   132→  return ft.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase());
   133→}
   134→
   135→function formatStatus(s: string): string {
   136→  return s.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase());
   137→}
   138→
   139→function formatDate(iso: string): string {
   140→  const d = new Date(iso);
   141→  return d.toLocaleDateString('en-US', {
   142→    month: 'short',
   143→    day: 'numeric',
   144→    hour: '2-digit',
   145→    minute: '2-digit',
   146→  });
   147→}

   205→function ChartTooltipContent({ active, payload, label }: any) {
   206→  if (!active || !payload?.length) return null;
   207→  return (
   208→    <div className="rounded-lg border bg-background px-3 py-2 shadow-md text-xs">
   209→      <p className="font-medium mb-1 capitalize">{label}</p>

   211→        <div key={idx} className="flex items-center gap-2">
