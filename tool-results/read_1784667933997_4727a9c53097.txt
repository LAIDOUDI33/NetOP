   107→
   108→function severityBadgeVariant(s: IncidentSeverity): 'default' | 'secondary' | 'destructive' | 'outline' {
   109→  if (s === 'critical' || s === 'high') return 'destructive';
   110→  if (s === 'medium') return 'secondary';
   111→  return 'default';
   112→}
   113→
   114→function statusBadgeVariant(s: IncidentStatus): 'default' | 'secondary' | 'destructive' | 'outline' {
   115→  if (s === 'open') return 'destructive';
   116→  if (s === 'investigating') return 'secondary';
   117→  if (s === 'resolved') return 'outline';
   118→  return 'secondary';
   119→}
   120→
   121→function computeMTTR(incident: Incident): string {
   122→  if (!incident.resolvedAt) return '—';
   123→  const created = new Date(incident.createdAt).getTime();
   124→  const resolved = new Date(incident.resolvedAt).getTime();
   125→  const minutes = (resolved - created) / 60000;
   126→  return `${Math.round(minutes)} min`;
   127→}
   128→
   129→function formatDate(iso: string): string {
   130→  const d = new Date(iso);
   131→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
   132→}

   204→const RADIAN = Math.PI / 180;
   205→function renderPieLabel({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) {
   206→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   207→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   208→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   209→  if (percent < 0.06) return null;
   210→  return (
   211→    <text x={x} y={y} fill="white" textAnchor="middle" dominantBaseline="central" fontSize={11} fontWeight={600}>
   212→      {`${(percent * 100).toFixed(0)}%`}
   213→    </text>
   214→  );
   215→}
