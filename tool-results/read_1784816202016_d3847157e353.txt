   120→   120  function formatCurrency(value: number): string {
   121→   121    if (value >= 1_000_000) return `$${((value ?? 0) / 1_000_000).toFixed(1)}M`;
   122→   122    if (value >= 1_000) return `$${((value ?? 0) / 1_000).toFixed(1)}K`;
   123→   123    return `$${value.toLocaleString()}`;
   124→   124  }
