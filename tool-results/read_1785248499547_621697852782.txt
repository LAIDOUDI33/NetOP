   149→model SecurityEvent {
   150→  id                String           @id @default(cuid())
   151→  eventId           String           @unique @map("event_id") // External event ID from source
   152→  
   153→  // Event Classification
   154→  eventType         String           @map("event_type") // authentication, network, endpoint, application, cloud, telecom
   155→  category          String           // intrusion, malware, policy_violation, anomaly, reconnaissance
   156→  severity          String           // critical, high, medium, low, informational
   157→  status            String           @default("new") // new, triaged, investigating, closed, false_positive
   158→  
   159→  // Source Information
   160→  sourceIp          String?          @map("source_ip")
   161→  destinationIp     String?          @map("destination_ip")
   162→  sourcePort        Int?             @map("source_port")
   163→  destinationPort   Int?             @map("destination_port")
   164→  protocol          String?
   165→  sourceHost        String?          @map("source_host")
   166→  destinationHost   String?          @map("destination_host")
   167→  
   168→  // Tool/Source Origin
   169→  toolName          String           @map("tool_name") // wazuh, suricata, zeek, splunk, etc.
   170→  toolSeverity      String?          @map("tool_severity") // Original severity from source tool
   171→  ruleId            String?          @map("rule_id") // Original rule/signature ID
   172→  ruleName          String?          @map("rule_name") // Human-readable rule name
   173→  ruleCategory      String?          @map("rule_category") // MITRE ATT&CK tactic
   174→  
   175→  // Event Details
   176→  title             String
   177→  description       String?          @db.Text
   178→  metadata          Json?            // Flexible JSON for additional data
   179→  rawLog            String?          @db.Text // Original log line
   180→  
   181→  // Telecom-Specific Fields
   182→  subscriberId      String?          @map("subscriber_id") // MSISDN reference
   183→  imsi              String?          // IMSI if available
   184→  imei              String?          // Device IMEI
   185→  cellId            String?          @map("cell_id") // Cell tower identifier
   186→  lac               String?          // Location Area Code
   187→  
   188→  // Correlation & Investigation
   189→  incidentId        String?          @map("incident_id") // Linked incident if any
   190→  iocMatched        String?          @map("ioc_matched") // Matched IOC hash/reference
   191→  correlationGroup  String?          @map("correlation_group") // Group ID for correlated events
   192→  confidence        Float?           @default(0.0) // 0.0-1.0 confidence score
   193→  
   194→  // Enrichment
   195→  geoLocation       Json?            @map("geo_location") // {country, city, lat, lon}
   196→  threatIntel       Json?            @map("threat_intel") // Enrichment from TI feeds
   197→  assetInfo         Json?            @map("asset_info") // Asset inventory enrichment
   198→  
   199→  // Processing Metadata
   200→  processedAt       DateTime         @map("processed_at") @default(now())
   201→  ingestedAt        DateTime         @map("ingested_at") @default(now()) // Original event timestamp
   202→  firstSeenAt       DateTime         @map("first_seen_at") @default(now())
   203→  lastSeenAt       DateTime         @map("last_seen_at") @default(now())
   204→  count             Int              @default(1) // Deduplication count
   205→  
   206→  // Audit
   207→  assignedTo        String?          @map("assigned_to") // Analyst user ID
   208→  assignedAt        DateTime?        @map("assigned_at")
   209→  triagedBy         String?          @map("triaged_by")
   210→  triagedAt         DateTime?        @map("triaged_at")
   211→  closedBy          String?          @map("closed_by")
   212→  closedAt         DateTime?        @map("closed_at")
   213→  closureReason     String?          @map("closure_reason")
   214→  
   215→  createdAt         DateTime         @default(now())
   216→  updatedAt         DateTime         @updatedAt
   217→
   218→  // Relations
   219→  incident          Incident?        @relation(fields: [incidentId], references: [id])
   220→  tags              SecurityEventTag[]
   221→
   222→  // Indexes for query performance
   223→  @@index([eventType])
   224→  @@index([severity])
   225→  @@index([status])
   226→  @@index([sourceIp])
   227→  @@index([destinationIp])
   228→  @@index([toolName])
   229→  @@index([ruleId])
   230→  @@index([ingestedAt]) // Primary partition column
   231→  @@index([processedAt])
   232→  @@index([incidentId])
   233→  @@index([assignedTo])
   234→  @@index([correlationGroup])
   235→  @@fulltext([title, description])
   236→  @@map("security_events")
   237→}

   315→  // Communication

   323→  mttContain        Int?             @map("mtt_contain") // Minutes to contain

   326→  // Audit

   329→  createdBy         User             @relation("IncidentCreator", fields: [createdById], references: [id])

   331→  deletedAt         DateTime?        @map("deleted_at")
   332→
   333→  // Relations

   338→  comments          Comment[]
   339→  tags              IncidentTag[]

   344→  @@index([type])
   345→  @@index([commanderId])
   346→  @@index([createdById])

  1053→model Notification {
  1054→  id                String           @id @default(cuid())
  1055→  
  1056→  // Recipient
  1057→  userId            String?          @map("user_id") // Null = system/broadcast
  1058→  channel           String           // email, sms, webhook, push, slack, teams, pagerduty
  1059→  recipientAddress  String           @map("recipient_address") // Email, phone, webhook URL
  1060→  
  1061→  // Content
  1062→  subject           String
  1063→  body              String           @db.Text
  1064→  bodyHtml          String?          @map("body_html")
  1065→  templateId        String?          @map("template_id")
  1066→  templateData      Json?            @map("template_data")
  1067→  
  1068→  // Priority & Type
  1069→  priority          String           @default("normal") // low, normal, high, urgent, critical
  1070→  category          String           // incident_assigned, incident_updated, system_alert, digest, mention, report_ready
  1071→  type              String           // transactional, marketing, system, alert
  1072→  
  1073→  // Status
  1074→  status            String           @default("pending") // pending, sent, delivered, failed, cancelled
  1075→  sendAttempts      Int              @default(0) @map("send_attempts")
  1076→  maxAttempts       Int              @default(3) @map("max_attempts")
  1077→  
  1078→  // Timing
  1079→  scheduledAt       DateTime?        @map("scheduled_at")
  1080→  sentAt            DateTime?         @map("sent_at")
  1081→  deliveredAt       DateTime?        @map("delivered_at")
  1082→  failedAt          DateTime?        @map("failed_at")
  1083→  nextRetryAt       DateTime?        @map("next_retry_at")
  1084→  
  1085→  // Error Handling
  1086→  errorCode         String?          @map("error_code")
  1087→  errorMessage      String?          @map("error_message")
  1088→  
  1089→  // Read Receipt
  1090→  readAt            DateTime?         @map("read_at")
  1091→  
  1092→  // Metadata
  1093→  relatedEntityType String?          @map("related_entity_type")
  1094→  relatedEntityId   String?          @map("related_entity_id")
  1095→  metadata          Json?
  1096→  
  1097→  createdBy         String?          @map("created_by")
  1098→  createdAt         DateTime         @default(now())
  1099→  updatedAt         DateTime         @updatedAt
  1100→
  1101→  @@index([userId])
  1102→  @@index([channel])
  1103→  @@index([status])
  1104→  @@index([priority])
  1105→  @@index([scheduledAt])
  1106→  @@index([createdAt])
  1107→  @@map("notifications")
  1108→}
