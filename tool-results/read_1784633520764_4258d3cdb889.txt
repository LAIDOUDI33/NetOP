    73→    73→function stdDev(arr: number[]) {
    74→    74→  if (arr.length === 0) return 0;
    75→    75→  const mean = arr.reduce((a, b) => a + b, 0) / arr.length;
    76→    76→  const squaredDiffs = arr.map(v => Math.pow(v - mean, 2));
    77→    77→  return Math.sqrt(squaredDiffs.reduce((a, b) => a + b, 0) / arr.length);
    78→    78→}
    79→    79→
    80→    80→function formatValue(value: number, metric: string): string {
    81→    81→  return PERCENT_METRICS.has(metric) ? `${value.toFixed(2)}%` : value.toFixed(2);
    82→    82→}
    83→    83→
    84→    84→function isBreach(value: number, metric: string, tech: string): boolean {
    85→    85→  for (const threshold of Object.values(BREACH_THRESHOLDS)) {
    86→    86→    if (threshold.metric !== metric) continue;
    87→    87→    if (threshold.tech && threshold.tech !== tech) continue;
    88→    88→    if (threshold.condition === 'lt' && value < threshold.value) return true;
    89→    89→    if (threshold.condition === 'gt' && value > threshold.value) return true;
    90→    90→  }
    91→    91→  return false;
    92→    92→}
    93→    93→
    94→    94→function countBreaches(sites: { technology: Technology; value: number }[], metric: string): number {
    95→    95→  let count = 0;
    96→    96→  for (const site of sites) {
    97→    97→    if (isBreach(site.value, metric, site.technology)) count++;
    98→    98→  }
    99→    99→  return count;
   100→   100→}

   107→   107→  }

   116→   116→  }

   387→   387→
   388→   388→    const now = new Date();
   389→   389→    const oneDayAgo = new Date(now.getTime() - 24 * 60 * 60 * 1000);
   390→   390→
   391→   391→    for (const mod of modules) {
   392→   392→      for (const action of mod.recentActions) {
   393→   393→        const actionDate = new Date(action.createdAt);
   394→   394→        if (actionDate >= oneDayAgo) {
   395→   395→          actions24h++;
   396→   396→          if (action.status === 'applied') successCount++;
   397→   397→          if (action.impactScore !== null && action.impactScore !== undefined) {
   398→   398→            totalImpact += action.impactScore;
   399→   399→            impactCount++;
   400→   400→          }
   401→   401→        }
   402→   402→      }
   403→   403→    }
   404→   404→
   405→   405→    const successRate = actions24h > 0 ? ((successCount / actions24h) * 100).toFixed(1) : '—';
   406→   406→    const avgImpact = impactCount > 0 ? (totalImpact / impactCount).toFixed(2) : '—';

   408→   408→    return { totalModules, actions24h, successRate, avgImpact };

   414→   414→      .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
   415→   415→  }, [modules]);

   435→   435→  }
   436→   436→
   437→   437→  return (
   438→   438→    <div className="space-y-4">
   439→   439→      {/* Summary Cards */}
   440→   440→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   441→   441→        <StatCard icon={Cpu} label="Total Modules" value={summaryCards.totalModules} />
