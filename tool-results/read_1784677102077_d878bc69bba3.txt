    95→function formatTime(ts: string | null | undefined): string {
    96→  if (!ts) return '—';
    97→  try {
    98→    const d = new Date(ts);
    99→    return d.toLocaleString(undefined, {
   100→      month: 'short',
   101→      day: 'numeric',
   102→      hour: '2-digit',
   103→      minute: '2-digit',
   104→    });
   105→  } catch {
   106→    return '—';
   107→  }
   108→}
   109→
   110→function formatDuration(ms: number | null): string {
   111→  if (!ms) return '—';
   112→  if (ms < 1000) return `${ms}ms`;
   113→  if (ms < 60000) return `${(ms / 1000).toFixed(1)}s`;
   114→  return `${(ms / 60000).toFixed(1)}m`;
   115→}

   117→function getTechBadgeClass(tech: string): string {
   118→  const map: Record<string, string> = {
   119→    '2G': 'bg-slate-500/10 text-slate-700 dark:text-slate-300 border-slate-500/20',
   120→    '3G': 'bg-cyan-500/10 text-cyan-700 dark:text-cyan-300 border-cyan-500/20',
   121→    '4G': 'bg-emerald-500/10 text-emerald-700 dark:text-emerald-300 border-emerald-500/20',
   122→    '5G': 'bg-amber-500/10 text-amber-700 dark:text-amber-300 border-amber-500/20',
   123→  };
   124→  return map[tech] ?? 'bg-slate-500/10 text-slate-700 dark:text-slate-300 border-slate-500/20';
   125→}

   550→                        : 'bg-slate-500/10 text-slate-700 dark:text-slate-300 border-slate-500/20'
   551→                  }`}
   552→                >
   553→                  <Activity className="h-3 w-3" />
   554→                  {kpiImpactEntries.length} KPI{kpiImpactEntries.length !== 1 ? 's' : ''}
   555→                </Badge>
   556→              </TooltipTrigger>
   557→              <TooltipContent>
   558→                <div className="space-y-1">
   559→                  {kpiImpactEntries.map(([k, v]) => (
   560→                    <p key={k} className="text-xs">
   561→                      {k}: <span className={Number(v) > 0 ? 'text-emerald-500' : Number(v) < 0 ? 'text-red-500' : ''}>{String(v)}</span>
   562→                    </p>
   563→                  ))}
   564→                </div>
   565→              </TooltipContent>
   566→            </Tooltip>
   567→          </TooltipProvider>
   568→        ) : (
   569→          <span className="text-xs text-muted-foreground">—</span>
   570→        )}
   571→      </TableCell>
   572→
   573→      {/* Rollback */}
   574→      <TableCell className="max-w-[150px]">
   575→        {execution.status === 'rolled_back' && execution.rollbackReason ? (
   576→          <TooltipProvider>
   577→            <Tooltip>
   578→              <TooltipTrigger asChild>
   579→                <Badge variant="outline" className="text-xs gap-1 bg-slate-500/10 text-slate-600 dark:text-slate-400 border-slate-500/20 cursor-default">
   580→                  <RotateCcw className="h-3 w-3" />
   581→                  {t('status.rolledBack')}
   582→                </Badge>
   583→              </TooltipTrigger>
   584→              <TooltipContent className="max-w-[250px]">
   585→                <p className="text-xs">{execution.rollbackReason}</p>
   586→              </TooltipContent>
   587→            </Tooltip>
   588→          </TooltipProvider>
   589→        ) : (
   590→          <span className="text-xs text-muted-foreground">—</span>
   591→        )}
   592→      </TableCell>
   593→    </TableRow>
   594→  );
   595→}
