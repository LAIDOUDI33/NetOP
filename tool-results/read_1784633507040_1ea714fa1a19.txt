    92→    92→function formatTime(ts: string | null | undefined): string {
    93→    93→  if (!ts) return '—';
    94→    94→  try {
    95→    95→    const d = new Date(ts);
    96→    96→    return d.toLocaleString(undefined, {
    97→    97→      month: 'short',
    98→    98→      day: 'numeric',
    99→    99→      hour: '2-digit',
   100→   100→      minute: '2-digit',
   101→   101→    });
   102→   102→  } catch {
   103→   103→    return '—';
   104→   104→  }
   105→   105→}
   106→   106→
   107→   107→function formatDuration(ms: number | null): string {
   108→   108→  if (!ms) return '—';
   109→   109→  if (ms < 1000) return `${ms}ms`;
   110→   110→  if (ms < 60000) return `${(ms / 1000).toFixed(1)}s`;
   111→   111→  return `${(ms / 60000).toFixed(1)}m`;
   112→   112→}
