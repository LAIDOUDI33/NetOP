   114→// ─── Helper Functions ──────────────────────────────────────────────────
   115→
   116→function probabilityBarColor(pct: number): string {
   117→  if (pct >= 80) return 'bg-red-500';
   118→  if (pct >= 60) return 'bg-amber-500';
   119→  if (pct >= 40) return 'bg-amber-400';
   120→  return 'bg-emerald-500';
   121→}
   122→
   123→function probabilityProgressClass(pct: number): string {
   124→  if (pct >= 80) return '[&>div]:bg-red-500';
   125→  if (pct >= 60) return '[&>div]:bg-amber-500';
   126→  if (pct >= 40) return '[&>div]:bg-amber-400';
   127→  return '[&>div]:bg-emerald-500';
   128→}
   129→
   130→function formatFaultType(ft: string): string {
   131→  return ft.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase());
   132→}
   133→
   134→function formatStatus(s: string): string {
   135→  return s.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase());
   136→}
   137→
   138→function formatDate(iso: string): string {
   139→  const d = new Date(iso);
   140→  return d.toLocaleDateString('en-US', {
   141→    month: 'short',
   142→    day: 'numeric',
   143→    hour: '2-digit',
   144→    minute: '2-digit',
   145→  });
   146→}

   204→function ChartTooltipContent({ active, payload, label }: any) {
   205→  if (!active || !payload?.length) return null;
   206→  return (
   207→    <div className="rounded-lg border bg-background px-3 py-2 shadow-md text-xs">
   208→      <p className="font-medium mb-1 capitalize">{label}</p>
   209→      {payload.map((entry: any, idx: number) => (
   210→        <div key={idx} className="flex items-center gap-2">
   211→          <div className="w-2.5 h-2.5 rounded-sm" style={{ backgroundColor: entry.color }} />
   212→          <span className="text-muted-foreground">{entry.name}:</span>

   227→  const { data, isLoading, isError } = useQuery<FaultsResponse>({
   228→    queryKey: ['faults', severityFilter, statusFilter, componentFilter],
   229→    queryFn: () => {
   230→      const params = new URLSearchParams();
   231→      if (severityFilter !== 'all') params.set('severity', severityFilter);
   232→      if (statusFilter !== 'all') params.set('status', statusFilter);
   233→      if (componentFilter !== 'all') params.set('component', componentFilter);

   236→    },
   237→    refetchInterval: 30000,
   238→  });
   239→
   240→  const predictions = data?.predictions ?? [];
   241→  const summary = data?.summary;
   242→
   243→  const severityChartData = SEVERITIES.map((s) => ({
   244→    severity: s,

   258→    heatmapData[comp]._total = 0;

   260→  for (const p of predictions) {
   261→    if (p.component in heatmapData && p.severity in heatmapData[p.component]) {
   262→      heatmapData[p.component][p.severity]++;
   263→      heatmapData[p.component]._total++;
   264→    }
   265→  }
   266→
   267→  // ─── Render: Loading State ──────────────────────────────────────────
   268→  if (isLoading) {
   269→    return (
   270→      <div className="space-y-6 p-6">
   271→        <div>
   272→          <Skeleton className="h-8 w-72 mb-2" />
   273→          <Skeleton className="h-4 w-96" />

   275→        <KpiCardsSkeleton />
   276→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">

   279→        </div>

   281→      </div>

   291→        <p className="text-sm mt-1">Please try again later.</p>
   292→      </div>
   293→    );
   294→  }
   295→
   296→  // ─── Render: Empty State ────────────────────────────────────────────

   298→    return (
   299→      <div className="flex flex-col items-center justify-center py-24 text-muted-foreground">
   300→        <Brain className="h-12 w-12 mb-4" />
   301→        <p className="text-lg font-medium">No Fault Predictions Available</p>
   302→        <p className="text-sm mt-1">
   303→          AI fault prediction data has not been generated yet.

   307→  }
   308→
   309→  // ─── Render: Main View ─────────────────────────────────────────────
   310→  return (

   312→      {/* Header */}

   315→          <Brain className="h-6 w-6 text-amber-500" />
   316→          AI Fault Prediction
   317→        </h1>

   319→          Predictive maintenance and failure forecasting

   330→              Total Predictions
