   108→
   109→function severityBadgeVariant(s: IncidentSeverity): 'default' | 'secondary' | 'destructive' | 'outline' {
   110→  if (s === 'critical' || s === 'high') return 'destructive';
   111→  if (s === 'medium') return 'secondary';
   112→  return 'default';
   113→}
   114→
   115→function statusBadgeVariant(s: IncidentStatus): 'default' | 'secondary' | 'destructive' | 'outline' {
   116→  if (s === 'open') return 'destructive';
   117→  if (s === 'investigating') return 'secondary';
   118→  if (s === 'resolved') return 'outline';
   119→  return 'secondary';
   120→}
   121→
   122→function computeMTTR(incident: Incident): string {
   123→  if (!incident.resolvedAt) return '—';
   124→  const created = new Date(incident.createdAt).getTime();
   125→  const resolved = new Date(incident.resolvedAt).getTime();
   126→  const minutes = (resolved - created) / 60000;
   127→  return `${Math.round(minutes)} min`;
   128→}
   129→
   130→function formatDate(iso: string): string {
   131→  const d = new Date(iso);
   132→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
   133→}

   205→const RADIAN = Math.PI / 180;
   206→function renderPieLabel({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) {
   207→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   208→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   209→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   210→  if (percent < 0.06) return null;
   211→  return (
   212→    <text x={x} y={y} fill="white" textAnchor="middle" dominantBaseline="central" fontSize={11} fontWeight={600}>
   213→      {`${(percent * 100).toFixed(0)}%`}
   214→    </text>
   215→  );
   216→}
