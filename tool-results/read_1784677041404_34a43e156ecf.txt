    75→function stdDev(arr: number[]) {
    76→  if (arr.length === 0) return 0;
    77→  const mean = arr.reduce((a, b) => a + b, 0) / arr.length;
    78→  const squaredDiffs = arr.map(v => Math.pow(v - mean, 2));
    79→  return Math.sqrt(squaredDiffs.reduce((a, b) => a + b, 0) / arr.length);
    80→}
    81→
    82→function formatValue(value: number, metric: string): string {
    83→  return PERCENT_METRICS.has(metric) ? `${value.toFixed(2)}%` : value.toFixed(2);
    84→}
    85→
    86→function isBreach(value: number, metric: string, tech: string): boolean {
    87→  for (const threshold of Object.values(BREACH_THRESHOLDS)) {
    88→    if (threshold.metric !== metric) continue;
    89→    if (threshold.tech && threshold.tech !== tech) continue;
    90→    if (threshold.condition === 'lt' && value < threshold.value) return true;
    91→    if (threshold.condition === 'gt' && value > threshold.value) return true;
    92→  }
    93→  return false;
    94→}
    95→
    96→function countBreaches(sites: { technology: Technology; value: number }[], metric: string): number {
    97→  let count = 0;
    98→  for (const site of sites) {
    99→    if (isBreach(site.value, metric, site.technology)) count++;
   100→  }
   101→  return count;
   102→}

   104→function formatTime(iso: string): string {
   105→  try {
   106→    return new Date(iso).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
   107→  } catch {
   108→    return iso;
   109→  }
   110→}
   111→
   112→function formatDateTime(iso: string | null): string {
   113→  if (!iso) return '—';
   114→  try {
   115→    return new Date(iso).toLocaleString([], { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' });
   116→  } catch {
   117→    return iso;
   118→  }
   119→}

   175→function KpiReportTab() {
   176→  const t = useT();
   177→  const [metric, setMetric] = useState('downloadThroughput');
   178→
   179→  const { data, isLoading } = useQuery<KpiResponse>({
   180→    queryKey: ['kpi-report', metric],
   181→    queryFn: () => fetch(`/api/kpi?technology=all&metric=${metric}`).then(r => r.json()),
   182→    refetchInterval: 60000,
   183→  });
   184→
   185→  const chartData = useMemo(() => {
   186→    if (!data) return [];
   187→    return data.timestamps.map((ts, i) => {
   188→      const point: Record<string, string | number> = { time: ts };
   189→      if (data.data) {
   190→        Object.entries(data.data).forEach(([tech, techData]) => {
   191→          point[tech] = techData.values[i];
   192→        });
   193→      }
   194→      return point;
   195→    });
   196→  }, [data]);
   197→
   198→  const rankedSites = useMemo(() => {
   199→    if (!data?.data) return [];
   200→    return Object.values(data.data)
   201→      .flatMap(d => d.sites)
   202→      .map(s => ({
   203→        ...s,
   204→        displayValue: formatValue(s.value, metric),
   205→        breached: isBreach(s.value, metric, s.technology),
   206→      }))
   207→      .sort((a, b) => b.value - a.value);
   208→  }, [data, metric]);
   209→
   210→  const summaryStats = useMemo(() => {
   211→    if (!data?.data) return { min: '0', max: '0', avg: '0', stddev: '0', breaches: 0 };
   212→    const allValues = Object.values(data.data).flatMap(d => d.values);
   213→    const allSites = Object.values(data.data).flatMap(d => d.sites);
   214→    if (allValues.length === 0) return { min: '0', max: '0', avg: '0', stddev: '0', breaches: 0 };
   215→
   216→    const min = Math.min(...allValues);
   217→    const max = Math.max(...allValues);
   218→    const avg = allValues.reduce((a, b) => a + b, 0) / allValues.length;
   219→    const sd = stdDev(allValues);
   220→    const breaches = countBreaches(allSites, metric);
   221→
   222→    return {
   223→      min: formatValue(min, metric),
   224→      max: formatValue(max, metric),
   225→      avg: formatValue(avg, metric),
   226→      stddev: formatValue(sd, metric),
   227→      breaches,
   228→    };
   229→  }, [data, metric]);
   230→
   231→  if (isLoading || !data) {
   232→    return (
   233→      <div className="space-y-4">
   234→        <div className="no-print flex flex-wrap gap-4 items-center">
   235→          <Skeleton className="h-10 w-56" />
   236→        </div>
   237→        <ChartSkeleton />
   238→        <SummarySkeleton />
   239→        <TableSkeleton />
   240→      </div>
   241→    );
   242→  }
   243→
   244→  const techs = data.technologies || [];
   245→
   246→  return (
   247→    <div className="space-y-4">
   248→      {/* Metric Filter */}
   249→      <div className="no-print flex flex-wrap gap-4 items-center">
   250→        <Select value={metric} onValueChange={setMetric}>
   251→          <SelectTrigger className="w-[280px]">
   252→            <SelectValue placeholder="Select Metric" />
   253→          </SelectTrigger>
   254→          <SelectContent>
   255→            {METRICS.map((m) => (
   256→              <SelectItem key={m.value} value={m.value}>{m.label}</SelectItem>
   257→            ))}
   258→          </SelectContent>
   259→        </Select>
   260→      </div>
   261→
   262→      {/* Combined Trend Chart */}
   263→      <Card>
   264→        <CardHeader className="pb-2">
   265→          <CardTitle className="text-base font-semibold">
   266→            {t('rpt.allTechTrend', { metric: METRICS.find(m => m.value === metric)?.label || metric })}
   267→          </CardTitle>
   268→        </CardHeader>
   269→        <CardContent className="p-4">
   270→          <div className="h-72">
   271→            <ResponsiveContainer width="100%" height="100%">
   272→              <LineChart data={chartData} margin={{ top: 5, right: 10, left: -10, bottom: 5 }}>
   273→                <CartesianGrid strokeDasharray="3 3" className="opacity-30" />
   274→                <XAxis dataKey="time" tick={{ fontSize: 11 }} />
   275→                <YAxis tick={{ fontSize: 11 }} />
   276→                <Tooltip
   277→                  contentStyle={{ borderRadius: '8px', fontSize: '12px', border: '1px solid hsl(var(--border))' }}
   278→                  labelStyle={{ fontWeight: 600 }}
   279→                />
   280→                <Legend wrapperStyle={{ fontSize: '12px' }} />
   281→                {techs.map((tech) => (
   282→                  <Line
   283→                    key={tech}
   284→                    type="monotone"
   285→                    dataKey={tech}
   286→                    stroke={TECH_COLORS[tech]}
   287→                    name={tech}
   288→                    strokeWidth={2}
   289→                    dot={false}
   290→                  />
   291→                ))}
   292→              </LineChart>
   293→            </ResponsiveContainer>
   294→          </div>
   295→        </CardContent>
   296→      </Card>
   297→
   298→      {/* Summary Statistics */}
   299→      <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-5 gap-4">
   300→        <StatCard icon={ArrowDownRight} label={t('rpt.minimum')} value={summaryStats.min} />
   301→        <StatCard icon={ArrowUpRight} label={t('rpt.maximum')} value={summaryStats.max} />
   302→        <StatCard icon={TrendingUp} label={t('rpt.average')} value={summaryStats.avg} />
   303→        <StatCard icon={TrendingDown} label={t('rpt.stdDev')} value={summaryStats.stddev} />
   304→        <StatCard icon={AlertTriangle} label={t('rpt.breachCount')} value={summaryStats.breaches} variant={summaryStats.breaches > 0 ? 'warning' : 'default'} />
   305→      </div>
   306→
   307→      {/* Site Performance Ranking */}
   308→      <Card>
   309→        <CardHeader className="pb-2">
   310→          <CardTitle className="text-base font-semibold">
   311→            {t('rpt.siteRanking', { metric: METRICS.find(m => m.value === metric)?.label || metric })}
   312→          </CardTitle>
   313→          <ExportButton data={rankedSites} filenamePrefix="reports" columns={[{ key: 'siteName', header: 'Site' }, { key: 'technology', header: 'Technology' }, { key: 'status', header: 'Status' }, { key: 'displayValue', header: 'Value' }]} />
   314→        </CardHeader>
   315→        <CardContent className="p-4">
   316→          <ScrollArea className="max-h-96">
   317→            <div className="min-w-[540px]">
   318→              <Table>
   319→                <TableHeader>
   320→                  <TableRow>
   321→                    <TableHead className="text-xs w-12">#</TableHead>
   322→                    <TableHead className="text-xs">{t('th.site')}</TableHead>
   323→                    <TableHead className="text-xs">{t('th.technology')}</TableHead>
   324→                    <TableHead className="text-xs">{t('th.status')}</TableHead>
   325→                    <TableHead className="text-xs text-right">{t('th.value')}</TableHead>
   326→                  </TableRow>
   327→                </TableHeader>
   328→                <TableBody>
   329→                  {rankedSites.map((site, i) => (
   330→                    <TableRow key={site.siteId}>
   331→                      <TableCell className="text-xs text-muted-foreground">
   332→                        {i < 3 ? (
   333→                          <Badge variant={i === 0 ? 'default' : 'secondary'} className="w-6 justify-center text-xs font-bold">
   334→                            {i + 1}
   335→                          </Badge>
   336→                        ) : (
   337→                          <span>{i + 1}</span>
   338→                        )}
   339→                      </TableCell>
   340→                      <TableCell className="text-xs font-medium">{site.siteName}</TableCell>
   341→                      <TableCell>
   342→                        <Badge
   343→                          className="text-xs"
   344→                          style={{ backgroundColor: TECH_COLORS[site.technology as Technology], color: '#fff' }}
   345→                        >
   346→                          {site.technology}
   347→                        </Badge>
   348→                      </TableCell>
   349→                      <TableCell>
   350→                        <Badge variant={STATUS_VARIANT[site.status]} className="text-xs">
   351→                          {site.status}
   352→                        </Badge>
   353→                      </TableCell>
   354→                      <TableCell
   355→                        className={`text-xs text-right font-medium ${
   356→                          site.breached ? 'text-red-600 bg-red-50 dark:bg-red-950/30 rounded px-1' : ''
   357→                        }`}

   359→                        {site.displayValue}

   362→                    </TableRow>
   363→                  ))}
   364→                </TableBody>
   365→              </Table>
   366→            </div>
   367→          </ScrollArea>
   368→        </CardContent>
   369→      </Card>
   370→    </div>
   371→  );
   372→}

   376→function SonActivityTab() {
   377→  const t = useT();
   378→  const { data, isLoading } = useQuery<SonResponse>({
   379→    queryKey: ['son-report'],
   380→    queryFn: () => fetch('/api/son').then(r => r.json()),
   381→    refetchInterval: 60000,
   382→  });
   383→
   384→  const modules = data?.modules || [];
   385→
   386→  const summaryCards = useMemo(() => {
   387→    const totalModules = modules.length;
   388→    let actions24h = 0;
   389→    let successCount = 0;
   390→    let totalImpact = 0;
   391→    let impactCount = 0;
   392→
   393→    const now = new Date();
   394→    const oneDayAgo = new Date(now.getTime() - 24 * 60 * 60 * 1000);
   395→
   396→    for (const mod of modules) {
   397→      for (const action of mod.recentActions) {
   398→        const actionDate = new Date(action.createdAt);
   399→        if (actionDate >= oneDayAgo) {
   400→          actions24h++;
   401→          if (action.status === 'applied') successCount++;
   402→          if (action.impactScore !== null && action.impactScore !== undefined) {
   403→            totalImpact += action.impactScore;
   404→            impactCount++;
   405→          }
   406→        }
   407→      }
   408→    }
   409→
   410→    const successRate = actions24h > 0 ? ((successCount / actions24h) * 100).toFixed(1) : '—';
   411→    const avgImpact = impactCount > 0 ? (totalImpact / impactCount).toFixed(2) : '—';
   412→
   413→    return { totalModules, actions24h, successRate, avgImpact };
   414→  }, [modules]);
   415→
   416→  const allActions = useMemo(() => {
   417→    return modules
   418→      .flatMap(m => m.recentActions.map(a => ({ ...a, moduleName: m.displayName, moduleTechnology: m.technology })))
   419→      .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
   420→  }, [modules]);
   421→
   422→  if (isLoading) {
   423→    return (
   424→      <div className="space-y-4">
   425→        <SummarySkeleton />
   426→        <TableSkeleton />
   427→      </div>
   428→    );
   429→  }
   430→
   431→  if (modules.length === 0) {
   432→    return (
   433→      <Card>
   434→        <CardContent className="p-12 text-center text-muted-foreground">
   435→          <Cpu className="h-10 w-10 mx-auto mb-3 opacity-40" />
   436→          <p className="text-sm">{t('rpt.noSonModules')}</p>
   437→        </CardContent>
   438→      </Card>
   439→    );
   440→  }
   441→
   442→  return (
   443→    <div className="space-y-4">
   444→      {/* Summary Cards */}
   445→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   446→        <StatCard icon={Cpu} label={t('rpt.totalModules')} value={summaryCards.totalModules} />
   447→        <StatCard icon={Activity} label={t('rpt.actions24h')} value={summaryCards.actions24h} />
   448→        <StatCard icon={CheckCircle} label={t('rpt.successRate')} value={`${summaryCards.successRate}%`} />
   449→        <StatCard icon={TrendingUp} label={t('rpt.avgImpact')} value={summaryCards.avgImpact} />
   450→      </div>
   451→
   452→      <Separator />
   453→
   454→      {/* Recent Actions Table */}
   455→      <Card>
   456→        <CardHeader className="pb-2">
   457→          <CardTitle className="text-base font-semibold">{t('rpt.recentSonActions')}</CardTitle>
   458→        </CardHeader>
   459→        <CardContent className="p-4">
   460→          <ScrollArea className="max-h-96">
   461→            <div className="min-w-[800px]">
   462→              <Table>
   463→                <TableHeader>
   464→                  <TableRow>
   465→                    <TableHead className="text-xs">{t('rpt.time')}</TableHead>
   466→                    <TableHead className="text-xs">{t('rpt.module')}</TableHead>
   467→                    <TableHead className="text-xs">{t('th.site')}</TableHead>
   468→                    <TableHead className="text-xs">{t('rpt.actionType')}</TableHead>
   469→                    <TableHead className="text-xs">{t('th.parameter')}</TableHead>
   470→                    <TableHead className="text-xs">{t('rpt.beforeAfter')}</TableHead>
   471→                    <TableHead className="text-xs text-right">{t('rpt.impact')}</TableHead>
   472→                    <TableHead className="text-xs">{t('th.status')}</TableHead>
   473→                  </TableRow>
   474→                </TableHeader>
   475→                <TableBody>
   476→                  {allActions.length === 0 ? (
   477→                    <TableRow>
   478→                      <TableCell colSpan={8} className="text-center text-muted-foreground text-xs py-8">
   479→                        {t('rpt.noActions')}
   480→                      </TableCell>
   481→                    </TableRow>
   482→                  ) : (
   483→                    allActions.map((action) => (
   484→                      <TableRow key={action.id}>
   485→                        <TableCell className="text-xs text-muted-foreground whitespace-nowrap">
   486→                          <span className="flex items-center gap-1"><Clock className="h-3 w-3" />{formatTime(action.createdAt)}</span>

   490→                        <TableCell className="text-xs">
   491→                          <Badge variant="outline" className="text-xs font-mono">{action.actionType}</Badge>
   492→                        </TableCell>
   493→                        <TableCell className="text-xs font-mono">{action.parameter}</TableCell>
   494→                        <TableCell className="text-xs whitespace-nowrap">
   495→                          <span className="text-muted-foreground">{action.previousValue}</span>
   496→                          <span className="mx-1.5 text-muted-foreground">→</span>

   507→                        </TableCell>

   509→                          {action.status === 'applied' && (
   510→                            <Badge variant="default" className="text-xs gap-1"><CheckCircle className="h-3 w-3" />{t('status.applied')}</Badge>
   511→                          )}
   512→                          {action.status === 'pending' && (
   513→                            <Badge variant="secondary" className="text-xs gap-1"><Clock className="h-3 w-3" />{t('status.pending')}</Badge>

   520→                          )}
   521→                        </TableCell>

   537→function PolicyReportTab() {
   538→  const t = useT();
   539→  const { data, isLoading } = useQuery<PolicyResponse>({
   540→    queryKey: ['policy-report'],
   541→    queryFn: () => fetch('/api/policies').then(r => r.json()),
   542→    refetchInterval: 60000,
   543→  });
   544→
   545→  const policies = data?.policies || [];
   546→
   547→  const summaryCards = useMemo(() => {
   548→    const totalPolicies = policies.length;
   549→    const activePolicies = policies.filter(p => p.enabled).length;
   550→    const totalExecutions = policies.reduce((sum, p) => sum + p.executionStats.totalRuns, 0);
   551→    const rates = policies.filter(p => p.executionStats.totalRuns > 0).map(p => p.executionStats.successRate);
   552→    const avgSuccessRate = rates.length > 0
   553→      ? (rates.reduce((a, b) => a + b, 0) / rates.length).toFixed(1)
   554→      : '—';
   555→
   556→    return { totalPolicies, activePolicies, totalExecutions, avgSuccessRate };
   557→  }, [policies]);
   558→
   559→  if (isLoading) {
   560→    return (
   561→      <div className="space-y-4">
   562→        <SummarySkeleton />
   563→        <TableSkeleton />
   564→      </div>
   565→    );
   566→  }
   567→
   568→  if (policies.length === 0) {
   569→    return (
   570→      <Card>
   571→        <CardContent className="p-12 text-center text-muted-foreground">
   572→          <Shield className="h-10 w-10 mx-auto mb-3 opacity-40" />
   573→          <p className="text-sm">{t('rpt.noPolicies')}</p>
   574→        </CardContent>
   575→      </Card>
   576→    );
   577→  }
   578→
   579→  return (
   580→    <div className="space-y-4">
   581→      {/* Summary Cards */}
   582→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   583→        <StatCard icon={Shield} label={t('rpt.totalPolicies')} value={summaryCards.totalPolicies} />
   584→        <StatCard icon={Activity} label={t('rpt.activePolicies')} value={summaryCards.activePolicies} />
   585→        <StatCard icon={Cpu} label={t('rpt.totalExecutions')} value={summaryCards.totalExecutions} />
   586→        <StatCard icon={TrendingUp} label={t('rpt.avgSuccessRate')} value={`${summaryCards.avgSuccessRate}%`} />
   587→      </div>
   588→
   589→      <Separator />
   590→
   591→      {/* Policies Table */}
   592→      <Card>
   593→        <CardHeader className="pb-2">
   594→          <CardTitle className="text-base font-semibold">{t('rpt.policyExecSummary')}</CardTitle>
   595→        </CardHeader>
   596→        <CardContent className="p-4">
   597→          <ScrollArea className="max-h-96">
   598→            <div className="min-w-[700px]">
   599→              <Table>
   600→                <TableHeader>
   601→                  <TableRow>
   602→                    <TableHead className="text-xs">{t('th.name')}</TableHead>
   603→                    <TableHead className="text-xs">{t('th.tech')}</TableHead>
   604→                    <TableHead className="text-xs">{t('th.trigger')}</TableHead>
   605→                    <TableHead className="text-xs">{t('rpt.scope')}</TableHead>
   606→                    <TableHead className="text-xs text-center">Priority</TableHead>
   607→                    <TableHead className="text-xs text-center">Enabled</TableHead>

   609→                    <TableHead className="text-xs text-right">{t('rpt.successRate')}</TableHead>
   610→                    <TableHead className="text-xs">{t('rpt.lastRun')}</TableHead>
   611→                  </TableRow>
   612→                </TableHeader>
   613→                <TableBody>
   614→                  {policies.map((policy) => (
   615→                    <TableRow key={policy.id}>
   616→                      <TableCell className="text-xs font-medium">{policy.name}</TableCell>
   617→                      <TableCell>
   618→                        <Badge
   619→                          className="text-xs"
   620→                          style={{ backgroundColor: TECH_COLORS[policy.technology as Technology] || '#94A3B8', color: '#fff' }}
   621→                        >

   623→                        </Badge>
   624→                      </TableCell>
   625→                      <TableCell className="text-xs">

   629→                      <TableCell className="text-xs text-center">
   630→                        <Badge variant={policy.priority <= 3 ? 'destructive' : policy.priority <= 7 ? 'secondary' : 'outline'} className="text-xs">
   631→                          {policy.priority}
   632→                        </Badge>
   633→                      </TableCell>
   634→                      <TableCell className="text-xs text-center">
   635→                        {policy.enabled ? (
   636→                          <CheckCircle className="h-4 w-4 text-emerald-500 mx-auto" />

   642→                      <TableCell className="text-xs text-right font-medium">
