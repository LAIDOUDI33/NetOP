   108→function StatusPipelineIndicator({ status }: { status: OnboardingStatus }) {
   109→  const isFailed = status === 'failed';
   110→  const currentStepIdx = isFailed
   111→    ? -1
   112→    : PIPELINE_STEPS.indexOf(status as (typeof PIPELINE_STEPS)[number]);
   113→
   114→  return (
   115→    <div className="flex items-center gap-1">
   116→      {PIPELINE_STEPS.map((step, idx) => {
   117→        const isCompleted = !isFailed && idx < currentStepIdx;
   118→        const isCurrent = !isFailed && idx === currentStepIdx;
   119→        const isFuture = !isCompleted && !isCurrent;
   120→
   121→        return (
   122→          <div key={step} className="flex items-center">
   123→            <div className="relative flex items-center justify-center">
   124→              {isCurrent ? (
   125→                <span className="absolute inline-flex h-4 w-4 animate-ping rounded-full bg-emerald-400 opacity-60" />
   126→              ) : null}
   127→              <div
   128→                className={`h-3 w-3 rounded-full border-[1.5px] transition-all ${
   129→                  isCompleted
   130→                    ? 'border-emerald-500 bg-emerald-500'
   131→                    : isCurrent
   132→                      ? 'border-emerald-400 bg-emerald-400'
   133→                      : 'border-muted-foreground/30 bg-transparent'
   134→                }`}
   135→              />
   136→            </div>
   137→            {idx < PIPELINE_STEPS.length - 1 && (
   138→              <div
   139→                className={`mx-[2px] h-[1.5px] w-3 ${
   140→                  !isFailed && idx < currentStepIdx
   141→                    ? 'bg-emerald-500'
   142→                    : 'bg-muted-foreground/20'
   143→                }`}
   144→              />
   145→            )}
   146→          </div>
   147→        );
   148→      })}
   149→      {isFailed && (
   150→        <>
   151→          <div className="mx-[2px] h-[1.5px] w-3 bg-red-400" />
   152→          <div className="relative flex items-center justify-center">
   153→            <span className="absolute inline-flex h-4 w-4 animate-ping rounded-full bg-red-400 opacity-50" />
   154→            <div className="h-3 w-3 rounded-full border-[1.5px] border-red-500 bg-red-500" />
   155→          </div>
   156→        </>
   157→      )}
   158→    </div>
   159→  );
   160→}
