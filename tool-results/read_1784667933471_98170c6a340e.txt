    99→function healthScoreColor(score: number): string {
   100→  if (score >= 80) return 'text-emerald-600 dark:text-emerald-400';
   101→  if (score >= 60) return 'text-amber-600 dark:text-amber-400';
   102→  return 'text-red-600 dark:text-red-400';
   103→}
   104→
   105→function healthBarColor(score: number): string {
   106→  if (score >= 80) return 'bg-emerald-500';
   107→  if (score >= 60) return 'bg-amber-500';
   108→  return 'bg-red-500';

   163→function ChartTooltipContent({ active, payload, label }: any) {
   164→  if (!active || !payload?.length) return null;
   165→  return (
   166→    <div className="rounded-lg border bg-background px-3 py-2 shadow-md text-xs">
   167→      <p className="font-medium mb-1">{label}</p>
   168→      {payload.map((entry: any, idx: number) => (
   169→        <div key={idx} className="flex items-center gap-2">
   170→          <div className="w-2.5 h-2.5 rounded-sm" style={{ backgroundColor: entry.color }} />
   171→          <span className="text-muted-foreground">{entry.name}:</span>
   172→          <span className="font-medium">{formatNumber(entry.value)}</span>
   173→        </div>
   174→      ))}
   175→    </div>
   176→  );

   179→// ─── Pie Label ─────────────────────────────────────────────────────────
   180→
   181→const RADIAN = Math.PI / 180;
   182→
   183→function renderPieLabel({ cx, cy, midAngle, innerRadius, outerRadius, percent, name }: any) {
   184→  const radius = innerRadius + (outerRadius - innerRadius) * 1.6;
   185→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   186→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   187→  return (
   188→    <text x={x} y={y} textAnchor={x > cx ? 'start' : 'end'} className="fill-muted-foreground text-xs">
   189→      {name} ({(percent * 100).toFixed(0)}%)
   190→    </text>
   191→  );
   192→}
