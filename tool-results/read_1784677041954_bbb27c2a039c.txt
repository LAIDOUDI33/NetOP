    39→// ─── Helper Functions ──────────────────────────────────────────────────
    40→
    41→function formatCompact(n: number): string {
    42→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
    43→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
    44→  return n.toLocaleString();
    45→}
    46→
    47→function healthColor(score: number): string {
    48→  if (score >= 80) return 'text-emerald-600 dark:text-emerald-400';
    49→  if (score >= 60) return 'text-amber-600 dark:text-amber-400';
    50→  return 'text-red-600 dark:text-red-400';
    51→}
    52→
    53→function healthBg(score: number): string {
    54→  if (score >= 80) return 'bg-emerald-500';
    55→  if (score >= 60) return 'bg-amber-500';
    56→  return 'bg-red-500';
    57→}
    58→
    59→function mosColor(score: number): string {
    60→  if (score >= 4) return 'text-emerald-600 dark:text-emerald-400';
    61→  if (score >= 3) return 'text-amber-600 dark:text-amber-400';
    62→  return 'text-red-600 dark:text-red-400';
    63→}
    64→
    65→// ─── Custom Pie Tooltip ────────────────────────────────────────────────
    66→
    67→function PieTooltipContent({ active, payload }: any) {
    68→  if (!active || !payload?.length) return null;
    69→  return (
    70→    <div className="rounded-lg border bg-background px-3 py-2 shadow-md text-xs">
    71→      {payload.map((entry: any, idx: number) => (
    72→        <div key={idx} className="flex items-center gap-2">
    73→          <div className="w-2.5 h-2.5 rounded-sm" style={{ backgroundColor: entry.color }} />
    74→          <span className="text-muted-foreground">{entry.name}:</span>
    75→          <span className="font-medium">{entry.value}</span>
    76→        </div>
    77→      ))}
    78→    </div>
    79→  );
    80→}
    81→
    82→// ─── Star Rating Component ─────────────────────────────────────────────
    83→
    84→function StarRating({ score, max = 5 }: { score: number; max?: number }) {
    85→  const fullStars = Math.floor(score);
    86→  const hasHalf = score - fullStars >= 0.3;
    87→  const emptyStars = max - fullStars - (hasHalf ? 1 : 0);
    88→  return (
    89→    <div className="flex items-center gap-0.5">
    90→      {Array.from({ length: fullStars }).map((_, i) => (
    91→        <Star key={`f${i}`} className="h-4 w-4 fill-amber-400 text-amber-400" />
    92→      ))}
    93→      {hasHalf && (
    94→        <div className="relative h-4 w-4">
    95→          <Star className="absolute inset-0 h-4 w-4 text-muted-foreground/30" />
    96→          <div className="absolute inset-0 overflow-hidden w-[50%]">
    97→            <Star className="h-4 w-4 fill-amber-400 text-amber-400" />
    98→          </div>
    99→        </div>
   100→      )}
   101→      {Array.from({ length: emptyStars }).map((_, i) => (
   102→        <Star key={`e${i}`} className="h-4 w-4 text-muted-foreground/30" />
   103→      ))}
   104→    </div>
   105→  );
   106→}
