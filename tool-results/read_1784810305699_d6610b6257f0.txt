   100→// ─── Helper Functions ──────────────────────────────────────────────────
   101→
   102→function mosColor(score: number | null): string {
   103→  if (score == null) return 'text-muted-foreground';
   104→  if (score >= 4) return 'text-emerald-600 dark:text-emerald-400';
   105→  if (score >= 3) return 'text-amber-600 dark:text-amber-400';
   106→  return 'text-red-600 dark:text-red-400';
   107→}
   108→
   109→function satisfactionColor(index: number | null): string {
   110→  if (index == null) return 'text-muted-foreground';
   111→  if (index >= 70) return 'text-emerald-600 dark:text-emerald-400';
   112→  if (index >= 50) return 'text-amber-600 dark:text-amber-400';
   113→  return 'text-red-600 dark:text-red-400';
   114→}
   115→
   116→function satisfactionBg(index: number | null): string {
   117→  if (index == null) return 'bg-muted';
   118→  if (index >= 70) return 'bg-emerald-500';
   119→  if (index >= 50) return 'bg-amber-500';
   120→  return 'bg-red-500';
   121→}
   122→
   123→function computeOverallAvgMos(sites: QoESiteSummary[]): number {
   124→  const valid = sites.filter((s) => s.mosScore != null);
   125→  if (valid.length === 0) return 0;
   126→  return Number((valid.reduce((sum, s) => sum + (s.mosScore ?? 0), 0) / valid.length).toFixed(2));
   127→}
   128→
   129→function computeOverallAvgSatisfaction(sites: QoESiteSummary[]): number {
   130→  const valid = sites.filter((s) => s.satisfactionIndex != null);
   131→  if (valid.length === 0) return 0;
   132→  return Number((valid.reduce((sum, s) => sum + (s.satisfactionIndex ?? 0), 0) / valid.length).toFixed(1));
   133→}
   134→
   135→function formatBucketTime(bucket: string): string {
   136→  const d = new Date(bucket);
   137→  return `${String(d.getHours()).padStart(2, '0')}:00`;
   138→}
