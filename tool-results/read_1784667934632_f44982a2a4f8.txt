    74→function stdDev(arr: number[]) {
    75→  if (arr.length === 0) return 0;
    76→  const mean = arr.reduce((a, b) => a + b, 0) / arr.length;
    77→  const squaredDiffs = arr.map(v => Math.pow(v - mean, 2));
    78→  return Math.sqrt(squaredDiffs.reduce((a, b) => a + b, 0) / arr.length);
    79→}
    80→
    81→function formatValue(value: number, metric: string): string {
    82→  return PERCENT_METRICS.has(metric) ? `${value.toFixed(2)}%` : value.toFixed(2);
    83→}
    84→
    85→function isBreach(value: number, metric: string, tech: string): boolean {
    86→  for (const threshold of Object.values(BREACH_THRESHOLDS)) {
    87→    if (threshold.metric !== metric) continue;
    88→    if (threshold.tech && threshold.tech !== tech) continue;
    89→    if (threshold.condition === 'lt' && value < threshold.value) return true;
    90→    if (threshold.condition === 'gt' && value > threshold.value) return true;
    91→  }
    92→  return false;
    93→}
    94→
    95→function countBreaches(sites: { technology: Technology; value: number }[], metric: string): number {
    96→  let count = 0;
    97→  for (const site of sites) {
    98→    if (isBreach(site.value, metric, site.technology)) count++;
    99→  }
   100→  return count;
   101→}

   103→function formatTime(iso: string): string {
   104→  try {
   105→    return new Date(iso).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
   106→  } catch {
   107→    return iso;
   108→  }
   109→}
   110→
   111→function formatDateTime(iso: string | null): string {
   112→  if (!iso) return '—';
   113→  try {
   114→    return new Date(iso).toLocaleString([], { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' });
   115→  } catch {
   116→    return iso;
   117→  }
   118→}

   174→function KpiReportTab() {
   175→  const t = useT();
   176→  const [metric, setMetric] = useState('downloadThroughput');
   177→
   178→  const { data, isLoading } = useQuery<KpiResponse>({
   179→    queryKey: ['kpi-report', metric],
   180→    queryFn: () => fetch(`/api/kpi?technology=all&metric=${metric}`).then(r => r.json()),
   181→    refetchInterval: 60000,
   182→  });
   183→
   184→  const chartData = useMemo(() => {
   185→    if (!data) return [];
   186→    return data.timestamps.map((ts, i) => {
   187→      const point: Record<string, string | number> = { time: ts };
   188→      if (data.data) {
   189→        Object.entries(data.data).forEach(([tech, techData]) => {
   190→          point[tech] = techData.values[i];
   191→        });
   192→      }
   193→      return point;
   194→    });
   195→  }, [data]);
   196→
   197→  const rankedSites = useMemo(() => {
   198→    if (!data?.data) return [];
   199→    return Object.values(data.data)
   200→      .flatMap(d => d.sites)
   201→      .map(s => ({
   202→        ...s,
   203→        displayValue: formatValue(s.value, metric),
   204→        breached: isBreach(s.value, metric, s.technology),
   205→      }))
   206→      .sort((a, b) => b.value - a.value);
   207→  }, [data, metric]);
   208→
   209→  const summaryStats = useMemo(() => {
   210→    if (!data?.data) return { min: '0', max: '0', avg: '0', stddev: '0', breaches: 0 };
   211→    const allValues = Object.values(data.data).flatMap(d => d.values);
   212→    const allSites = Object.values(data.data).flatMap(d => d.sites);
   213→    if (allValues.length === 0) return { min: '0', max: '0', avg: '0', stddev: '0', breaches: 0 };
   214→
   215→    const min = Math.min(...allValues);
   216→    const max = Math.max(...allValues);
   217→    const avg = allValues.reduce((a, b) => a + b, 0) / allValues.length;
   218→    const sd = stdDev(allValues);
   219→    const breaches = countBreaches(allSites, metric);
   220→
   221→    return {
   222→      min: formatValue(min, metric),
   223→      max: formatValue(max, metric),
   224→      avg: formatValue(avg, metric),
   225→      stddev: formatValue(sd, metric),
   226→      breaches,
   227→    };
   228→  }, [data, metric]);
   229→
   230→  if (isLoading || !data) {
   231→    return (
   232→      <div className="space-y-4">
   233→        <div className="no-print flex flex-wrap gap-4 items-center">
   234→          <Skeleton className="h-10 w-56" />
   235→        </div>
   236→        <ChartSkeleton />
   237→        <SummarySkeleton />
   238→        <TableSkeleton />
   239→      </div>
   240→    );
   241→  }
   242→
   243→  const techs = data.technologies || [];
   244→
   245→  return (
   246→    <div className="space-y-4">
   247→      {/* Metric Filter */}
   248→      <div className="no-print flex flex-wrap gap-4 items-center">
   249→        <Select value={metric} onValueChange={setMetric}>
   250→          <SelectTrigger className="w-[280px]">
   251→            <SelectValue placeholder="Select Metric" />
   252→          </SelectTrigger>
   253→          <SelectContent>
   254→            {METRICS.map((m) => (
   255→              <SelectItem key={m.value} value={m.value}>{m.label}</SelectItem>
   256→            ))}
   257→          </SelectContent>
   258→        </Select>
   259→      </div>
   260→
   261→      {/* Combined Trend Chart */}
   262→      <Card>
   263→        <CardHeader className="pb-2">
   264→          <CardTitle className="text-base font-semibold">
   265→            {t('rpt.allTechTrend', { metric: METRICS.find(m => m.value === metric)?.label || metric })}
   266→          </CardTitle>
   267→        </CardHeader>
   268→        <CardContent className="p-4">
   269→          <div className="h-72">
   270→            <ResponsiveContainer width="100%" height="100%">
   271→              <LineChart data={chartData} margin={{ top: 5, right: 10, left: -10, bottom: 5 }}>
   272→                <CartesianGrid strokeDasharray="3 3" className="opacity-30" />
   273→                <XAxis dataKey="time" tick={{ fontSize: 11 }} />
   274→                <YAxis tick={{ fontSize: 11 }} />
   275→                <Tooltip
   276→                  contentStyle={{ borderRadius: '8px', fontSize: '12px', border: '1px solid hsl(var(--border))' }}
   277→                  labelStyle={{ fontWeight: 600 }}
   278→                />
   279→                <Legend wrapperStyle={{ fontSize: '12px' }} />
   280→                {techs.map((tech) => (
   281→                  <Line
   282→                    key={tech}
   283→                    type="monotone"
   284→                    dataKey={tech}
   285→                    stroke={TECH_COLORS[tech]}
   286→                    name={tech}
   287→                    strokeWidth={2}
   288→                    dot={false}
   289→                  />
   290→                ))}
   291→              </LineChart>
   292→            </ResponsiveContainer>
   293→          </div>
   294→        </CardContent>
   295→      </Card>
   296→
   297→      {/* Summary Statistics */}
   298→      <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-5 gap-4">
   299→        <StatCard icon={ArrowDownRight} label={t('rpt.minimum')} value={summaryStats.min} />
   300→        <StatCard icon={ArrowUpRight} label={t('rpt.maximum')} value={summaryStats.max} />
   301→        <StatCard icon={TrendingUp} label={t('rpt.average')} value={summaryStats.avg} />
   302→        <StatCard icon={TrendingDown} label={t('rpt.stdDev')} value={summaryStats.stddev} />
   303→        <StatCard icon={AlertTriangle} label={t('rpt.breachCount')} value={summaryStats.breaches} variant={summaryStats.breaches > 0 ? 'warning' : 'default'} />
   304→      </div>
   305→
   306→      {/* Site Performance Ranking */}
   307→      <Card>
   308→        <CardHeader className="pb-2">
   309→          <CardTitle className="text-base font-semibold">
   310→            {t('rpt.siteRanking', { metric: METRICS.find(m => m.value === metric)?.label || metric })}
   311→          </CardTitle>
   312→        </CardHeader>
   313→        <CardContent className="p-4">
   314→          <ScrollArea className="max-h-96">
   315→            <div className="min-w-[540px]">
   316→              <Table>
   317→                <TableHeader>
   318→                  <TableRow>
   319→                    <TableHead className="text-xs w-12">#</TableHead>
   320→                    <TableHead className="text-xs">{t('th.site')}</TableHead>
   321→                    <TableHead className="text-xs">{t('th.technology')}</TableHead>
   322→                    <TableHead className="text-xs">{t('th.status')}</TableHead>
   323→                    <TableHead className="text-xs text-right">{t('th.value')}</TableHead>
   324→                  </TableRow>
   325→                </TableHeader>
   326→                <TableBody>
   327→                  {rankedSites.map((site, i) => (
   328→                    <TableRow key={site.siteId}>
   329→                      <TableCell className="text-xs text-muted-foreground">
   330→                        {i < 3 ? (
   331→                          <Badge variant={i === 0 ? 'default' : 'secondary'} className="w-6 justify-center text-xs font-bold">
   332→                            {i + 1}
   333→                          </Badge>
   334→                        ) : (
   335→                          <span>{i + 1}</span>
   336→                        )}
   337→                      </TableCell>
   338→                      <TableCell className="text-xs font-medium">{site.siteName}</TableCell>
   339→                      <TableCell>
   340→                        <Badge
   341→                          className="text-xs"
   342→                          style={{ backgroundColor: TECH_COLORS[site.technology as Technology], color: '#fff' }}
   343→                        >
   344→                          {site.technology}
   345→                        </Badge>
   346→                      </TableCell>
   347→                      <TableCell>
   348→                        <Badge variant={STATUS_VARIANT[site.status]} className="text-xs">
   349→                          {site.status}
   350→                        </Badge>
   351→                      </TableCell>
   352→                      <TableCell
   353→                        className={`text-xs text-right font-medium ${
   354→                          site.breached ? 'text-red-600 bg-red-50 dark:bg-red-950/30 rounded px-1' : ''
   355→                        }`}
   356→                      >
   357→                        {site.displayValue}
   358→                        {site.breached && <AlertTriangle className="inline h-3 w-3 ml-1 text-red-500" />}
   359→                      </TableCell>
   360→                    </TableRow>
   361→                  ))}
   362→                </TableBody>
   363→              </Table>
   364→            </div>
   365→          </ScrollArea>
   366→        </CardContent>
   367→      </Card>
   368→    </div>
   369→  );
   370→}

   374→function SonActivityTab() {
   375→  const t = useT();
   376→  const { data, isLoading } = useQuery<SonResponse>({
   377→    queryKey: ['son-report'],
   378→    queryFn: () => fetch('/api/son').then(r => r.json()),
   379→    refetchInterval: 60000,
   380→  });
   381→
   382→  const modules = data?.modules || [];
   383→
   384→  const summaryCards = useMemo(() => {
   385→    const totalModules = modules.length;
   386→    let actions24h = 0;
   387→    let successCount = 0;
   388→    let totalImpact = 0;
   389→    let impactCount = 0;
   390→
   391→    const now = new Date();
   392→    const oneDayAgo = new Date(now.getTime() - 24 * 60 * 60 * 1000);
   393→
   394→    for (const mod of modules) {
   395→      for (const action of mod.recentActions) {
   396→        const actionDate = new Date(action.createdAt);
   397→        if (actionDate >= oneDayAgo) {
   398→          actions24h++;
   399→          if (action.status === 'applied') successCount++;
   400→          if (action.impactScore !== null && action.impactScore !== undefined) {
   401→            totalImpact += action.impactScore;
   402→            impactCount++;
   403→          }
   404→        }
   405→      }
   406→    }
   407→
   408→    const successRate = actions24h > 0 ? ((successCount / actions24h) * 100).toFixed(1) : '—';
   409→    const avgImpact = impactCount > 0 ? (totalImpact / impactCount).toFixed(2) : '—';
   410→
   411→    return { totalModules, actions24h, successRate, avgImpact };
   412→  }, [modules]);
   413→
   414→  const allActions = useMemo(() => {
   415→    return modules
   416→      .flatMap(m => m.recentActions.map(a => ({ ...a, moduleName: m.displayName, moduleTechnology: m.technology })))
   417→      .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
   418→  }, [modules]);
   419→
   420→  if (isLoading) {
   421→    return (
   422→      <div className="space-y-4">
   423→        <SummarySkeleton />
   424→        <TableSkeleton />
   425→      </div>
   426→    );
   427→  }
   428→
   429→  if (modules.length === 0) {
   430→    return (
   431→      <Card>
   432→        <CardContent className="p-12 text-center text-muted-foreground">
   433→          <Cpu className="h-10 w-10 mx-auto mb-3 opacity-40" />
   434→          <p className="text-sm">{t('rpt.noSonModules')}</p>
   435→        </CardContent>
   436→      </Card>
   437→    );
   438→  }
   439→
   440→  return (
   441→    <div className="space-y-4">
   442→      {/* Summary Cards */}
   443→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   444→        <StatCard icon={Cpu} label={t('rpt.totalModules')} value={summaryCards.totalModules} />
   445→        <StatCard icon={Activity} label={t('rpt.actions24h')} value={summaryCards.actions24h} />
   446→        <StatCard icon={CheckCircle} label={t('rpt.successRate')} value={`${summaryCards.successRate}%`} />
   447→        <StatCard icon={TrendingUp} label={t('rpt.avgImpact')} value={summaryCards.avgImpact} />
   448→      </div>
   449→
   450→      <Separator />
   451→
   452→      {/* Recent Actions Table */}
   453→      <Card>
   454→        <CardHeader className="pb-2">
   455→          <CardTitle className="text-base font-semibold">{t('rpt.recentSonActions')}</CardTitle>
   456→        </CardHeader>
   457→        <CardContent className="p-4">
   458→          <ScrollArea className="max-h-96">
   459→            <div className="min-w-[800px]">
   460→              <Table>
   461→                <TableHeader>
   462→                  <TableRow>
   463→                    <TableHead className="text-xs">{t('rpt.time')}</TableHead>
   464→                    <TableHead className="text-xs">{t('rpt.module')}</TableHead>
   465→                    <TableHead className="text-xs">{t('th.site')}</TableHead>
   466→                    <TableHead className="text-xs">{t('rpt.actionType')}</TableHead>
   467→                    <TableHead className="text-xs">{t('th.parameter')}</TableHead>
   468→                    <TableHead className="text-xs">{t('rpt.beforeAfter')}</TableHead>
   469→                    <TableHead className="text-xs text-right">{t('rpt.impact')}</TableHead>
   470→                    <TableHead className="text-xs">{t('th.status')}</TableHead>
   471→                  </TableRow>
   472→                </TableHeader>
   473→                <TableBody>
   474→                  {allActions.length === 0 ? (
   475→                    <TableRow>
   476→                      <TableCell colSpan={8} className="text-center text-muted-foreground text-xs py-8">
   477→                        {t('rpt.noActions')}
   478→                      </TableCell>
   479→                    </TableRow>
   480→                  ) : (
   481→                    allActions.map((action) => (
   482→                      <TableRow key={action.id}>
   483→                        <TableCell className="text-xs text-muted-foreground whitespace-nowrap">
   484→                          <span className="flex items-center gap-1"><Clock className="h-3 w-3" />{formatTime(action.createdAt)}</span>

   490→                        </TableCell>
   491→                        <TableCell className="text-xs font-mono">{action.parameter}</TableCell>

   493→                          <span className="text-muted-foreground">{action.previousValue}</span>

   505→                        </TableCell>

   507→                          {action.status === 'applied' && (
   508→                            <Badge variant="default" className="text-xs gap-1"><CheckCircle className="h-3 w-3" />{t('status.applied')}</Badge>
   509→                          )}
   510→                          {action.status === 'pending' && (
   511→                            <Badge variant="secondary" className="text-xs gap-1"><Clock className="h-3 w-3" />{t('status.pending')}</Badge>

   534→
   535→function PolicyReportTab() {
   536→  const t = useT();
   537→  const { data, isLoading } = useQuery<PolicyResponse>({
   538→    queryKey: ['policy-report'],
   539→    queryFn: () => fetch('/api/policies').then(r => r.json()),
   540→    refetchInterval: 60000,
   541→  });
   542→
   543→  const policies = data?.policies || [];
   544→
   545→  const summaryCards = useMemo(() => {
   546→    const totalPolicies = policies.length;
   547→    const activePolicies = policies.filter(p => p.enabled).length;
   548→    const totalExecutions = policies.reduce((sum, p) => sum + p.executionStats.totalRuns, 0);
   549→    const rates = policies.filter(p => p.executionStats.totalRuns > 0).map(p => p.executionStats.successRate);
   550→    const avgSuccessRate = rates.length > 0
   551→      ? (rates.reduce((a, b) => a + b, 0) / rates.length).toFixed(1)
   552→      : '—';
   553→
   554→    return { totalPolicies, activePolicies, totalExecutions, avgSuccessRate };
   555→  }, [policies]);
   556→
   557→  if (isLoading) {
   558→    return (
   559→      <div className="space-y-4">
   560→        <SummarySkeleton />
   561→        <TableSkeleton />
   562→      </div>
   563→    );
   564→  }
   565→
   566→  if (policies.length === 0) {
   567→    return (
   568→      <Card>
   569→        <CardContent className="p-12 text-center text-muted-foreground">
   570→          <Shield className="h-10 w-10 mx-auto mb-3 opacity-40" />
   571→          <p className="text-sm">{t('rpt.noPolicies')}</p>
   572→        </CardContent>
   573→      </Card>
   574→    );
   575→  }
   576→
   577→  return (
   578→    <div className="space-y-4">
   579→      {/* Summary Cards */}
   580→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   581→        <StatCard icon={Shield} label={t('rpt.totalPolicies')} value={summaryCards.totalPolicies} />
   582→        <StatCard icon={Activity} label={t('rpt.activePolicies')} value={summaryCards.activePolicies} />
   583→        <StatCard icon={Cpu} label={t('rpt.totalExecutions')} value={summaryCards.totalExecutions} />
   584→        <StatCard icon={TrendingUp} label={t('rpt.avgSuccessRate')} value={`${summaryCards.avgSuccessRate}%`} />
   585→      </div>
   586→
   587→      <Separator />
   588→
   589→      {/* Policies Table */}
   590→      <Card>
   591→        <CardHeader className="pb-2">
   592→          <CardTitle className="text-base font-semibold">{t('rpt.policyExecSummary')}</CardTitle>
   593→        </CardHeader>
   594→        <CardContent className="p-4">
   595→          <ScrollArea className="max-h-96">
   596→            <div className="min-w-[700px]">
   597→              <Table>
   598→                <TableHeader>
   599→                  <TableRow>
   600→                    <TableHead className="text-xs">{t('th.name')}</TableHead>
   601→                    <TableHead className="text-xs">{t('th.tech')}</TableHead>
   602→                    <TableHead className="text-xs">{t('th.trigger')}</TableHead>
   603→                    <TableHead className="text-xs">{t('rpt.scope')}</TableHead>
   604→                    <TableHead className="text-xs text-center">Priority</TableHead>
   605→                    <TableHead className="text-xs text-center">Enabled</TableHead>
   606→                    <TableHead className="text-xs text-right">{t('rpt.totalRuns')}</TableHead>
   607→                    <TableHead className="text-xs text-right">{t('rpt.successRate')}</TableHead>
   608→                    <TableHead className="text-xs">{t('rpt.lastRun')}</TableHead>
   609→                  </TableRow>
   610→                </TableHeader>
   611→                <TableBody>
   612→                  {policies.map((policy) => (
   613→                    <TableRow key={policy.id}>
   614→                      <TableCell className="text-xs font-medium">{policy.name}</TableCell>
   615→                      <TableCell>
   616→                        <Badge
   617→                          className="text-xs"
   618→                          style={{ backgroundColor: TECH_COLORS[policy.technology as Technology] || '#94A3B8', color: '#fff' }}
   619→                        >
   620→                          {policy.technology}
   621→                        </Badge>
   622→                      </TableCell>
   623→                      <TableCell className="text-xs">
   624→                        <Badge variant="outline" className="text-xs font-mono">{policy.triggerType}</Badge>
   625→                      </TableCell>
   626→                      <TableCell className="text-xs">{policy.scope}{policy.scopeValue ? ` (${policy.scopeValue})` : ''}</TableCell>
   627→                      <TableCell className="text-xs text-center">
   628→                        <Badge variant={policy.priority <= 3 ? 'destructive' : policy.priority <= 7 ? 'secondary' : 'outline'} className="text-xs">
   629→                          {policy.priority}
   630→                        </Badge>
   631→                      </TableCell>
   632→                      <TableCell className="text-xs text-center">
   633→                        {policy.enabled ? (
   634→                          <CheckCircle className="h-4 w-4 text-emerald-500 mx-auto" />
   635→                        ) : (
   636→                          <XCircle className="h-4 w-4 text-muted-foreground mx-auto" />
   637→                        )}
   638→                      </TableCell>
   639→                      <TableCell className="text-xs text-right font-medium">{policy.executionStats.totalRuns}</TableCell>
   640→                      <TableCell className="text-xs text-right font-medium">
   641→                        {policy.executionStats.totalRuns > 0 ? (
   642→                          <span className={policy.executionStats.successRate >= 80 ? 'text-emerald-600' : policy.executionStats.successRate >= 50 ? 'text-amber-600' : 'text-red-600'}>

   648→                        {formatDateTime(policy.executionStats.lastRun)}

   651→                  ))}
   652→                </TableBody>
   653→              </Table>
   654→            </div>
   655→          </ScrollArea>
   656→        </CardContent>
   657→      </Card>
   658→    </div>
   659→  );
