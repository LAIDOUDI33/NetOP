   105→   105→
   106→   106→function severityBadgeVariant(s: IncidentSeverity): 'default' | 'secondary' | 'destructive' | 'outline' {
   107→   107→  if (s === 'critical' || s === 'high') return 'destructive';
   108→   108→  if (s === 'medium') return 'secondary';
   109→   109→  return 'default';
   110→   110→}
   111→   111→
   112→   112→function statusBadgeVariant(s: IncidentStatus): 'default' | 'secondary' | 'destructive' | 'outline' {
   113→   113→  if (s === 'open') return 'destructive';
   114→   114→  if (s === 'investigating') return 'secondary';
   115→   115→  if (s === 'resolved') return 'outline';
   116→   116→  return 'secondary';
   117→   117→}
   118→   118→
   119→   119→function computeMTTR(incident: Incident): string {
   120→   120→  if (!incident.resolvedAt) return '—';
   121→   121→  const created = new Date(incident.createdAt).getTime();
   122→   122→  const resolved = new Date(incident.resolvedAt).getTime();
   123→   123→  const minutes = (resolved - created) / 60000;
   124→   124→  return `${Math.round(minutes)} min`;
   125→   125→}
   126→   126→
   127→   127→function formatDate(iso: string): string {
   128→   128→  const d = new Date(iso);
   129→   129→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
   130→   130→}

   202→   202→const RADIAN = Math.PI / 180;
   203→   203→function renderPieLabel({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) {
   204→   204→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   205→   205→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   206→   206→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   207→   207→  if (percent < 0.06) return null;
   208→   208→  return (
   209→   209→    <text x={x} y={y} fill="white" textAnchor="middle" dominantBaseline="central" fontSize={11} fontWeight={600}>
   210→   210→      {`${(percent * 100).toFixed(0)}%`}
   211→   211→    </text>
   212→   212→  );
   213→   213→}
