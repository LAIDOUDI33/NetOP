   106→   106→function StatusPipelineIndicator({ status }: { status: OnboardingStatus }) {
   107→   107→  const isFailed = status === 'failed';
   108→   108→  const currentStepIdx = isFailed
   109→   109→    ? -1
   110→   110→    : PIPELINE_STEPS.indexOf(status as (typeof PIPELINE_STEPS)[number]);
   111→   111→
   112→   112→  return (
   113→   113→    <div className="flex items-center gap-1">
   114→   114→      {PIPELINE_STEPS.map((step, idx) => {
   115→   115→        const isCompleted = !isFailed && idx < currentStepIdx;
   116→   116→        const isCurrent = !isFailed && idx === currentStepIdx;
   117→   117→        const isFuture = !isCompleted && !isCurrent;
   118→   118→
   119→   119→        return (
   120→   120→          <div key={step} className="flex items-center">
   121→   121→            <div className="relative flex items-center justify-center">
   122→   122→              {isCurrent ? (
   123→   123→                <span className="absolute inline-flex h-4 w-4 animate-ping rounded-full bg-emerald-400 opacity-60" />
   124→   124→              ) : null}
   125→   125→              <div
   126→   126→                className={`h-3 w-3 rounded-full border-[1.5px] transition-all ${
   127→   127→                  isCompleted
   128→   128→                    ? 'border-emerald-500 bg-emerald-500'
   129→   129→                    : isCurrent
   130→   130→                      ? 'border-emerald-400 bg-emerald-400'
   131→   131→                      : 'border-muted-foreground/30 bg-transparent'
   132→   132→                }`}
   133→   133→              />
   134→   134→            </div>
   135→   135→            {idx < PIPELINE_STEPS.length - 1 && (
   136→   136→              <div
   137→   137→                className={`mx-[2px] h-[1.5px] w-3 ${
   138→   138→                  !isFailed && idx < currentStepIdx
   139→   139→                    ? 'bg-emerald-500'
   140→   140→                    : 'bg-muted-foreground/20'
   141→   141→                }`}
   142→   142→              />
   143→   143→            )}
   144→   144→          </div>
   145→   145→        );
   146→   146→      })}
   147→   147→      {isFailed && (
   148→   148→        <>
   149→   149→          <div className="mx-[2px] h-[1.5px] w-3 bg-red-400" />
   150→   150→          <div className="relative flex items-center justify-center">
   151→   151→            <span className="absolute inline-flex h-4 w-4 animate-ping rounded-full bg-red-400 opacity-50" />
   152→   152→            <div className="h-3 w-3 rounded-full border-[1.5px] border-red-500 bg-red-500" />
   153→   153→          </div>
   154→   154→        </>
   155→   155→      )}
   156→   156→    </div>
   157→   157→  );
   158→   158→}

   238→   238→export default function OnboardingView() {
   239→   239→  const queryClient = useQueryClient();
   240→   240→
   241→   241→  // ── Data fetching ──
   242→   242→
   243→   243→  const { data, isLoading } = useQuery<OnboardingApiResponse>({
   244→   244→    queryKey: ['onboarding'],
   245→   245→    queryFn: () => fetch('/api/onboarding').then((r) => r.json()),
   246→   246→    refetchInterval: 15000,
   247→   247→  });
   248→   248→
   249→   249→  const records = data?.records ?? [];
   250→   250→  const counts = data?.countsByStatus ?? {};
   251→   251→
   252→   252→  const totalSites = records.length;
   253→   253→  const inProgress =
   254→   254→    (counts.provisioning ?? 0) +
   255→   255→    (counts.configuring ?? 0) +
   256→   256→    (counts.verifying ?? 0);
   257→   257→  const completed = counts.completed ?? 0;
   258→   258→  const failed = counts.failed ?? 0;
   259→   259→
   260→   260→  // ── Create mutation ──
   261→   261→
   262→   262→  const createMutation = useMutation({
   263→   263→    mutationFn: (values: OnboardingFormValues) =>
   264→   264→      fetch('/api/onboarding', {
   265→   265→        method: 'POST',
   266→   266→        headers: { 'Content-Type': 'application/json' },
   267→   267→        body: JSON.stringify(values),
   268→   268→      }).then((r) => {
   269→   269→        if (!r.ok) return r.json().then((e) => Promise.reject(new Error(e.error)));
   270→   270→        return r.json();
   271→   271→      }),
   272→   272→    onSuccess: () => {
   273→   273→      queryClient.invalidateQueries({ queryKey: ['onboarding'] });
   274→   274→      toast.success('Site onboarding initiated successfully');
   275→   275→      form.reset();
   276→   276→    },
   277→   277→    onError: (err: Error) => {
   278→   278→      toast.error(err.message || 'Failed to create onboarding');
   279→   279→    },
   280→   280→  });
   281→   281→
   282→   282→  // ── Advance mutation ──
   283→   283→
   284→   284→  const advanceMutation = useMutation({
   285→   285→    mutationFn: ({ onboardingId }: { onboardingId: string }) =>
   286→   286→      fetch('/api/onboarding', {
   287→   287→        method: 'PATCH',
   288→   288→        headers: { 'Content-Type': 'application/json' },
   289→   289→        body: JSON.stringify({ onboardingId, action: 'advance' }),
   290→   290→      }).then((r) => {
   291→   291→        if (!r.ok) return r.json().then((e) => Promise.reject(new Error(e.error)));
   292→   292→        return r.json();
   293→   293→      }),
   294→   294→    onSuccess: () => {
   295→   295→      queryClient.invalidateQueries({ queryKey: ['onboarding'] });
   296→   296→      toast.success('Pipeline advanced to next stage');
   297→   297→    },
   298→   298→    onError: (err: Error) => {
   299→   299→      toast.error(err.message || 'Failed to advance pipeline');
   300→   300→    },
   301→   301→  });
   302→   302→
   303→   303→  // ── Retry mutation (resets failed → pending) ──
   304→   304→
   305→   305→  const retryMutation = useMutation({
   306→   306→    mutationFn: ({ onboardingId }: { onboardingId: string }) =>
   307→   307→      fetch('/api/onboarding', {
   308→   308→        method: 'PATCH',
   309→   309→        headers: { 'Content-Type': 'application/json' },
   310→   310→        body: JSON.stringify({ onboardingId, action: 'advance', status: 'pending' }),
   311→   311→      }).then((r) => {
   312→   312→        if (!r.ok) return r.json().then((e) => Promise.reject(new Error(e.error)));
   313→   313→        return r.json();
   314→   314→      }),
   315→   315→    onSuccess: () => {
   316→   316→      queryClient.invalidateQueries({ queryKey: ['onboarding'] });
   317→   317→      toast.success('Site onboarding retry initiated');
   318→   318→    },
   319→   319→    onError: (err: Error) => {
   320→   320→      toast.error(err.message || 'Failed to retry onboarding');
   321→   321→    },
   322→   322→  });
   323→   323→
   324→   324→  // ── Form ──
   325→   325→
   326→   326→  const form = useForm<OnboardingFormValues>({
   327→   327→    resolver: zodResolver(onboardingFormSchema),
   328→   328→    defaultValues: {
   329→   329→      siteName: '',
   330→   330→      siteCode: '',
   331→   331→      technology: '',
   332→   332→      region: '',
   333→   333→      vendor: '',
   334→   334→      latitude: 0,
   335→   335→      longitude: 0,
   336→   336→      altitude: 0,
   337→   337→      frequency: '',
   338→   338→      bandwidth: 0,
   339→   339→      maxCapacity: 0,
   340→   340→    },
   341→   341→  });
   342→   342→
   343→   343→  const onSubmit = (values: OnboardingFormValues) => {
   344→   344→    createMutation.mutate(values);
   345→   345→  };
   346→   346→
   347→   347→  // ── Helpers ──
   348→   348→
   349→   349→  const formatDate = (iso: string) => {
   350→   350→    try {
   351→   351→      return new Date(iso).toLocaleDateString('en-US', {
   352→   352→        month: 'short',
   353→   353→        day: 'numeric',
   354→   354→        hour: '2-digit',
   355→   355→        minute: '2-digit',
   356→   356→      });
   357→   357→    } catch {
   358→   358→      return '—';
   359→   359→    }
   360→   360→  };
   361→   361→
   362→   362→  const isAdvanceable = (status: OnboardingStatus) =>
   363→   363→    status !== 'completed' && status !== 'failed';
   364→   364→
   365→   365→  const isRetryable = (status: OnboardingStatus) => status === 'failed';
   366→   366→
   367→   367→  // ── Render ──
   368→   368→
   369→   369→  return (
   370→   370→    <div className="space-y-6">
   371→   371→      {/* ── Header ── */}
   372→   372→      <div>
   373→   373→        <h1 className="text-2xl font-bold tracking-tight">Site Onboarding</h1>
   374→   374→        <p className="text-sm text-muted-foreground">
   375→   375→          Plug &amp; Play Zero-Touch Provisioning
   376→   376→        </p>
   377→   377→      </div>
   378→   378→
   379→   379→      {/* ── Stats Row ── */}
   380→   380→      <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   381→   381→        <StatCard
   382→   382→          icon={Server}
   383→   383→          label="Total Sites"
   384→   384→          value={totalSites}
   385→   385→          color="text-slate-700 dark:text-slate-300"
   386→   386→          bgColor="bg-slate-100 dark:bg-slate-800"
   387→   387→        />
   388→   388→        <StatCard
   389→   389→          icon={Radio}
   390→   390→          label="In Progress"
   391→   391→          value={inProgress}
   392→   392→          color="text-blue-600 dark:text-blue-400"
   393→   393→          bgColor="bg-blue-100 dark:bg-blue-900/40"
   394→   394→        />
   395→   395→        <StatCard
   396→   396→          icon={CheckCircle}
   397→   397→          label="Completed"
   398→   398→          value={completed}
   399→   399→          color="text-emerald-600 dark:text-emerald-400"

   401→   401→        />

   408→   408→        />
   409→   409→      </div>
   410→   410→
   411→   411→      {/* ── Main Content Grid ── */}

   413→   413→        {/* ── Left: New Site Form ── */}
   414→   414→        <Card className="lg:col-span-1">
