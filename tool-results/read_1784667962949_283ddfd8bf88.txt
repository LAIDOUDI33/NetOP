   101→// ─── Helper Functions ──────────────────────────────────────────────────
   102→
   103→function loadBarColor(load: number): string {
   104→  if (load >= 80) return 'bg-red-500';
   105→  if (load >= 60) return 'bg-amber-500';
   106→  return 'bg-emerald-500';
   107→}
   108→
   109→function loadTextColor(load: number): string {
   110→  if (load >= 80) return 'text-red-600 dark:text-red-400';
   111→  if (load >= 60) return 'text-amber-600 dark:text-amber-400';
   112→  return 'text-emerald-600 dark:text-emerald-400';
   113→}
   114→
   115→function loadProgressClass(load: number): string {
   116→  if (load >= 80) return '[&>div]:bg-red-500';
   117→  if (load >= 60) return '[&>div]:bg-amber-500';
   118→  return '[&>div]:bg-emerald-500';
   119→}
   120→
   121→function computeTypeStats(slices: NetworkSlice[], type: string) {
   122→  const typed = slices.filter((s) => s.sliceType === type);
   123→  if (typed.length === 0) return { count: 0, avgLoad: 0, totalUsers: 0 };
   124→  const avgLoad = Number((typed.reduce((sum, s) => sum + s.currentLoad, 0) / typed.length).toFixed(1));
   125→  const totalUsers = typed.reduce((sum, s) => sum + s.activeUsers, 0);
   126→  return { count: typed.length, avgLoad, totalUsers };
   127→}
