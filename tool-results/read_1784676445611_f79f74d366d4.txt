   248→function SidebarNav({ collapsed, onNavigate, allowedViews }: { collapsed: boolean; onNavigate: (view: ViewType) => void; allowedViews: Set<string> }) {
   249→  const { currentView, setCurrentView } = useAppStore();
   250→  const t = useT();
   251→  const groups = ['Operations', 'Analytics', 'Intelligence', 'AI Engine', 'Automation', 'System'] as const;
   252→
   253→  const renderGroup = (items: typeof NAV_ITEMS, label: string) => {
   254→    if (items.length === 0) return null;
   255→    return (
   256→      <div className="space-y-0.5">
   257→        {!collapsed && <p className="px-3 py-1.5 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">{t(NAV_GROUP_KEYS[label] ?? label)}</p>}
   258→        {items.map((item) => {
   259→          const Icon = item.icon;
   260→          const isActive = currentView === item.view;
   261→          const label = t(item.labelKey);
   262→          return (
   263→            <TooltipProvider key={item.view} delayDuration={0}>
   264→              <Tooltip>
   265→                <TooltipTrigger asChild>
   266→                  <button
   267→                    onClick={() => { setCurrentView(item.view); onNavigate(item.view); }}
   268→                    className={`flex items-center gap-3 rounded-lg px-3 py-2 text-sm font-medium transition-all w-full text-left
   269→                      ${isActive ? 'bg-primary text-primary-foreground shadow-sm' : 'text-muted-foreground hover:bg-accent hover:text-accent-foreground'}
   270→                      ${collapsed ? 'justify-center px-2' : ''}`}
   271→                    aria-current={isActive ? 'page' : undefined}
   272→                  >
   273→                    <Icon className="h-4 w-4 shrink-0" />
   274→                    {!collapsed && <span>{label}</span>}
   275→                  </button>
   276→                </TooltipTrigger>
   277→                {collapsed && <TooltipContent side="right"><p>{label}</p></TooltipContent>}
   278→              </Tooltip>
   279→            </TooltipProvider>
   280→          );
   281→        })}
   282→      </div>
   283→    );
   284→  };
   285→
   286→  return (
   287→    <nav className="flex flex-col gap-3 px-2 py-4" role="navigation" aria-label={t('app.mainNav')}>
   288→      {groups.map(g => {
   289→        const items = NAV_ITEMS.filter(n => n.group === g && allowedViews.has(n.view));
   290→        return items.length > 0 ? <div key={g}>{renderGroup(items, g)}</div> : null;
   291→      })}
   292→    </nav>
   293→  );
   294→}

   355→function LoginView() {
   356→  const [email, setEmail] = useState('');
   357→  const [password, setPassword] = useState('');
   358→  const [showPassword, setShowPassword] = useState(false);
   359→  const [error, setError] = useState('');
   360→  const [loading, setLoading] = useState(false);
   361→  const t = useT();
   362→
   363→  const handleSubmit = async (e: React.FormEvent) => {
   364→    e.preventDefault();
   365→    setError('');
   366→    setLoading(true);
   367→    try {
   368→      const result = await signIn('credentials', { email, password, redirect: false });
   369→      if (result?.error) {
   370→        setError(t('auth.invalidCredentials'));
   371→      }
   372→    } catch {
   373→      setError(t('auth.error'));
   374→    } finally {
   375→      setLoading(false);
   376→    }
   377→  };
   378→
   379→  return (
   380→    <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-slate-900 via-slate-800 to-slate-900 p-4">
   381→      <div className="absolute inset-0 opacity-5">

   383→      </div>

   390→            <h1 className="text-2xl font-bold text-white">NetOptima</h1>

   396→          <div className="space-y-2">
   397→            <label htmlFor="login-email" className="text-sm text-slate-300">{t('auth.email')}</label>
