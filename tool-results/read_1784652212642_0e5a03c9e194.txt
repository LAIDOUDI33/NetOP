   106→
   107→function severityBadgeVariant(s: IncidentSeverity): 'default' | 'secondary' | 'destructive' | 'outline' {
   108→  if (s === 'critical' || s === 'high') return 'destructive';
   109→  if (s === 'medium') return 'secondary';
   110→  return 'default';
   111→}
   112→
   113→function statusBadgeVariant(s: IncidentStatus): 'default' | 'secondary' | 'destructive' | 'outline' {
   114→  if (s === 'open') return 'destructive';
   115→  if (s === 'investigating') return 'secondary';
   116→  if (s === 'resolved') return 'outline';
   117→  return 'secondary';
   118→}
   119→
   120→function computeMTTR(incident: Incident): string {
   121→  if (!incident.resolvedAt) return '—';
   122→  const created = new Date(incident.createdAt).getTime();
   123→  const resolved = new Date(incident.resolvedAt).getTime();
   124→  const minutes = (resolved - created) / 60000;
   125→  return `${Math.round(minutes)} min`;
   126→}
   127→
   128→function formatDate(iso: string): string {
   129→  const d = new Date(iso);
   130→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
   131→}

   203→const RADIAN = Math.PI / 180;
   204→function renderPieLabel({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) {
   205→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   206→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   207→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   208→  if (percent < 0.06) return null;
   209→  return (
   210→    <text x={x} y={y} fill="white" textAnchor="middle" dominantBaseline="central" fontSize={11} fontWeight={600}>
   211→      {`${(percent * 100).toFixed(0)}%`}
   212→    </text>
   213→  );
   214→}
