   179→   179→export default function SonView() {
   180→   180→  const queryClient = useQueryClient();
   181→   181→  const selectedTechnology = useAppStore((s) => s.selectedTechnology);
   182→   182→  const [modeFilter, setModeFilter] = useState<SonModuleMode | 'ALL'>('ALL');
   183→   183→  const [activeTab, setActiveTab] = useState('actions');
   184→   184→

   187→   187→
   188→   188→  // ─── Queries ──────────────────────────────────────────────────────
   189→   189→  // Note: SON modules use compound technology (e.g. "4G,5G"), so we
   190→   190→  // fetch all modules without technology filter.
   191→   191→
   192→   192→  const {
   193→   193→    data: modulesData,
   194→   194→    isLoading: modulesLoading,
   195→   195→  } = useQuery<{ modules: SonModuleItem[] }>({
   196→   196→    queryKey: ['son-modules'],
   197→   197→    queryFn: () => fetch('/api/son').then((r) => r.json()),
   198→   198→    refetchInterval: 15000,
   199→   199→  });
   200→   200→
   201→   201→  const {
   202→   202→    data: actionsData,
   203→   203→    isLoading: actionsLoading,
   204→   204→  } = useQuery<{ actions: SonActionItem[]; pagination?: any }>({
   205→   205→    queryKey: ['son-actions'],
   206→   206→    queryFn: () =>
   207→   207→      fetch('/api/son/actions?limit=50').then((r) => r.json()),
   208→   208→    refetchInterval: 15000,
   209→   209→  });
   210→   210→
   211→   211→  const {
   212→   212→    data: neighborsData,
   213→   213→    isLoading: neighborsLoading,
   214→   214→  } = useQuery<{ neighbors: NeighborRelationItem[] }>({
   215→   215→    queryKey: ['son-neighbors'],
   216→   216→    queryFn: () =>
   217→   217→      fetch('/api/son/neighbors').then((r) => r.json()),
   218→   218→    refetchInterval: 15000,
   219→   219→  });
   220→   220→
   221→   221→  // ─── Mutations ────────────────────────────────────────────────────
   222→   222→
   223→   223→  const moduleMutation = useMutation({
   224→   224→    mutationFn: (body: {
   225→   225→      moduleId: string;
   226→   226→      action: 'toggle' | 'execute' | 'rollback';
   227→   227→    }) =>
   228→   228→      fetch('/api/son', {
   229→   229→        method: 'PATCH',
   230→   230→        headers: { 'Content-Type': 'application/json' },
   231→   231→        body: JSON.stringify(body),
   232→   232→      }).then((r) => {
   233→   233→        if (!r.ok) return r.json().then((e) => Promise.reject(e));
   234→   234→        return r.json();
   235→   235→      }),
   236→   236→    onSuccess: (_data, variables) => {
   237→   237→      const actionLabel =
   238→   238→        variables.action === 'toggle'
   239→   239→          ? 'toggled'
   240→   240→          : variables.action === 'execute'
   241→   241→          ? 'executed'
   242→   242→          : 'rolled back';
   243→   243→      toast.success(`Module ${actionLabel} successfully`);
   244→   244→      queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   245→   245→      queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   246→   246→    },
   247→   247→    onError: (error: any) => {
   248→   248→      toast.error(error.error || `Failed to ${error.action} module`);
   249→   249→    },
   250→   250→  });
   251→   251→
   252→   252→  const rollbackActionMutation = useMutation({
   253→   253→    mutationFn: (actionId: string) =>
   254→   254→      fetch('/api/son/actions', {
   255→   255→        method: 'PATCH',
   256→   256→        headers: { 'Content-Type': 'application/json' },
   257→   257→        body: JSON.stringify({ actionId, action: 'rollback' }),
   258→   258→      }).then((r) => {
   259→   259→        if (!r.ok) return r.json().then((e) => Promise.reject(e));
   260→   260→        return r.json();
   261→   261→      }),
   262→   262→    onSuccess: () => {
   263→   263→      toast.success('Action rolled back successfully');
   264→   264→      queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   265→   265→      queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   266→   266→    },
   267→   267→    onError: (error: any) => {
   268→   268→      toast.error(error.error || 'Failed to rollback action');
   269→   269→    },
   270→   270→  });
   271→   271→
   272→   272→  // ─── Derived Data ─────────────────────────────────────────────────
   273→   273→
   274→   274→  const modules = modulesData?.modules ?? [];

   276→   276→  const filteredModules = useMemo(() => {
   277→   277→    if (modeFilter === 'ALL') return modules;
   278→   278→    return modules.filter((m) => m.mode === modeFilter);
   279→   279→  }, [modules, modeFilter]);
   280→   280→
   281→   281→  const stats = useMemo(() => {

   283→   283→    const activeModules = modules.filter((m) => m.enabled).length;
   284→   284→    const totalActions = modules.reduce(
   285→   285→      (sum, m) => sum + (m.actionCount ?? 0),
   286→   286→      0
   287→   287→    );
   288→   288→    const allImpactScores = (actionsData?.actions ?? [])
   289→   289→      .map((a) => a.impactScore)
   290→   290→      .filter((s): s is number => s !== null && s !== undefined);
   291→   291→    const avgImpact =
   292→   292→      allImpactScores.length > 0
   293→   293→        ? allImpactScores.reduce((a, b) => a + b, 0) / allImpactScores.length
   294→   294→        : 0;
   295→   295→
   296→   296→    return { totalModules, activeModules, totalActions, avgImpact };
   297→   297→  }, [modules, actionsData]);
   298→   298→
   299→   299→  const actions = actionsData?.actions ?? [];
   300→   300→  const neighbors = neighborsData?.neighbors ?? [];
   301→   301→
   302→   302→  // ─── Handlers ─────────────────────────────────────────────────────
   303→   303→
   304→   304→  const handleToggleModule = (moduleId: string) => {
   305→   305→    moduleMutation.mutate({ moduleId, action: 'toggle' });
   306→   306→  };
   307→   307→
   308→   308→  const handleExecuteModule = (moduleId: string, displayName: string) => {
   309→   309→    moduleMutation.mutate(
   310→   310→      { moduleId, action: 'execute' },
   311→   311→      {
   312→   312→        onSuccess: () => {
   313→   313→          toast.success(`"${displayName}" executed successfully`);
   314→   314→          queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   315→   315→          queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   316→   316→        },
   317→   317→      }
   318→   318→    );
   319→   319→  };
   320→   320→
   321→   321→  const handleRollbackAction = (actionId: string) => {
   322→   322→    rollbackActionMutation.mutate(actionId);
   323→   323→  };
   324→   324→
   325→   325→  const handleRefresh = () => {
   326→   326→    queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   327→   327→    queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   328→   328→    queryClient.invalidateQueries({ queryKey: ['son-neighbors'] });
   329→   329→    toast.success('Data refreshed');
   330→   330→  };
   331→   331→
   332→   332→  const formatTime = (ts: string) => {
   333→   333→    try {
   334→   334→      return new Date(ts).toLocaleString(undefined, {
   335→   335→        month: 'short',
   336→   336→        day: 'numeric',
   337→   337→        hour: '2-digit',
   338→   338→        minute: '2-digit',
   339→   339→      });
   340→   340→    } catch {
   341→   341→      return ts;
   342→   342→    }
   343→   343→  };
   344→   344→
   345→   345→  // ─── Render ───────────────────────────────────────────────────────
   346→   346→
   347→   347→  return (
   348→   348→    <div className="space-y-6">
   349→   349→      {/* ─── Header ───────────────────────────────────────────────── */}
   350→   350→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   351→   351→        <div className="flex items-center gap-3">
   352→   352→          <div className="flex items-center justify-center h-10 w-10 rounded-lg bg-primary/10">
   353→   353→            <Cpu className="h-5 w-5 text-primary" />
   354→   354→          </div>
   355→   355→          <div>
   356→   356→            <h1 className="text-2xl font-bold tracking-tight">
   357→   357→              SON Automation
   358→   358→            </h1>
   359→   359→            <p className="text-sm text-muted-foreground">
   360→   360→              Self-Organizing Network module management &amp; action history
   361→   361→            </p>
   362→   362→          </div>
   363→   363→        </div>
   364→   364→        <div className="flex items-center gap-3">
   365→   365→          <div className="flex items-center gap-2">
   366→   366→            <Filter className="h-4 w-4 text-muted-foreground" />
   367→   367→            <Select
   368→   368→              value={selectedTechnology}
   369→   369→              onValueChange={(v) =>
   370→   370→                useAppStore.getState().setSelectedTechnology(v as Technology)
   371→   371→              }
   372→   372→            >
   373→   373→              <SelectTrigger size="sm" className="w-[90px]">
   374→   374→                <SelectValue />

   416→   416→      {/* ─── Stats Cards ─────────────────────────────────────────── */}
   417→   417→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   418→   418→        <StatsCard

   421→   421→          icon={<Layers className="h-5 w-5" />}

   424→   424→          loading={modulesLoading}
   425→   425→        />
   426→   426→        <StatsCard
   427→   427→          title="Active Modules"
   428→   428→          value={stats.activeModules}

   434→   434→        />

   441→   441→          loading={actionsLoading}
   442→   442→        />

   444→   444→          title="Avg Impact Score"
   445→   445→          value={stats.avgImpact}

   451→   451→        />

   454→   454→      {/* ─── Module Cards Grid ────────────────────────────────────── */}
   455→   455→      <div>
   456→   456→        <h2 className="text-lg font-semibold mb-4">Modules</h2>
   457→   457→        {modulesLoading ? (
   458→   458→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   459→   459→            {Array.from({ length: 6 }).map((_, i) => (

   463→   463→        ) : filteredModules.length === 0 ? (
   464→   464→          <Card className="py-16">
   465→   465→            <CardContent className="flex flex-col items-center justify-center text-center gap-2">
   466→   466→              <Cpu className="h-10 w-10 text-muted-foreground/50" />
   467→   467→              <p className="text-sm text-muted-foreground">
   468→   468→                No SON modules found for the selected filters.
   469→   469→              </p>
   470→   470→            </CardContent>
   471→   471→          </Card>
   472→   472→        ) : (
   473→   473→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   474→   474→            {filteredModules.map((mod) => (
   475→   475→              <ModuleCard
   476→   476→                key={mod.id}
