   252→function SidebarNav({ collapsed, onNavigate, allowedViews }: { collapsed: boolean; onNavigate: (view: ViewType) => void; allowedViews: Set<string> }) {
   253→  const { currentView, setCurrentView } = useAppStore();
   254→  const t = useT();
   255→  const groups = ['Operations', 'Analytics', 'Intelligence', 'AI Engine', 'Automation', 'System'] as const;
   256→
   257→  const renderGroup = (items: typeof NAV_ITEMS, label: string) => {
   258→    if (items.length === 0) return null;
   259→    return (
   260→      <div className="space-y-0.5">
   261→        {!collapsed && <p className="px-3 py-1.5 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">{t(NAV_GROUP_KEYS[label] ?? label)}</p>}
   262→        {items.map((item) => {
   263→          const Icon = item.icon;
   264→          const isActive = currentView === item.view;
   265→          const label = t(item.labelKey);
   266→          return (
   267→            <TooltipProvider key={item.view} delayDuration={0}>
   268→              <Tooltip>
   269→                <TooltipTrigger asChild>
   270→                  <button
   271→                    onClick={() => { setCurrentView(item.view); onNavigate(item.view); }}
   272→                    className={`flex items-center gap-3 rounded-lg px-3 py-2 text-sm font-medium transition-all w-full text-left
   273→                      ${isActive ? 'bg-primary text-primary-foreground shadow-sm' : 'text-muted-foreground hover:bg-accent hover:text-accent-foreground'}
   274→                      ${collapsed ? 'justify-center px-2' : ''}`}
   275→                    aria-current={isActive ? 'page' : undefined}
   276→                  >
   277→                    <Icon className="h-4 w-4 shrink-0" />
   278→                    {!collapsed && <span>{label}</span>}
   279→                  </button>
   280→                </TooltipTrigger>
   281→                {collapsed && <TooltipContent side="right"><p>{label}</p></TooltipContent>}
   282→              </Tooltip>
   283→            </TooltipProvider>
   284→          );
   285→        })}
   286→      </div>
   287→    );
   288→  };
   289→
   290→  return (
   291→    <nav className="flex flex-col gap-3 px-2 py-4" role="navigation" aria-label={t('app.mainNav')}>
   292→      {groups.map(g => {
   293→        const items = NAV_ITEMS.filter(n => n.group === g && allowedViews.has(n.view));
   294→        return items.length > 0 ? <div key={g}>{renderGroup(items, g)}</div> : null;
   295→      })}
   296→    </nav>
   297→  );
   298→}

   361→function LoginView() {

   363→  const [password, setPassword] = useState('');
   364→  const [showPassword, setShowPassword] = useState(false);
   365→  const [error, setError] = useState('');
   366→  const [loading, setLoading] = useState(false);

   370→    e.preventDefault();
   371→    setError('');
   372→    setLoading(true);
   373→    try {
   374→      const result = await signIn('credentials', { email, password, redirect: false });
   375→      if (result?.error) {
   376→        setError(t('auth.invalidCredentials'));
   377→      }
   378→    } catch {
   379→      setError(t('auth.error'));
   380→    } finally {
   381→      setLoading(false);
   382→    }
   383→  };
   384→
   385→  return (
   386→    <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-slate-900 via-slate-800 to-slate-900 p-4">
   387→      <div className="absolute inset-0 opacity-5">

   396→            <h1 className="text-2xl font-bold text-white">NetOptima Algérie</h1>
