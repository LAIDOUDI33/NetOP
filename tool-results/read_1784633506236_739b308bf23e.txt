   177→// ─── Component ───────────────────────────────────────────────────────
   178→
   179→export default function SonView() {
   180→  const queryClient = useQueryClient();
   181→  const selectedTechnology = useAppStore((s) => s.selectedTechnology);
   182→  const [modeFilter, setModeFilter] = useState<SonModuleMode | 'ALL'>('ALL');
   183→  const [activeTab, setActiveTab] = useState('actions');
   184→
   185→  const techParam =
   186→    selectedTechnology === 'ALL' ? undefined : selectedTechnology;
   187→
   188→  // ─── Queries ──────────────────────────────────────────────────────
   189→  // Note: SON modules use compound technology (e.g. "4G,5G"), so we
   190→  // fetch all modules without technology filter.
   191→
   192→  const {
   193→    data: modulesData,
   194→    isLoading: modulesLoading,
   195→  } = useQuery<{ modules: SonModuleItem[] }>({
   196→    queryKey: ['son-modules'],
   197→    queryFn: () => fetch('/api/son').then((r) => r.json()),
   198→    refetchInterval: 15000,
   199→  });
   200→
   201→  const {
   202→    data: actionsData,
   203→    isLoading: actionsLoading,
   204→  } = useQuery<{ actions: SonActionItem[]; pagination?: any }>({
   205→    queryKey: ['son-actions'],
   206→    queryFn: () =>
   207→      fetch('/api/son/actions?limit=50').then((r) => r.json()),
   208→    refetchInterval: 15000,
   209→  });
   210→
   211→  const {
   212→    data: neighborsData,
   213→    isLoading: neighborsLoading,
   214→  } = useQuery<{ neighbors: NeighborRelationItem[] }>({
   215→    queryKey: ['son-neighbors'],
   216→    queryFn: () =>
   217→      fetch('/api/son/neighbors').then((r) => r.json()),
   218→    refetchInterval: 15000,
   219→  });
   220→
   221→  // ─── Mutations ────────────────────────────────────────────────────
   222→
   223→  const moduleMutation = useMutation({
   224→    mutationFn: (body: {
   225→      moduleId: string;
   226→      action: 'toggle' | 'execute' | 'rollback';
   227→    }) =>
   228→      fetch('/api/son', {
   229→        method: 'PATCH',
   230→        headers: { 'Content-Type': 'application/json' },
   231→        body: JSON.stringify(body),
   232→      }).then((r) => {
   233→        if (!r.ok) return r.json().then((e) => Promise.reject(e));
   234→        return r.json();
   235→      }),
   236→    onSuccess: (_data, variables) => {
   237→      const actionLabel =
   238→        variables.action === 'toggle'
   239→          ? 'toggled'
   240→          : variables.action === 'execute'
   241→          ? 'executed'
   242→          : 'rolled back';
   243→      toast.success(`Module ${actionLabel} successfully`);
   244→      queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   245→      queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   246→    },
   247→    onError: (error: any) => {
   248→      toast.error(error.error || `Failed to ${error.action} module`);
   249→    },
   250→  });
   251→
   252→  const rollbackActionMutation = useMutation({
   253→    mutationFn: (actionId: string) =>
   254→      fetch('/api/son/actions', {
   255→        method: 'PATCH',
   256→        headers: { 'Content-Type': 'application/json' },
   257→        body: JSON.stringify({ actionId, action: 'rollback' }),
   258→      }).then((r) => {
   259→        if (!r.ok) return r.json().then((e) => Promise.reject(e));
   260→        return r.json();
   261→      }),
   262→    onSuccess: () => {
   263→      toast.success('Action rolled back successfully');
   264→      queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   265→      queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   266→    },
   267→    onError: (error: any) => {
   268→      toast.error(error.error || 'Failed to rollback action');
   269→    },
   270→  });
   271→
   272→  // ─── Derived Data ─────────────────────────────────────────────────
   273→
   274→  const modules = modulesData?.modules ?? [];
   275→
   276→  const filteredModules = useMemo(() => {
   277→    if (modeFilter === 'ALL') return modules;
   278→    return modules.filter((m) => m.mode === modeFilter);
   279→  }, [modules, modeFilter]);
   280→
   281→  const stats = useMemo(() => {
   282→    const totalModules = modules.length;
   283→    const activeModules = modules.filter((m) => m.enabled).length;
   284→    const totalActions = modules.reduce(
   285→      (sum, m) => sum + (m.actionCount ?? 0),
   286→      0
   287→    );
   288→    const allImpactScores = (actionsData?.actions ?? [])
   289→      .map((a) => a.impactScore)
   290→      .filter((s): s is number => s !== null && s !== undefined);
   291→    const avgImpact =
   292→      allImpactScores.length > 0
   293→        ? allImpactScores.reduce((a, b) => a + b, 0) / allImpactScores.length
   294→        : 0;
   295→
   296→    return { totalModules, activeModules, totalActions, avgImpact };
   297→  }, [modules, actionsData]);
   298→
   299→  const actions = actionsData?.actions ?? [];
   300→  const neighbors = neighborsData?.neighbors ?? [];
   301→
   302→  // ─── Handlers ─────────────────────────────────────────────────────
   303→
   304→  const handleToggleModule = (moduleId: string) => {
   305→    moduleMutation.mutate({ moduleId, action: 'toggle' });
   306→  };
   307→
   308→  const handleExecuteModule = (moduleId: string, displayName: string) => {
   309→    moduleMutation.mutate(
   310→      { moduleId, action: 'execute' },
   311→      {
   312→        onSuccess: () => {
   313→          toast.success(`"${displayName}" executed successfully`);
   314→          queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   315→          queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   316→        },
   317→      }
   318→    );
   319→  };
   320→
   321→  const handleRollbackAction = (actionId: string) => {
   322→    rollbackActionMutation.mutate(actionId);
   323→  };
   324→
   325→  const handleRefresh = () => {
   326→    queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   327→    queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   328→    queryClient.invalidateQueries({ queryKey: ['son-neighbors'] });
   329→    toast.success('Data refreshed');
   330→  };
   331→
   332→  const formatTime = (ts: string) => {
   333→    try {
   334→      return new Date(ts).toLocaleString(undefined, {
   335→        month: 'short',
   336→        day: 'numeric',
   337→        hour: '2-digit',
   338→        minute: '2-digit',
   339→      });
   340→    } catch {
   341→      return ts;
   342→    }
   343→  };
   344→
   345→  // ─── Render ───────────────────────────────────────────────────────
   346→
   347→  return (
   348→    <div className="space-y-6">
   349→      {/* ─── Header ───────────────────────────────────────────────── */}
   350→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   351→        <div className="flex items-center gap-3">
   352→          <div className="flex items-center justify-center h-10 w-10 rounded-lg bg-primary/10">
   353→            <Cpu className="h-5 w-5 text-primary" />
   354→          </div>
   355→          <div>
   356→            <h1 className="text-2xl font-bold tracking-tight">
   357→              SON Automation
   358→            </h1>
   359→            <p className="text-sm text-muted-foreground">
   360→              Self-Organizing Network module management &amp; action history
   361→            </p>
   362→          </div>
   363→        </div>
   364→        <div className="flex items-center gap-3">
   365→          <div className="flex items-center gap-2">
   366→            <Filter className="h-4 w-4 text-muted-foreground" />
   367→            <Select
   368→              value={selectedTechnology}
   369→              onValueChange={(v) =>
   370→                useAppStore.getState().setSelectedTechnology(v as Technology)
   371→              }
   372→            >
   373→              <SelectTrigger size="sm" className="w-[90px]">
   374→                <SelectValue />
   375→              </SelectTrigger>
   376→              <SelectContent>
   377→                {TECH_OPTIONS.map((t) => (
   378→                  <SelectItem key={t} value={t}>
   379→                    {t}
   380→                  </SelectItem>
   381→                ))}
   382→              </SelectContent>
   383→            </Select>
   384→          </div>
   385→          <Select
   386→            value={modeFilter}
   387→            onValueChange={(v) =>
   388→              setModeFilter(v as SonModuleMode | 'ALL')
   389→            }
   390→          >
   391→            <SelectTrigger size="sm" className="w-[170px]">
   392→              <SelectValue placeholder="Mode" />
   393→            </SelectTrigger>
   394→            <SelectContent>
   395→              {MODE_OPTIONS.map((m) => (
   396→                <SelectItem key={m} value={m}>
   397→                  {m === 'ALL'
   398→                    ? 'All Modes'
   399→                    : MODE_BADGE_CONFIG[m as SonModuleMode].label}
   400→                </SelectItem>
   401→              ))}
   402→            </SelectContent>
   403→          </Select>
   404→          <Button
   405→            variant="outline"
   406→            size="sm"
   407→            onClick={handleRefresh}
   408→            className="h-8"
   409→          >
   410→            <RefreshCw className="h-3.5 w-3.5 mr-1.5" />
   411→            Refresh
   412→          </Button>
   413→        </div>
   414→      </div>
   415→
   416→      {/* ─── Stats Cards ─────────────────────────────────────────── */}
   417→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   418→        <StatsCard
   419→          title="Total Modules"
   420→          value={stats.totalModules}
   421→          icon={<Layers className="h-5 w-5" />}
   422→          iconColor="text-slate-500"
   423→          iconBg="bg-slate-500/10"
   424→          loading={modulesLoading}
   425→        />
   426→        <StatsCard
   427→          title="Active Modules"
   428→          value={stats.activeModules}
   429→          subtitle={`of ${stats.totalModules} total`}
   430→          icon={<Zap className="h-5 w-5" />}
   431→          iconColor="text-emerald-500"
   432→          iconBg="bg-emerald-500/10"
   433→          loading={modulesLoading}
   434→        />
   435→        <StatsCard
   436→          title="Total Actions"
   437→          value={stats.totalActions}
   438→          icon={<Activity className="h-5 w-5" />}
   439→          iconColor="text-amber-500"
   440→          iconBg="bg-amber-500/10"
   441→          loading={actionsLoading}
   442→        />
   443→        <StatsCard
   444→          title="Avg Impact Score"
   445→          value={stats.avgImpact}
   446→          isDecimal
   447→          icon={<Gauge className="h-5 w-5" />}
   448→          iconColor="text-cyan-500"
   449→          iconBg="bg-cyan-500/10"
   450→          loading={actionsLoading}
   451→        />
   452→      </div>
   453→
   454→      {/* ─── Module Cards Grid ────────────────────────────────────── */}
   455→      <div>
   456→        <h2 className="text-lg font-semibold mb-4">Modules</h2>
   457→        {modulesLoading ? (
   458→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   459→            {Array.from({ length: 6 }).map((_, i) => (
   460→              <Skeleton key={i} className="h-72 rounded-xl" />
   461→            ))}
   462→          </div>
   463→        ) : filteredModules.length === 0 ? (
   464→          <Card className="py-16">
   465→            <CardContent className="flex flex-col items-center justify-center text-center gap-2">
   466→              <Cpu className="h-10 w-10 text-muted-foreground/50" />
   467→              <p className="text-sm text-muted-foreground">
   468→                No SON modules found for the selected filters.
   469→              </p>
   470→            </CardContent>
   471→          </Card>
   472→        ) : (
   473→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   474→            {filteredModules.map((mod) => (
   475→              <ModuleCard
   476→                key={mod.id}
   477→                module={mod}
   478→                onToggle={handleToggleModule}
   479→                onExecute={handleExecuteModule}
   480→                isMutating={moduleMutation.isPending}
   481→              />
   482→            ))}
   483→          </div>
   484→        )}
   485→      </div>
   486→
   487→      {/* ─── Tabs: Actions & Neighbors ────────────────────────────── */}
   488→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   489→        <TabsList className="grid w-full grid-cols-2 max-w-md">
   490→          <TabsTrigger value="actions" className="flex items-center gap-2">
   491→            <ArrowRightLeft className="h-3.5 w-3.5" />
   492→            Action History
   493→            {!actionsLoading && actions.length > 0 && (
   494→              <Badge
   495→                variant="secondary"
   496→                className="ml-1 h-5 px-1.5 text-[10px] font-mono"
   497→              >
   498→                {actions.length}
   499→              </Badge>
   500→            )}
   501→          </TabsTrigger>
   502→          <TabsTrigger value="neighbors" className="flex items-center gap-2">
   503→            <Network className="h-3.5 w-3.5" />
   504→            Neighbor Relations
   505→            {!neighborsLoading && neighbors.length > 0 && (
   506→              <Badge
   507→                variant="secondary"
   508→                className="ml-1 h-5 px-1.5 text-[10px] font-mono"
   509→              >
   510→                {neighbors.length}
   511→              </Badge>
   512→            )}
   513→          </TabsTrigger>
   514→        </TabsList>
   515→
   516→        {/* ── Action History Tab ──────────────────────────────────── */}
   517→        <TabsContent value="actions" className="mt-4">
   518→          <Card>
   519→            <CardHeader className="pb-3">
   520→              <CardTitle className="text-base font-semibold flex items-center gap-2">
   521→                <Activity className="h-4 w-4 text-muted-foreground" />
   522→                Recent Actions
   523→              </CardTitle>
   524→            </CardHeader>
   525→            <CardContent className="p-0">
   526→              {actionsLoading ? (
   527→                <div className="p-6 space-y-3">
   528→                  {Array.from({ length: 5 }).map((_, i) => (
   529→                    <Skeleton key={i} className="h-12 w-full" />
   530→                  ))}
   531→                </div>
   532→              ) : actions.length === 0 ? (
   533→                <div className="py-12 text-center text-sm text-muted-foreground">
   534→                  No actions recorded yet.
   535→                </div>
   536→              ) : (
   537→                <ScrollArea className="max-h-96">
   538→                  <Table>
   539→                    <TableHeader>
   540→                      <TableRow className="hover:bg-transparent">
   541→                        <TableHead className="w-[150px] text-xs">Time</TableHead>
   542→                        <TableHead className="text-xs">Module</TableHead>
   543→                        <TableHead className="text-xs">Site</TableHead>
   544→                        <TableHead className="text-xs">Action Type</TableHead>
   545→                        <TableHead className="text-xs">Parameter</TableHead>
   546→                        <TableHead className="text-xs">Change</TableHead>
   547→                        <TableHead className="text-xs">Status</TableHead>
   548→                        <TableHead className="text-xs text-right">Impact</TableHead>
   549→                        <TableHead className="text-xs text-right w-[80px]">
   550→                          Rollback
   551→                        </TableHead>
   552→                      </TableRow>
   553→                    </TableHeader>
   554→                    <TableBody>
   555→                      {actions.map((action) => {
   556→                        const statusCfg =
   557→                          ACTION_STATUS_BADGE[action.status];
   558→                        return (
   559→                          <TableRow key={action.id}>
   560→                            <TableCell className="text-xs text-muted-foreground py-3">
   561→                              {formatTime(action.createdAt)}
   562→                            </TableCell>
   563→                            <TableCell className="py-3">
   564→                              <div className="text-xs font-medium">
   565→                                {action.moduleDisplayName ||
   566→                                  action.moduleName}
   567→                              </div>
   568→                              <Badge
   569→                                className="text-[9px] px-1 py-0 mt-0.5"
   570→                                style={{
   571→                                  backgroundColor: TECH_COLORS[action.technology as Technology],
   572→                                  color: '#fff',
   573→                                }}
   574→                              >
   575→                                {action.technology}
   576→                              </Badge>
   577→                            </TableCell>
   578→                            <TableCell className="text-xs py-3">
   579→                              <div className="font-medium">
   580→                                {action.siteName || '—'}
