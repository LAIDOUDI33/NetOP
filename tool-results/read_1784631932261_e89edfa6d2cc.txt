    92→function formatTime(ts: string | null | undefined): string {
    93→  if (!ts) return '—';
    94→  try {
    95→    const d = new Date(ts);
    96→    return d.toLocaleString(undefined, {
    97→      month: 'short',
    98→      day: 'numeric',
    99→      hour: '2-digit',
   100→      minute: '2-digit',
   101→    });
   102→  } catch {
   103→    return '—';
   104→  }
   105→}
   106→
   107→function formatDuration(ms: number | null): string {
   108→  if (!ms) return '—';
   109→  if (ms < 1000) return `${ms}ms`;
   110→  if (ms < 60000) return `${(ms / 1000).toFixed(1)}s`;
   111→  return `${(ms / 60000).toFixed(1)}m`;
   112→}

   114→function getTechBadgeClass(tech: string): string {
   115→  const map: Record<string, string> = {
   116→    '2G': 'bg-slate-500/10 text-slate-700 dark:text-slate-300 border-slate-500/20',
   117→    '3G': 'bg-cyan-500/10 text-cyan-700 dark:text-cyan-300 border-cyan-500/20',
   118→    '4G': 'bg-emerald-500/10 text-emerald-700 dark:text-emerald-300 border-emerald-500/20',
   119→    '5G': 'bg-amber-500/10 text-amber-700 dark:text-amber-300 border-amber-500/20',
   120→  };
   121→  return map[tech] ?? 'bg-slate-500/10 text-slate-700 dark:text-slate-300 border-slate-500/20';
   122→}

   553→        {execution.status === 'rolled_back' && execution.rollbackReason ? (

   555→            <Tooltip>
   556→              <TooltipTrigger asChild>
   557→                <Badge variant="outline" className="text-xs gap-1 bg-slate-500/10 text-slate-600 dark:text-slate-400 border-slate-500/20 cursor-default">
   558→                  <RotateCcw className="h-3 w-3" />
   559→                  Rollback
   560→                </Badge>
   561→              </TooltipTrigger>
   562→              <TooltipContent className="max-w-[250px]">
   563→                <p className="text-xs">{execution.rollbackReason}</p>
   564→              </TooltipContent>
   565→            </Tooltip>
   566→          </TooltipProvider>
   567→        ) : (
   568→          <span className="text-xs text-muted-foreground">—</span>
   569→        )}
   570→      </TableCell>
   571→    </TableRow>
   572→  );
   573→}

   665→export default function PoliciesView() {
   666→  const queryClient = useQueryClient();
   667→  const { selectedTechnology } = useAppStore();
   668→  const [techFilter, setTechFilter] = useState<string>(selectedTechnology);
   669→
   670→  // ── Queries ──
   671→
   672→  const { data: policiesData, isLoading: policiesLoading } = useQuery<{
   673→    policies: PolicyItem[];
   674→  }>({
   675→    queryKey: ['policies', techFilter],
   676→    queryFn: () =>
   677→      fetch(`/api/policies?technology=${techFilter}`)
   678→        .then((r) => r.json()),
   679→    refetchInterval: 15000,
   680→  });
   681→
   682→  const { data: executionsData, isLoading: executionsLoading } = useQuery<{
   683→    executions: PolicyExecutionItem[];
   684→  }>({
   685→    queryKey: ['policy-executions', techFilter],
   686→    queryFn: () =>
   687→      fetch(`/api/policies/executions?technology=${techFilter}`)
   688→        .then((r) => r.json()),
   689→    refetchInterval: 15000,
   690→  });
   691→
   692→  // ── Mutations ──
   693→
   694→  const patchMutation = useMutation({
   695→    mutationFn: (body: { policyId: string; action: 'toggle' | 'trigger'; triggerReason?: string }) =>
   696→      fetch('/api/policies', {
   697→        method: 'PATCH',
   698→        headers: { 'Content-Type': 'application/json' },
   699→        body: JSON.stringify(body),
   700→      }).then((r) => r.json()),
   701→    onSuccess: (_data, variables) => {
   702→      queryClient.invalidateQueries({ queryKey: ['policies'] });
   703→      queryClient.invalidateQueries({ queryKey: ['policy-executions'] });
   704→      if (variables.action === 'toggle') {
   705→        toast.success('Policy toggled successfully');
   706→      } else {
   707→        toast.success('Policy triggered successfully');
   708→      }
   709→    },
   710→    onError: () => {
   711→      toast.error('Failed to perform action on policy');
   712→    },
   713→  });
   714→
   715→  const handleToggle = (id: string) => {
   716→    patchMutation.mutate({ policyId: id, action: 'toggle' });
   717→  };
   718→
   719→  const handleTrigger = (id: string) => {
   720→    patchMutation.mutate({
   721→      policyId: id,
   722→      action: 'trigger',
   723→      triggerReason: 'Manual trigger from UI',
   724→    });
   725→  };

   742→  const totalExecutions = filteredExecutions.length;
   743→  const avgSuccessRate =
   744→    filteredPolicies.length > 0
   745→      ? (

   747→          filteredPolicies.length
   748→        ).toFixed(1)
   749→      : '0.0';
   750→
   751→  const isMutating = patchMutation.isPending;
   752→
   753→  return (
   754→    <TooltipProvider>
   755→      <div className="space-y-6">
   756→        {/* ── Header ── */}
   757→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   758→          <div className="space-y-1">
   759→            <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   760→              <Shield className="h-6 w-6" />
   761→              Automation Policies
   762→            </h1>
   763→            <p className="text-sm text-muted-foreground">
   764→              Closed-loop policy engine for automated network optimization
   765→            </p>
   766→          </div>
   767→          <div className="flex items-center gap-3">
   768→            <Select value={techFilter} onValueChange={setTechFilter}>
   769→              <SelectTrigger className="w-[140px]">
   770→                <SelectValue placeholder="Filter by tech" />
   771→              </SelectTrigger>
   772→              <SelectContent>
   773→                <SelectItem value="all">All Technologies</SelectItem>
   774→                <SelectItem value="2G">2G</SelectItem>
   775→                <SelectItem value="3G">3G</SelectItem>
   776→                <SelectItem value="4G">4G</SelectItem>
   777→                <SelectItem value="5G">5G</SelectItem>
   778→              </SelectContent>
   779→            </Select>
   780→            <Button
   781→              variant="outline"
   782→              size="icon"
   783→              onClick={() => {
   784→                queryClient.invalidateQueries({ queryKey: ['policies'] });
   785→                queryClient.invalidateQueries({ queryKey: ['policy-executions'] });
   786→                toast.success('Data refreshed');
   787→              }}
   788→            >
   789→              <RefreshCw className="h-4 w-4" />
   790→            </Button>
   791→          </div>

   793→
   794→        {/* ── Stats Cards ── */}
   795→        {policiesLoading ? (
   796→          <StatCardsSkeleton />
   797→        ) : (
   798→          <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   799→            <StatCard
   800→              title="Total Policies"
   801→              value={totalPolicies}
   802→              icon={Shield}
   803→              color="bg-slate-500/10 text-slate-600 dark:text-slate-400"
