   179→// ─── Component ───────────────────────────────────────────────────────
   180→
   181→export default function SonView() {
   182→  const t = useT();
   183→  const queryClient = useQueryClient();
   184→  const selectedTechnology = useAppStore((s) => s.selectedTechnology);
   185→  const [modeFilter, setModeFilter] = useState<SonModuleMode | 'ALL'>('ALL');
   186→  const [activeTab, setActiveTab] = useState('actions');
   187→
   188→  const techParam =
   189→    selectedTechnology === 'ALL' ? undefined : selectedTechnology;
   190→
   191→  // ─── Queries ──────────────────────────────────────────────────────
   192→  // Note: SON modules use compound technology (e.g. "4G,5G"), so we
   193→  // fetch all modules without technology filter.
   194→
   195→  const {
   196→    data: modulesData,
   197→    isLoading: modulesLoading,
   198→  } = useQuery<{ modules: SonModuleItem[] }>({
   199→    queryKey: ['son-modules'],
   200→    queryFn: () => fetch('/api/son').then((r) => r.json()),
   201→    refetchInterval: 15000,
   202→  });
   203→
   204→  const {
   205→    data: actionsData,
   206→    isLoading: actionsLoading,
   207→  } = useQuery<{ actions: SonActionItem[]; pagination?: any }>({
   208→    queryKey: ['son-actions'],
   209→    queryFn: () =>
   210→      fetch('/api/son/actions?limit=50').then((r) => r.json()),
   211→    refetchInterval: 15000,
   212→  });
   213→
   214→  const {
   215→    data: neighborsData,
   216→    isLoading: neighborsLoading,
   217→  } = useQuery<{ neighbors: NeighborRelationItem[] }>({
   218→    queryKey: ['son-neighbors'],
   219→    queryFn: () =>
   220→      fetch('/api/son/neighbors').then((r) => r.json()),
   221→    refetchInterval: 15000,
   222→  });
   223→
   224→  // ─── Mutations ────────────────────────────────────────────────────
   225→
   226→  const moduleMutation = useMutation({
   227→    mutationFn: (body: {
   228→      moduleId: string;
   229→      action: 'toggle' | 'execute' | 'rollback';
   230→    }) =>
   231→      fetch('/api/son', {
   232→        method: 'PATCH',
   233→        headers: { 'Content-Type': 'application/json' },
   234→        body: JSON.stringify(body),
   235→      }).then((r) => {
   236→        if (!r.ok) return r.json().then((e) => Promise.reject(e));
   237→        return r.json();
   238→      }),
   239→    onSuccess: (_data, variables) => {
   240→      const actionLabel =
   241→        variables.action === 'toggle'
   242→          ? t('son.toggled')
   243→          : variables.action === 'execute'
   244→          ? t('son.executed')
   245→          : t('son.rolledBack');
   246→      toast.success(t('son.moduleToggled', { action: actionLabel }));
   247→      queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   248→      queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   249→    },
   250→    onError: (error: any) => {
   251→      toast.error(error.error || t('son.failedToAction', { action: error.action || variables.action }));
   252→    },
   253→  });
   254→
   255→  const rollbackActionMutation = useMutation({
   256→    mutationFn: (actionId: string) =>
   257→      fetch('/api/son/actions', {
   258→        method: 'PATCH',
   259→        headers: { 'Content-Type': 'application/json' },
   260→        body: JSON.stringify({ actionId, action: 'rollback' }),
   261→      }).then((r) => {
   262→        if (!r.ok) return r.json().then((e) => Promise.reject(e));
   263→        return r.json();
   264→      }),
   265→    onSuccess: () => {
   266→      toast.success(t('son.actionRolledBack'));
   267→      queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   268→      queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   269→    },
   270→    onError: (error: any) => {
   271→      toast.error(error.error || t('son.failedToRollback'));
   272→    },
   273→  });
   274→
   275→  // ─── Derived Data ─────────────────────────────────────────────────
   276→
   277→  const modules = modulesData?.modules ?? [];
   278→
   279→  const filteredModules = useMemo(() => {
   280→    if (modeFilter === 'ALL') return modules;
   281→    return modules.filter((m) => m.mode === modeFilter);
   282→  }, [modules, modeFilter]);
   283→
   284→  const stats = useMemo(() => {
   285→    const totalModules = modules.length;
   286→    const activeModules = modules.filter((m) => m.enabled).length;
   287→    const totalActions = modules.reduce(
   288→      (sum, m) => sum + (m.actionCount ?? 0),
   289→      0
   290→    );
   291→    const allImpactScores = (actionsData?.actions ?? [])
   292→      .map((a) => a.impactScore)
   293→      .filter((s): s is number => s !== null && s !== undefined);
   294→    const avgImpact =
   295→      allImpactScores.length > 0
   296→        ? allImpactScores.reduce((a, b) => a + b, 0) / allImpactScores.length
   297→        : 0;
   298→
   299→    return { totalModules, activeModules, totalActions, avgImpact };
   300→  }, [modules, actionsData]);
   301→
   302→  const actions = actionsData?.actions ?? [];
   303→  const neighbors = neighborsData?.neighbors ?? [];
   304→
   305→  // ─── Handlers ─────────────────────────────────────────────────────
   306→
   307→  const handleToggleModule = (moduleId: string) => {
   308→    moduleMutation.mutate({ moduleId, action: 'toggle' });
   309→  };
   310→
   311→  const handleExecuteModule = (moduleId: string, displayName: string) => {
   312→    moduleMutation.mutate(
   313→      { moduleId, action: 'execute' },
   314→      {
   315→        onSuccess: () => {
   316→          toast.success(t('son.moduleExecuted', { name: displayName }));
   317→          queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   318→          queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   319→        },
   320→      }
   321→    );
   322→  };
   323→
   324→  const handleRollbackAction = (actionId: string) => {
   325→    rollbackActionMutation.mutate(actionId);
   326→  };
   327→
   328→  const handleRefresh = () => {
   329→    queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   330→    queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   331→    queryClient.invalidateQueries({ queryKey: ['son-neighbors'] });
   332→    toast.success(t('toast.dataRefreshed'));
   333→  };
   334→
   335→  const formatTime = (ts: string) => {
   336→    try {
   337→      return new Date(ts).toLocaleString(undefined, {
   338→        month: 'short',
   339→        day: 'numeric',
   340→        hour: '2-digit',
   341→        minute: '2-digit',
   342→      });
   343→    } catch {
   344→      return ts;
   345→    }
   346→  };
   347→
   348→  // ─── Render ───────────────────────────────────────────────────────
   349→
   350→  return (
   351→    <div className="space-y-6">
   352→      {/* ─── Header ───────────────────────────────────────────────── */}
   353→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   354→        <div className="flex items-center gap-3">
   355→          <div className="flex items-center justify-center h-10 w-10 rounded-lg bg-primary/10">
   356→            <Cpu className="h-5 w-5 text-primary" />
   357→          </div>
   358→          <div>
   359→            <h1 className="text-2xl font-bold tracking-tight">
   360→              {t('title.sonAutomation', { defaultValue: 'SON Automation' })}
   361→            </h1>
   362→            <p className="text-sm text-muted-foreground">
   363→              {t('son.subtitle')}
   364→            </p>
   365→          </div>
   366→        </div>
   367→        <div className="flex items-center gap-3">
   368→          <div className="flex items-center gap-2">
   369→            <Filter className="h-4 w-4 text-muted-foreground" />
   370→            <Select
   371→              value={selectedTechnology}
   372→              onValueChange={(v) =>
   373→                useAppStore.getState().setSelectedTechnology(v as Technology)
   374→              }
   375→            >
   376→              <SelectTrigger size="sm" className="w-[90px]">
   377→                <SelectValue />
   378→              </SelectTrigger>
   379→              <SelectContent>
   380→                {TECH_OPTIONS.map((t) => (
   381→                  <SelectItem key={t} value={t}>
   382→                    {t}
   383→                  </SelectItem>
   384→                ))}
   385→              </SelectContent>
   386→            </Select>
   387→          </div>
   388→          <Select
   389→            value={modeFilter}
   390→            onValueChange={(v) =>
   391→              setModeFilter(v as SonModuleMode | 'ALL')
   392→            }
   393→          >
   394→            <SelectTrigger size="sm" className="w-[170px]">
   395→              <SelectValue placeholder="Mode" />
   396→            </SelectTrigger>
   397→            <SelectContent>
   398→              {MODE_OPTIONS.map((m) => (
   399→                <SelectItem key={m} value={m}>
   400→                  {m === 'ALL'
   401→                    ? t('filter.allModes')
   402→                    : MODE_BADGE_CONFIG[m as SonModuleMode].label}
   403→                </SelectItem>
   404→              ))}
   405→            </SelectContent>
   406→          </Select>
   407→          <Button
   408→            variant="outline"
   409→            size="sm"
   410→            onClick={handleRefresh}
   411→            className="h-8"
   412→          >
   413→            <RefreshCw className="h-3.5 w-3.5 mr-1.5" />
   414→            Refresh
   415→          </Button>
   416→        </div>
   417→      </div>
   418→
   419→      {/* ─── Stats Cards ─────────────────────────────────────────── */}
   420→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   421→        <StatsCard
   422→          title={t('son.totalModules')}
   423→          value={stats.totalModules}
   424→          icon={<Layers className="h-5 w-5" />}
   425→          iconColor="text-slate-500"
   426→          iconBg="bg-slate-500/10"
   427→          loading={modulesLoading}
   428→        />
   429→        <StatsCard
   430→          title={t('son.activeModules')}
   431→          value={stats.activeModules}
   432→          subtitle={t('son.ofTotal', { n: stats.totalModules })}
   433→          icon={<Zap className="h-5 w-5" />}
   434→          iconColor="text-emerald-500"
   435→          iconBg="bg-emerald-500/10"
   436→          loading={modulesLoading}
   437→        />
   438→        <StatsCard
   439→          title={t('son.totalActions')}
   440→          value={stats.totalActions}
   441→          icon={<Activity className="h-5 w-5" />}
   442→          iconColor="text-amber-500"
   443→          iconBg="bg-amber-500/10"
   444→          loading={actionsLoading}
   445→        />
   446→        <StatsCard
   447→          title={t('son.avgImpact')}
   448→          value={stats.avgImpact}
   449→          isDecimal
   450→          icon={<Gauge className="h-5 w-5" />}
   451→          iconColor="text-cyan-500"
   452→          iconBg="bg-cyan-500/10"
   453→          loading={actionsLoading}
   454→        />
   455→      </div>
   456→
   457→      {/* ─── Module Cards Grid ────────────────────────────────────── */}
   458→      <div>
   459→        <h2 className="text-lg font-semibold mb-4">{t('son.modules')}</h2>
   460→        {modulesLoading ? (
   461→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   462→            {Array.from({ length: 6 }).map((_, i) => (
   463→              <Skeleton key={i} className="h-72 rounded-xl" />
   464→            ))}
   465→          </div>
   466→        ) : filteredModules.length === 0 ? (
   467→          <Card className="py-16">
   468→            <CardContent className="flex flex-col items-center justify-center text-center gap-2">
   469→              <Cpu className="h-10 w-10 text-muted-foreground/50" />
   470→              <p className="text-sm text-muted-foreground">
   471→                {t('son.noModules')}
   472→              </p>
   473→            </CardContent>
   474→          </Card>
   475→        ) : (
   476→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   477→            {filteredModules.map((mod) => (
   478→              <ModuleCard
   479→                key={mod.id}
   480→                module={mod}
   481→                onToggle={handleToggleModule}
   482→                onExecute={handleExecuteModule}
   483→                isMutating={moduleMutation.isPending}
   484→              />
   485→            ))}
   486→          </div>
   487→        )}
   488→      </div>
   489→
   490→      {/* ─── Tabs: Actions & Neighbors ────────────────────────────── */}
   491→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   492→        <TabsList className="grid w-full grid-cols-2 max-w-md">
   493→          <TabsTrigger value="actions" className="flex items-center gap-2">
   494→            <ArrowRightLeft className="h-3.5 w-3.5" />
   495→            {t('son.actionHistory')}
   496→            {!actionsLoading && actions.length > 0 && (
   497→              <Badge
   498→                variant="secondary"
   499→                className="ml-1 h-5 px-1.5 text-[10px] font-mono"
   500→              >
   501→                {actions.length}
   502→              </Badge>
   503→            )}
   504→          </TabsTrigger>
   505→          <TabsTrigger value="neighbors" className="flex items-center gap-2">
   506→            <Network className="h-3.5 w-3.5" />
   507→            {t('son.neighborRelations')}
   508→            {!neighborsLoading && neighbors.length > 0 && (
   509→              <Badge
   510→                variant="secondary"
   511→                className="ml-1 h-5 px-1.5 text-[10px] font-mono"
   512→              >
   513→                {neighbors.length}
   514→              </Badge>
   515→            )}
   516→          </TabsTrigger>
   517→        </TabsList>
   518→
   519→        {/* ── Action History Tab ──────────────────────────────────── */}
   520→        <TabsContent value="actions" className="mt-4">
   521→          <Card>
   522→            <CardHeader className="pb-3">
   523→              <CardTitle className="text-base font-semibold flex items-center gap-2">
   524→                <Activity className="h-4 w-4 text-muted-foreground" />
   525→                {t('son.recentActions')}
   526→              </CardTitle>
   527→              <ExportButton data={actions} filenamePrefix="son" columns={[{ key: 'createdAt', header: 'Time' }, { key: 'moduleName', header: 'Module' }, { key: 'siteName', header: 'Site' }, { key: 'actionType', header: 'Action Type' }, { key: 'parameter', header: 'Parameter' }, { key: 'oldValue', header: 'Old Value' }, { key: 'newValue', header: 'New Value' }, { key: 'status', header: 'Status' }]} />
   528→            </CardHeader>
   529→            <CardContent className="p-0">
   530→              {actionsLoading ? (
   531→                <div className="p-6 space-y-3">
   532→                  {Array.from({ length: 5 }).map((_, i) => (
   533→                    <Skeleton key={i} className="h-12 w-full" />
   534→                  ))}
   535→                </div>
   536→              ) : actions.length === 0 ? (
   537→                <div className="py-12 text-center text-sm text-muted-foreground">
   538→                  {t('son.noActions')}
   539→                </div>
   540→              ) : (
   541→                <ScrollArea className="max-h-96">
   542→                  <Table>
   543→                    <TableHeader>
   544→                      <TableRow className="hover:bg-transparent">
   545→                        <TableHead className="w-[150px] text-xs">{t('son.time')}</TableHead>
   546→                        <TableHead className="text-xs">{t('son.module')}</TableHead>
   547→                        <TableHead className="text-xs">{t('son.site')}</TableHead>
   548→                        <TableHead className="text-xs">{t('son.actionType')}</TableHead>
   549→                        <TableHead className="text-xs">{t('son.parameter')}</TableHead>
   550→                        <TableHead className="text-xs">{t('son.change')}</TableHead>
   551→                        <TableHead className="text-xs">{t('th.status')}</TableHead>
   552→                        <TableHead className="text-xs text-right">{t('son.impact')}</TableHead>
   553→                        <TableHead className="text-xs text-right w-[80px]">
   554→                          {t('th.rollback')}
   555→                        </TableHead>
   556→                      </TableRow>
   557→                    </TableHeader>
   558→                    <TableBody>
   559→                      {actions.map((action) => {
   560→                        const statusCfg =
