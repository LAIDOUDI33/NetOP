   446→export class OpenvasDefectDojoClient extends EventEmitter {
   447→  private config: VulnerabilityIntegrationConfig;
   448→  private openvasToken?: string;
   449→  private isConnected: boolean = false;
   450→  
   451→  // Statistics cache
   452→  public stats: VulnerabilityStatistics = this.initializeStats();
   453→
   454→  constructor(config: VulnerabilityIntegrationConfig) {
   455→    super();
   456→    this.config = config;
   457→
   458→    // Set up internal event handlers
   459→    this.on('error', (err) => {
   460→      console.error('[VULN] Error:', err.message);
   461→    });
   462→
   463→    this.on('finding', (vuln: UnifiedVulnerability) => {
   464→      console.log(`[VULN] Finding (${vuln.severity}): ${vuln.title} on ${vuln.host}`);
   465→      
   466→      // Check SLA breach
   467→      if (vuln.slaExceeded) {
   468→        this.emit('slaBreach', vuln);
   469→      }
   470→
   471→      // Auto-alert on critical
   472→      if (vuln.severity === 'Critical') {
   473→        this.emit('criticalFinding', vuln);
   474→      }
   475→    });
   476→  }
   477→
   478→  // ============================================================
   479→  // Connection Management
   480→  // ============================================================
   481→
   482→  /**
   483→   * Initialize connections to both OpenVAS and DefectDojo
   484→   */
   485→  async connect(): Promise<void> {
   486→    try {
   487→      console.log('[VULN] Connecting to OpenVAS at', this.config.openvas.apiUrl);
   488→
   489→      // Authenticate with OpenVAS/GVM
   490→      const authResponse = await this.fetchOpenvas('/auth', {
   491→        method: 'POST',
   492→        body: JSON.stringify({
   493→          username: this.config.openvas.username,
   494→          password: this.config.openvas.password,
   495→        }),
   496→      });
   497→
   498→      if (!authResponse.ok) {
   499→        throw new Error(`OpenVAS authentication failed: ${authResponse.status}`);
   500→      }
   501→
   502→      const authData = await authResponse.json();
   503→      this.openvasToken = authData.token || authData.access_token;
   504→      console.log('[VULN] OpenVAS authentication successful');
   505→
   506→      // Test DefectDojo connection
   507→      console.log('[VULN] Testing DefectDojo connection...');
   508→      const dojoHealth = await this.fetchDojo('/api/v2/products/?limit=1');
   509→      
   510→      if (dojoHealth.ok) {
   511→        console.log('[VULN] DefectDojo connected successfully');
   512→      } else if (dojoHealth.status === 403) {
   513→        console.warn('[VULN] DefectDojo auth issue, check API key');
   514→      } else {
   515→        console.warn('[VULN] DefectDojo returned:', dojoHealth.status);
   516→      }
   517→
   518→      this.isConnected = true;
   519→
   520→      // Refresh initial statistics
   521→      await this.refreshStats();
   522→
   523→      this.emit('connected', {
   524→        openvas: true,
   525→        defectdojo: dojoHealth.ok,
   526→        timestamp: new Date(),
   527→      });
   528→
   529→    } catch (error) {
   530→      this.isConnected = false;
   531→      this.emit('error', error);
   532→      throw error;
   533→    }
   534→  }
   535→
   536→  /**
   537→   * Disconnect from all services
   538→   */
   539→  async disconnect(): Promise<void> {
   540→    // Logout from OpenVAS to invalidate token
   541→    if (this.openvasToken) {
   542→      try {
   543→        await this.fetchOpenvas('/logout', { method: 'POST' });
   544→      } catch {
   545→        // Ignore logout errors
   546→      }
   547→      this.openvasToken = undefined;
   548→    }
   549→
   550→    this.isConnected = false;
   551→    this.emit('disconnected', { timestamp: new Date() });
   552→    console.log('[VULN] Disconnected from all vulnerability management services');
   553→  }
   554→
   555→  get connectionStatus(): boolean {
   556→    return this.isConnected;
   557→  }
   558→
   559→  // ============================================================
   560→  // OpenVAS Task Management
   561→  // ============================================================
   562→
   563→  /**
   564→   * Create a new vulnerability scan task
   565→   */
   566→  async createScan(config: ScanConfiguration): Promise<OpenvasTask> {
   567→    try {
   568→      // Step 1: Create target
   569→      const targetPayload: any = {
   570→        name: `Target-${config.name}-${Date.now()}`,
   571→        hosts: config.targets.join(', '),
   572→        ...(config.exclusions && config.exclusions.length > 0 && {
   573→          exclude_hosts: config.exclusions.join(', '),
   574→        }),
   575→        ...(config.portRange && { port_list_id: await this.createPortList(config.name, config.portRange) }),
   576→      };
   577→
   578→      const targetResponse = await this.fetchOpenvas('/targets', {
   579→        method: 'POST',
   580→        body: JSON.stringify(targetPayload),
   581→      }, true);
   582→
   583→      if (!targetResponse.ok) {
   584→        throw new Error(`Failed to create target: ${targetResponse.status}`);
   585→      }
   586→
   587→      const targetData = await targetResponse.json();
   588→      const targetId = targetData.id || targetData.target_id;
   589→
   590→      // Step 2: Get appropriate scan config based on profile
   591→      const scanConfigs = await this.getScanConfigs();
   592→      const selectedConfig = scanConfigs.find(c => 
   593→        c.name.toLowerCase().includes(config.scanProfile)
   594→      ) || scanConfigs[0];
   595→
   596→      // Step 3: Create task
   597→      const taskPayload = {
   598→        name: config.name,
   599→        target: { id: targetId },
   600→        config: { id: selectedConfig?.id },
   601→        scanner: { id: await this.getDefaultScanner() },
   602→        ...(config.schedule && {
   603→          schedule: await this.createSchedule(config.schedule),
   604→        }),
   605→      };
   606→
   607→      const taskResponse = await this.fetchOpenvas('/tasks', {
   608→        method: 'POST',
   609→        body: JSON.stringify(taskPayload),
   610→      }, true);
   611→
   612→      if (!taskResponse.ok) {
   613→        throw new Error(`Failed to create task: ${taskResponse.status}`);
   614→      }
   615→
   616→      const taskData = await taskResponse.json();
   617→      const task: OpenvasTask = {
   618→        taskId: taskData.task?.id || taskData.id,
   619→        name: config.name,
   620→        status: 'New',
   621→        targetId,
   622→        configId: selectedConfig?.id,
   623→        scannerId: taskData.task?.scanner?.id,
   624→        reportCount: 0,
   625→        createdAt: new Date().toISOString(),

   868→  /**
   869→   * Import OpenVAS scan results into DefectDojo
   870→   */
   871→  async importScanResults(params: {
   872→    productId: number;
   873→    engagementId?: number;
   874→    taskId: string;
   875→    scanType?: 'GVM Scan'; // DefectDojo scanner type name
   876→    closeOldFindings?: boolean;
   877→    pushToJira?: boolean;
   878→    createEngagementIfMissing?: boolean;
   879→  }): Promise<{
   880→    success: boolean;
   881→    testId?: number;
   882→    findingsImported?: number;
   883→    message: string;
   884→  }> {
   885→    const {
   886→      productId,
   887→      engagementId,
   888→      taskId,
   889→      scanType = 'GVM Scan',
   890→      closeOldFindings = true,
   891→      pushToJira = false,
   892→      createEngagementIfMissing = true,
   893→    } = params;
   894→
   895→    try {
   896→      // Step 1: Get the scan report from OpenVAS

   898→      
   899→      if (!reportData) {
   900→        throw new Error('No report available for task');
   901→      }
   902→

   907→          name: `Auto-Scan-${new Date().toISOString().split('T')[0]}`,
   908→          productId,
   909→          engType: 'CI/CD',
   910→          targetStart: new Date().toISOString(),
   911→          targetEnd: new Date(Date.now() + 86400000).toISOString(), // Tomorrow
   912→          status: 'Completed',
   913→          environment: 'Production',
   914→        });
   915→        finalEngagementId = engagement.id;
   916→      }
   917→
   918→      if (!finalEngagementId) {
   919→        throw new Error('No engagement available for import');
   920→      }
   921→
   922→      // Step 3: Import via DefectDojo API
   923→      const formData = new FormData();
   924→      formData.append('file', new Blob([reportData], { type: 'application/xml' }), 'report.xml');
   925→      formData.append('scan_type', scanType);
   926→      formData.append('engagement', finalEngagementId.toString());
   927→      formData.append('close_old_findings', closeOldFindings.toString());

   931→      formData.append('active', 'true');
   932→
   933→      const response = await this.fetchDojo('/api/v2/import-scan/', {
   934→        method: 'POST',
   935→        body: formData,

   937→
   938→      if (!response.ok) {
   939→        const errorText = await response.text();
   940→        throw new Error(`Import failed: ${response.status} - ${errorText}`);
   941→      }
   942→
   943→      const importResult = await response.json();
   944→
   945→      this.emit('importComplete', {
   946→        taskId,
   947→        productId,
   948→        findingsImported: importResult.count || importResult.new_findings || 0,
   949→      });
   950→
   951→      return {
   952→        success: true,
   953→        testId: importResult.test_id,
   954→        findingsImported: importResult.count || importResult.new_findings || 0,
   955→        message: `Successfully imported ${importResult.count || 0} findings`,
   956→      };
   957→
   958→    } catch (error) {
   959→      this.emit('error', error);
   960→      return { success: false, message: error.message };

  1129→  /**
  1130→   * Get comprehensive vulnerability statistics

  1132→  async getFullStats(): Promise<VulnerabilityStatistics> {
  1133→    try {
  1134→      // Gather statistics from multiple sources
  1135→      const [findings, products] = await Promise.all([
  1136→        this.getFindings({ limit: 5000 }), // Large limit for accurate stats
  1137→        this.getProducts({ limit: 200 }),
  1138→      ]);
  1139→
  1140→      // Calculate summary
  1141→      const summary = {
  1142→        totalFindings: findings.length,
  1143→        critical: findings.filter(f => f.severity === 'Critical').length,
  1144→        high: findings.filter(f => f.severity === 'High').length,
  1145→        medium: findings.filter(f => f.severity === 'Medium').length,
  1146→        low: findings.filter(f => f.severity === 'Low').length,
  1147→        info: findings.findings?.filter(f => f.severity === 'Info').length || 0,
  1148→        falsePositives: findings.filter(f => f.falsePositive).length,
  1149→        riskAccepted: findings.filter(f => f.riskAccepted).length,
  1150→      };
  1151→
  1152→      // Calculate by category
  1153→      const byCategory: Record<string, number> = {};
  1154→      findings.forEach(f => {
  1155→        // Extract category from title/description (simplified)
  1156→        const category = this.inferCategory(f);
  1157→        byCategory[category] = (byCategory[category] || 0) + 1;
  1158→      });
  1159→
  1160→      // Calculate by asset
  1161→      const assetMap = new Map<string, { vulnCount: number; criticalCount: number }>();
  1162→      findings.forEach(f => {
  1163→        f.endpoints?.forEach(ep => {
  1164→          const existing = assetMap.get(ep.host) || { vulnCount: 0, criticalCount: 0 };
  1165→          existing.vulnCount++;
  1166→          if (f.severity === 'Critical' || f.severity === 'High') {
  1167→            existing.criticalCount++;
  1168→          }
  1169→          assetMap.set(ep.host, existing);
  1170→        });
  1171→      });
  1172→
  1173→      const byAsset = Array.from(assetMap.entries())
  1174→        .map(([host, stats]) => ({ host, ...stats }))
  1175→        .sort((a, b) => b.vulnCount - a.vulnCount)
  1176→        .slice(0, 20);
  1177→
  1178→      // Top CVEs
  1179→      const cveMap = new Map<string, { occurrences: number; totalCvss: number }>();
  1180→      findings.forEach(f => {
  1181→        if (f.cve) {
  1182→          const existing = cveMap.get(f.cve) || { occurrences: 0, totalCvss: 0 };
  1183→          existing.occurrences++;
  1184→          existing.totalCvss += parseFloat(f.numericalSeverity || '0');
  1185→          cveMap.set(f.cve, existing);
  1186→        }
  1187→      });
  1188→
  1189→      const topCves = Array.from(cveMap.entries())
  1190→        .map(([cve, stats]) => ({
  1191→          cve,
  1192→          occurrences: stats.occurrences,
  1193→          avgCvss: stats.totalCvss / stats.occurrences,
  1194→        }))
  1195→        .sort((a, b) => b.occurrences - a.occurrences)
  1196→        .slice(0, 10);
  1197→
  1198→      // Update cached stats
  1199→      this.stats = {
  1200→        ...this.stats,
  1201→        summary,
  1202→        byCategory,
  1203→        byAsset,
  1204→        topCves,
  1205→        complianceStatus: this.calculateCompliance(findings),
  1206→      };
  1207→
  1208→      return this.stats;
  1209→
  1210→    } catch (error) {
  1211→      this.emit('error', error);
  1212→      return this.stats;
  1213→    }
  1214→  }

  1223→  async healthCheck(): Promise<{
  1224→    openvas: { status: 'healthy' | 'degraded' | 'unhealthy'; latency: number; runningScans: number };
  1225→    defectdojo: { status: 'healthy' | 'degraded' | 'unhealthy'; latency: number; findingCount: number };
  1226→    overall: 'operational' | 'degraded' | 'down';
  1227→  }> {
  1228→    const checks = await Promise.allSettled([
  1229→      this.checkOpenvasHealth(),
  1230→      this.checkDefectDojoHealth(),
  1231→    ]);
  1232→
  1233→    const openvas = checks[0].status === 'fulfilled' ? checks[0].value : { status: 'unhealthy' as const, latency: -1, runningScans: 0 };
  1234→    const defectdojo = checks[1].status === 'fulfilled' ? checks[1].value : { status: 'unhealthy' as const, latency: -1, findingCount: 0 };
  1235→
  1236→    const healthyCount = [openvas, defectdojo].filter(c => c.status === 'healthy').length;
  1237→
  1238→    return {
  1239→      openvas,
  1240→      defectdojo,
  1241→      overall: healthyCount === 2 ? 'operational' : healthyCount >= 1 ? 'degraded' : 'down',
  1242→    };
  1243→  }

  1249→  private async fetchOpenvas(path: string, init: RequestInit = {}, skipAuthCheck = false): Promise<Response> {
  1250→    const url = `${this.config.openvas.apiUrl.replace(/\/$/, '')}${path}`;
  1251→    
  1252→    if (!this.openvasToken && !skipAuthCheck) {
  1253→      throw new Error('Not authenticated with OpenVAS');
  1254→    }
  1255→
  1256→    const headers: HeadersInit = {
  1257→      ...(init.headers as Record<string, string>),
  1258→    };
  1259→
  1260→    // Add token for authenticated requests
  1261→    if (this.openvasToken && !skipAuthCheck) {
  1262→      headers['Authorization'] = `Bearer ${this.openvasToken}`;
  1263→    } else if (!skipAuthCheck) {
  1264→      // Basic auth fallback
  1265→      const credentials = Buffer.from(
  1266→        `${this.config.openvas.username}:${this.config.openvas.password}`
  1267→      ).toString('base64');
  1268→      headers['Authorization'] = `Basic ${credentials}`;
  1269→    }
  1270→
  1271→    // Set content type for non-form-data requests
  1272→    if (!(init.body instanceof FormData)) {
  1273→      headers['Content-Type'] = 'application/json';
  1274→    }
  1275→
  1276→    return fetch(url, {
  1277→      ...init,
  1278→      headers,
  1279→      signal: AbortSignal.timeout(this.config.openvas.timeout || 120000),
  1280→    });
  1281→  }

  1302→  private transformToUnified(finding: DojoFinding): UnifiedVulnerability {
  1303→    const now = new Date();
  1304→    const discoveredAt = new Date(finding.date || finding.createdAt);
  1305→    
  1306→    // Calculate SLA deadline based on severity
  1307→    const slaConfig = this.config.slaConfig || {
  1308→      criticalHours: 4,
  1309→      highHours: 24,
  1310→      mediumDays: 7,
  1311→      lowDays: 30,
  1312→    };
  1313→
  1314→    let slaDeadline: Date;
  1315→    switch (finding.severity) {
  1316→      case 'Critical':
  1317→        slaDeadline = new Date(discoveredAt.getTime() + slaConfig.criticalHours * 3600000);
  1318→        break;
  1319→      case 'High':
  1320→        slaDeadline = new Date(discoveredAt.getTime() + slaConfig.highHours * 3600000);
  1321→        break;
  1322→      case 'Medium':
  1323→        slaDeadline = new Date(discoveredAt.getTime() + slaConfig.mediumDays * 86400000);
  1324→        break;
  1325→      default:
  1326→        slaDeadline = new Date(discoveredAt.getTime() + slaConfig.lowDays * 86400000);
  1327→    }
  1328→
  1329→    const isSlaExceeded = now > slaDeadline && finding.active && !finding.riskAccepted;
  1330→
  1331→    // Determine status
  1332→    let status: UnifiedVulnerability['status'];
  1333→    if (finding.falsePositive) status = 'false_positive';
  1334→    else if (finding.duplicate) status = 'duplicate';
  1335→    else if (finding.riskAccepted) status = 'risk_accepted';
  1336→    else if (!finding.active) status = 'fixed';
  1337→    else if (finding.mitigation && finding.verified) status = 'in_progress';
  1338→    else status = 'open';
  1339→
  1340→    return {
  1341→      vulnId: `dd-${finding.id}`,
  1342→      source: 'defectdojo',
  1343→      title: finding.title,
  1344→      description: finding.description,
  1345→      severity: finding.severity,
  1346→      cvssScore: parseFloat(finding.numericalSeverity || '0'),
  1347→      cve: finding.cve,
  1348→      host: finding.endpoints?.[0]?.host || 'unknown',
  1349→      port: finding.endpoints?.[0]?.port,
  1350→      protocol: finding.endpoints?.[0]?.protocol,
  1351→      category: this.inferCategory(finding),
  1352→      solution: finding.mitigation,
  1353→      references: finding.references?.split('\n').filter(Boolean),
  1354→      status,
  1355→      discoveredAt: finding.date || finding.createdAt,
  1356→      lastSeenAt: finding.updatedAt,
  1357→      fixedAt: !finding.active ? finding.updatedAt : undefined,
  1358→      slaDeadline: slaDeadline.toISOString(),
  1359→      slaExceeded: isSlaExceeded,
  1360→      daysOpen: Math.floor((now.getTime() - discoveredAt.getTime()) / 86400000),
  1361→      findingId: finding.id,
  1362→      jiraTicket: finding.jiraIssue,
  1363→      complianceMapping: finding.complianceMapping,
  1364→    };
  1365→  }

  1367→  private inferCategory(finding: DojoFinding): string {
  1368→    const title = (finding.title || '').toLowerCase();
  1369→    const desc = (finding.description || '').toLowerCase();
  1370→    const combined = `${title} ${desc}`;
  1371→
  1372→    if (combined.match(/(sql injection|sqli|xss|cross-site|csrf|command injection|rce)/)) return 'Injection';
  1373→    if (combined.match(/(auth|password|login|session|token|jwt|oauth)/)) return 'Authentication';
  1374→    if (combined.match(/(encrypt|tls|ssl|certificate|crypto|key)/)) return 'Cryptography';
  1375→    if (combined.match(/(xxe|xml)/)) return 'XML Processing';
  1376→    if (combined.match(/(ssrf|redirect|open.forward|host.header)/)) return 'Input Validation';
  1377→    if (combined.match(/(file.upload|path.traversal|lfi|rfi)/)) return 'File Handling';
  1378→    if (combined.match(/(information.disclosure|debug|error.message|stack.trace)/)) return 'Information Disclosure';
  1379→    if (combined.match(/(cors|content.type|mime.sniffing)/)) return 'Security Configuration';
  1380→    if (combined.match(/(outdated|version|old|legacy|deprecated)/)) return 'Outdated Component';
  1381→    if (combined.match(/(dos|ddos|resource.exhaustion)/)) return 'Denial of Service';
  1382→    if (combined.match(/(privilege|access.control|authorization|rbac|abac)/)) return 'Access Control';
  1383→
  1384→    return 'Other';
  1385→  }
