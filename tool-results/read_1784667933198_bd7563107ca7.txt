   116→// ─── Helper Functions ──────────────────────────────────────────────────
   117→
   118→function probabilityBarColor(pct: number): string {
   119→  if (pct >= 80) return 'bg-red-500';
   120→  if (pct >= 60) return 'bg-amber-500';
   121→  if (pct >= 40) return 'bg-amber-400';
   122→  return 'bg-emerald-500';
   123→}
   124→
   125→function probabilityProgressClass(pct: number): string {
   126→  if (pct >= 80) return '[&>div]:bg-red-500';
   127→  if (pct >= 60) return '[&>div]:bg-amber-500';
   128→  if (pct >= 40) return '[&>div]:bg-amber-400';
   129→  return '[&>div]:bg-emerald-500';
   130→}
   131→
   132→function formatFaultType(ft: string): string {
   133→  return ft.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase());
   134→}
   135→
   136→function formatStatus(s: string): string {
   137→  return s.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase());
   138→}
   139→
   140→function formatDate(iso: string): string {
   141→  const d = new Date(iso);
   142→  return d.toLocaleDateString('en-US', {
   143→    month: 'short',
   144→    day: 'numeric',
   145→    hour: '2-digit',
   146→    minute: '2-digit',
   147→  });
   148→}

   206→function ChartTooltipContent({ active, payload, label }: any) {

   209→    <div className="rounded-lg border bg-background px-3 py-2 shadow-md text-xs">
