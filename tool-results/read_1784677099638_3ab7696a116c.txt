   100→function healthScoreColor(score: number): string {
   101→  if (score >= 80) return 'text-emerald-600 dark:text-emerald-400';
   102→  if (score >= 60) return 'text-amber-600 dark:text-amber-400';
   103→  return 'text-red-600 dark:text-red-400';
   104→}
   105→
   106→function healthBarColor(score: number): string {
   107→  if (score >= 80) return 'bg-emerald-500';
   108→  if (score >= 60) return 'bg-amber-500';
   109→  return 'bg-red-500';
   110→}

   164→function ChartTooltipContent({ active, payload, label }: any) {
   165→  if (!active || !payload?.length) return null;
   166→  return (
   167→    <div className="rounded-lg border bg-background px-3 py-2 shadow-md text-xs">
   168→      <p className="font-medium mb-1">{label}</p>
   169→      {payload.map((entry: any, idx: number) => (
   170→        <div key={idx} className="flex items-center gap-2">
   171→          <div className="w-2.5 h-2.5 rounded-sm" style={{ backgroundColor: entry.color }} />
   172→          <span className="text-muted-foreground">{entry.name}:</span>
   173→          <span className="font-medium">{formatNumber(entry.value)}</span>
   174→        </div>
   175→      ))}
   176→    </div>
   177→  );
   178→}

   180→// ─── Pie Label ─────────────────────────────────────────────────────────
   181→
   182→const RADIAN = Math.PI / 180;
   183→
   184→function renderPieLabel({ cx, cy, midAngle, innerRadius, outerRadius, percent, name }: any) {
   185→  const radius = innerRadius + (outerRadius - innerRadius) * 1.6;
   186→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   187→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   188→  return (
   189→    <text x={x} y={y} textAnchor={x > cx ? 'start' : 'end'} className="fill-muted-foreground text-xs">
   190→      {name} ({(percent * 100).toFixed(0)}%)
   191→    </text>
   192→  );
   193→}
