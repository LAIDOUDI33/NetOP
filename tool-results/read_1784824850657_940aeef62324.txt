   180→   180→
   181→   181→export default function SonView() {
   182→   182→  const t = useT();

   199→   199→    refetchInterval: 15000,
   200→   200→  });
   201→   201→
   202→   202→  const {
   203→   203→    data: actionsData,
   204→   204→    isLoading: actionsLoading,
   205→   205→  } = useQuery<{ actions: SonActionItem[]; pagination?: any }>({
   206→   206→    queryKey: ['son-actions'],
   207→   207→    queryFn: () =>
   208→   208→      fetch('/api/son/actions?limit=50').then((r) => { if (!r.ok) throw new Error('SON Actions API error: ' + r.status); return r.json(); }),
   209→   209→    refetchInterval: 15000,
   210→   210→  });

   212→   212→  const {
   213→   213→    data: neighborsData,

   216→   216→    queryKey: ['son-neighbors'],
   217→   217→    queryFn: () =>
   218→   218→      fetch('/api/son/neighbors').then((r) => { if (!r.ok) throw new Error('SON Neighbors API error: ' + r.status); return r.json(); }),
   219→   219→    refetchInterval: 15000,

   225→   225→    mutationFn: (body: {
   226→   226→      moduleId: string;
   227→   227→      action: 'toggle' | 'execute' | 'rollback';
   228→   228→    }) =>
   229→   229→      fetch('/api/son', {
   230→   230→        method: 'PATCH',
   231→   231→        headers: { 'Content-Type': 'application/json' },
   232→   232→        body: JSON.stringify(body),
   233→   233→      }).then((r) => {
   234→   234→        if (!r.ok) return r.json().then((e) => Promise.reject(e));
   235→   235→        return r.json();
   236→   236→      }),

   238→   238→      const actionLabel =
   239→   239→        variables.action === 'toggle'
   240→   240→          ? t('son.toggled')
   241→   241→          : variables.action === 'execute'
   242→   242→          ? t('son.executed')
   243→   243→          : t('son.rolledBack');

   245→   245→      queryClient.invalidateQueries({ queryKey: ['son-modules'] });

   247→   247→    },
   248→   248→    onError: (error: any, variables: any) => {
   249→   249→      toast.error(error.error || t('son.failedToAction', { action: error.action || variables.action }));

   251→   251→  });

   253→   253→  const rollbackActionMutation = useMutation({
   254→   254→    mutationFn: (actionId: string) =>

   256→   256→        method: 'PATCH',
   257→   257→        headers: { 'Content-Type': 'application/json' },
   258→   258→        body: JSON.stringify({ actionId, action: 'rollback' }),
   259→   259→      }).then((r) => {
   260→   260→        if (!r.ok) return r.json().then((e) => Promise.reject(e));
   261→   261→        return r.json();
   262→   262→      }),
   263→   263→    onSuccess: () => {
   264→   264→      toast.success(t('son.actionRolledBack'));
   265→   265→      queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   266→   266→      queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   267→   267→    },
   268→   268→    onError: (error: any) => {
   269→   269→      toast.error(error.error || t('son.failedToRollback'));
   270→   270→    },
   271→   271→  });
   272→   272→
   273→   273→  // ─── Derived Data ─────────────────────────────────────────────────
   274→   274→
   275→   275→  const modules = modulesData?.modules ?? [];

   280→   280→  }, [modules, modeFilter]);
   281→   281→
   282→   282→  const stats = useMemo(() => {

   285→   285→    const totalActions = modules.reduce(
   286→   286→      (sum, m) => sum + (m.actionCount ?? 0),
   287→   287→      0
   288→   288→    );
   289→   289→    const allImpactScores = (actionsData?.actions ?? [])
   290→   290→      .map((a) => a.impactScore)
   291→   291→      .filter((s): s is number => s !== null && s !== undefined);
   292→   292→    const avgImpact =
   293→   293→      allImpactScores.length > 0
   294→   294→        ? allImpactScores.reduce((a, b) => a + b, 0) / allImpactScores.length
   295→   295→        : 0;
   296→   296→
   297→   297→    return { totalModules, activeModules, totalActions, avgImpact };
   298→   298→  }, [modules, actionsData]);
   299→   299→
   300→   300→  const actions = actionsData?.actions ?? [];
   301→   301→  const neighbors = neighborsData?.neighbors ?? [];
   302→   302→
   303→   303→  // ─── Handlers ─────────────────────────────────────────────────────
   304→   304→
   305→   305→  const handleToggleModule = (moduleId: string) => {
   306→   306→    moduleMutation.mutate({ moduleId, action: 'toggle' });
   307→   307→  };
   308→   308→
   309→   309→  const handleExecuteModule = (moduleId: string, displayName: string) => {
   310→   310→    moduleMutation.mutate(
   311→   311→      { moduleId, action: 'execute' },
   312→   312→      {
   313→   313→        onSuccess: () => {
   314→   314→          toast.success(t('son.moduleExecuted', { name: displayName }));
   315→   315→          queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   316→   316→          queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   317→   317→        },
   318→   318→      }
   319→   319→    );
   320→   320→  };
   321→   321→
   322→   322→  const handleRollbackAction = (actionId: string) => {
   323→   323→    rollbackActionMutation.mutate(actionId);
   324→   324→  };
   325→   325→
   326→   326→  const handleRefresh = () => {
   327→   327→    queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   328→   328→    queryClient.invalidateQueries({ queryKey: ['son-actions'] });

   330→   330→    toast.success(t('toast.dataRefreshed'));
   331→   331→  };
   332→   332→
   333→   333→  const formatTime = (ts: string) => {

   335→   335→      return new Date(ts).toLocaleString(undefined, {
   336→   336→        month: 'short',
   337→   337→        day: 'numeric',
   338→   338→        hour: '2-digit',
   339→   339→        minute: '2-digit',
   340→   340→      });
   341→   341→    } catch {
   342→   342→      return ts;
   343→   343→    }
   344→   344→  };
   345→   345→
   346→   346→  // ─── Render ───────────────────────────────────────────────────────
   347→   347→
   348→   348→  return (
   349→   349→    <div className="space-y-6">
   350→   350→      {/* ─── Header ───────────────────────────────────────────────── */}
   351→   351→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   352→   352→        <div className="flex items-center gap-3">
   353→   353→          <div className="flex items-center justify-center h-10 w-10 rounded-lg bg-primary/10">
   354→   354→            <Cpu className="h-5 w-5 text-primary" />
   355→   355→          </div>
   356→   356→          <div>

   358→   358→              {t('title.sonAutomation', { defaultValue: 'SON Automation' })}

   361→   361→              {t('son.subtitle')}
   362→   362→            </p>
   363→   363→          </div>
   364→   364→        </div>
   365→   365→        <div className="flex items-center gap-3">
   366→   366→          <div className="flex items-center gap-2">
   367→   367→            <Filter className="h-4 w-4 text-muted-foreground" />
   368→   368→            <Select
   369→   369→              value={selectedTechnology}

   371→   371→                useAppStore.getState().setSelectedTechnology(v as Technology)

   416→   416→
   417→   417→      {/* ─── Stats Cards ─────────────────────────────────────────── */}
   418→   418→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   419→   419→        <StatsCard
   420→   420→          title={t('son.totalModules')}
   421→   421→          value={stats.totalModules}
   422→   422→          icon={<Layers className="h-5 w-5" />}

   425→   425→          loading={modulesLoading}
   426→   426→        />
   427→   427→        <StatsCard
   428→   428→          title={t('son.activeModules')}
   429→   429→          value={stats.activeModules}

   435→   435→        />
   436→   436→        <StatsCard
   437→   437→          title={t('son.totalActions')}
   438→   438→          value={stats.totalActions}
   439→   439→          icon={<Activity className="h-5 w-5" />}
   440→   440→          iconColor="text-amber-500"

   455→   455→      {/* ─── Module Cards Grid ────────────────────────────────────── */}

   458→   458→        {modulesLoading ? (
   459→   459→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   460→   460→            {Array.from({ length: 6 }).map((_, i) => (
   461→   461→              <Skeleton key={i} className="h-72 rounded-xl" />

   468→   468→              <p className="text-sm text-muted-foreground">

   471→   471→            </CardContent>
