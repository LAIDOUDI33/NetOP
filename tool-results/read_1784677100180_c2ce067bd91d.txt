   105→// ─── Helper Functions ──────────────────────────────────────────────────
   106→
   107→function progressColor(pct: number): string {
   108→  if (pct < 30) return 'bg-red-500';
   109→  if (pct < 70) return 'bg-amber-500';
   110→  return 'bg-emerald-500';
   111→}
   112→
   113→function progressTextColor(pct: number): string {
   114→  if (pct < 30) return 'text-red-600 dark:text-red-400';
   115→  if (pct < 70) return 'text-amber-600 dark:text-amber-400';
   116→  return 'text-emerald-600 dark:text-emerald-400';
   117→}
   118→
   119→function formatCurrency(n: number): string {
   120→  if (n >= 1_000_000) return `$${(n / 1_000_000).toFixed(1)}M`;
   121→  if (n >= 1_000) return `$${(n / 1_000).toFixed(1)}K`;
   122→  return `$${n.toLocaleString()}`;
   123→}
   124→
   125→function formatDate(d: string): string {
   126→  if (!d) return '—';
   127→  return new Date(d).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
   128→}

   183→function ChartTooltipContent({ active, payload, label }: any) {
   184→  if (!active || !payload?.length) return null;
   185→  return (
   186→    <div className="rounded-lg border bg-background px-3 py-2 shadow-md text-xs">
   187→      <p className="font-medium mb-1">{label}</p>
   188→      {payload.map((entry: any, idx: number) => (
   189→        <div key={idx} className="flex items-center gap-2">

   191→          <span className="text-muted-foreground">{entry.name}:</span>
   192→          <span className="font-medium">{entry.value}</span>

   194→      ))}
   195→    </div>
   196→  );
