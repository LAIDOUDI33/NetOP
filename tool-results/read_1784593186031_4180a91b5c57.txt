    99→    99→
   100→   100→function mosColor(score: number | null): string {
   101→   101→  if (score == null) return 'text-muted-foreground';
   102→   102→  if (score >= 4) return 'text-emerald-600 dark:text-emerald-400';
   103→   103→  if (score >= 3) return 'text-amber-600 dark:text-amber-400';
   104→   104→  return 'text-red-600 dark:text-red-400';
   105→   105→}
   106→   106→
   107→   107→function satisfactionColor(index: number | null): string {
   108→   108→  if (index == null) return 'text-muted-foreground';
   109→   109→  if (index >= 70) return 'text-emerald-600 dark:text-emerald-400';
   110→   110→  if (index >= 50) return 'text-amber-600 dark:text-amber-400';
   111→   111→  return 'text-red-600 dark:text-red-400';
   112→   112→}
   113→   113→
   114→   114→function satisfactionBg(index: number | null): string {
   115→   115→  if (index == null) return 'bg-muted';
   116→   116→  if (index >= 70) return 'bg-emerald-500';
   117→   117→  if (index >= 50) return 'bg-amber-500';
   118→   118→  return 'bg-red-500';
   119→   119→}
   120→   120→
   121→   121→function computeOverallAvgMos(sites: QoESiteSummary[]): number {
   122→   122→  const valid = sites.filter((s) => s.mosScore != null);
   123→   123→  if (valid.length === 0) return 0;
   124→   124→  return Number((valid.reduce((sum, s) => sum + (s.mosScore ?? 0), 0) / valid.length).toFixed(2));
   125→   125→}
   126→   126→
   127→   127→function computeOverallAvgSatisfaction(sites: QoESiteSummary[]): number {
   128→   128→  const valid = sites.filter((s) => s.satisfactionIndex != null);
   129→   129→  if (valid.length === 0) return 0;
   130→   130→  return Number((valid.reduce((sum, s) => sum + (s.satisfactionIndex ?? 0), 0) / valid.length).toFixed(1));
   131→   131→}

   133→   133→function formatBucketTime(bucket: string): string {
   134→   134→  const d = new Date(bucket);
   135→   135→  return `${String(d.getHours()).padStart(2, '0')}:00`;
   136→   136→}
