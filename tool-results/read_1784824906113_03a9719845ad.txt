   239→// Mini chart component (CSS-based)
   240→function MiniChart({ data, color = "#3b82f6" }: { data: number[]; color?: string }) {
   241→  const max = Math.max(...data);
   242→  const min = Math.min(...data);
   243→  const range = max - min || 1;
   244→  
   245→  const points = data.map((value, index) => {
   246→    const x = (index / (data.length - 1)) * 100;
   247→    const y = 100 - ((value - min) / range) * 80 - 10;
   248→    return `${x},${y}`;
   249→  }).join(" ");
   250→
   251→  return (
   252→    <svg className="w-full h-12" viewBox="0 0 100 100" preserveAspectRatio="none">
   253→      <defs>
   254→        <linearGradient id={`gradient-${color.replace('#', '')}`} x1="0%" y1="0%" x2="0%" y2="100%">
   255→          <stop offset="0%" stopColor={color} stopOpacity="0.3" />
   256→          <stop offset="100%" stopColor={color} stopOpacity="0" />
   257→        </linearGradient>
   258→      </defs>
   259→      <polygon
   260→        points={`0,100 ${points} 100,100`}
   261→        fill={`url(#gradient-${color.replace('#', '')})`}
   262→      />
   263→      <polyline
   264→        points={points}
   265→        fill="none"
   266→        stroke={color}
   267→        strokeWidth="2"
   268→        vectorEffect="non-scaling-stroke"
   269→      />
   270→    </svg>
   271→  );
   272→}
