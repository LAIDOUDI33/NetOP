   100→// ─── Helper Functions ──────────────────────────────────────────────────
   101→
   102→function loadBarColor(load: number): string {
   103→  if (load >= 80) return 'bg-red-500';
   104→  if (load >= 60) return 'bg-amber-500';
   105→  return 'bg-emerald-500';
   106→}
   107→
   108→function loadTextColor(load: number): string {
   109→  if (load >= 80) return 'text-red-600 dark:text-red-400';
   110→  if (load >= 60) return 'text-amber-600 dark:text-amber-400';
   111→  return 'text-emerald-600 dark:text-emerald-400';
   112→}
   113→
   114→function loadProgressClass(load: number): string {
   115→  if (load >= 80) return '[&>div]:bg-red-500';
   116→  if (load >= 60) return '[&>div]:bg-amber-500';
   117→  return '[&>div]:bg-emerald-500';
   118→}
   119→
   120→function computeTypeStats(slices: NetworkSlice[], type: string) {
   121→  const typed = slices.filter((s) => s.sliceType === type);
   122→  if (typed.length === 0) return { count: 0, avgLoad: 0, totalUsers: 0 };
   123→  const avgLoad = Number((typed.reduce((sum, s) => sum + s.currentLoad, 0) / typed.length).toFixed(1));
   124→  const totalUsers = typed.reduce((sum, s) => sum + s.activeUsers, 0);
   125→  return { count: typed.length, avgLoad, totalUsers };
   126→}
