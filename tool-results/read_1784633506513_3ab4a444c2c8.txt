   106→function StatusPipelineIndicator({ status }: { status: OnboardingStatus }) {
   107→  const isFailed = status === 'failed';
   108→  const currentStepIdx = isFailed
   109→    ? -1
   110→    : PIPELINE_STEPS.indexOf(status as (typeof PIPELINE_STEPS)[number]);
   111→
   112→  return (
   113→    <div className="flex items-center gap-1">
   114→      {PIPELINE_STEPS.map((step, idx) => {
   115→        const isCompleted = !isFailed && idx < currentStepIdx;
   116→        const isCurrent = !isFailed && idx === currentStepIdx;
   117→        const isFuture = !isCompleted && !isCurrent;
   118→
   119→        return (
   120→          <div key={step} className="flex items-center">
   121→            <div className="relative flex items-center justify-center">
   122→              {isCurrent ? (
   123→                <span className="absolute inline-flex h-4 w-4 animate-ping rounded-full bg-emerald-400 opacity-60" />
   124→              ) : null}
   125→              <div
   126→                className={`h-3 w-3 rounded-full border-[1.5px] transition-all ${
   127→                  isCompleted
   128→                    ? 'border-emerald-500 bg-emerald-500'
   129→                    : isCurrent
   130→                      ? 'border-emerald-400 bg-emerald-400'
   131→                      : 'border-muted-foreground/30 bg-transparent'
   132→                }`}
   133→              />
   134→            </div>
   135→            {idx < PIPELINE_STEPS.length - 1 && (
   136→              <div
   137→                className={`mx-[2px] h-[1.5px] w-3 ${
   138→                  !isFailed && idx < currentStepIdx
   139→                    ? 'bg-emerald-500'
   140→                    : 'bg-muted-foreground/20'
   141→                }`}
   142→              />
   143→            )}
   144→          </div>
   145→        );
   146→      })}
   147→      {isFailed && (
   148→        <>
   149→          <div className="mx-[2px] h-[1.5px] w-3 bg-red-400" />
   150→          <div className="relative flex items-center justify-center">
   151→            <span className="absolute inline-flex h-4 w-4 animate-ping rounded-full bg-red-400 opacity-50" />
   152→            <div className="h-3 w-3 rounded-full border-[1.5px] border-red-500 bg-red-500" />
   153→          </div>
   154→        </>
   155→      )}
   156→    </div>
   157→  );
   158→}

   257→  const completed = counts.completed ?? 0;
   258→  const failed = counts.failed ?? 0;
   259→
   260→  // ── Create mutation ──
   261→
   262→  const createMutation = useMutation({
   263→    mutationFn: (values: OnboardingFormValues) =>
   264→      fetch('/api/onboarding', {
   265→        method: 'POST',
   266→        headers: { 'Content-Type': 'application/json' },
   267→        body: JSON.stringify(values),
   268→      }).then((r) => {
   269→        if (!r.ok) return r.json().then((e) => Promise.reject(new Error(e.error)));
   270→        return r.json();
   271→      }),
   272→    onSuccess: () => {
   273→      queryClient.invalidateQueries({ queryKey: ['onboarding'] });
   274→      toast.success('Site onboarding initiated successfully');
   275→      form.reset();
   276→    },
   277→    onError: (err: Error) => {
   278→      toast.error(err.message || 'Failed to create onboarding');
   279→    },
   280→  });
   281→
   282→  // ── Advance mutation ──
   283→
   284→  const advanceMutation = useMutation({
   285→    mutationFn: ({ onboardingId }: { onboardingId: string }) =>
   286→      fetch('/api/onboarding', {
   287→        method: 'PATCH',
   288→        headers: { 'Content-Type': 'application/json' },
   289→        body: JSON.stringify({ onboardingId, action: 'advance' }),
   290→      }).then((r) => {
   291→        if (!r.ok) return r.json().then((e) => Promise.reject(new Error(e.error)));
   292→        return r.json();
   293→      }),
   294→    onSuccess: () => {
   295→      queryClient.invalidateQueries({ queryKey: ['onboarding'] });
   296→      toast.success('Pipeline advanced to next stage');
   297→    },
   298→    onError: (err: Error) => {
   299→      toast.error(err.message || 'Failed to advance pipeline');
   300→    },
   301→  });
   302→
   303→  // ── Retry mutation (resets failed → pending) ──
   304→
   305→  const retryMutation = useMutation({
   306→    mutationFn: ({ onboardingId }: { onboardingId: string }) =>
   307→      fetch('/api/onboarding', {
   308→        method: 'PATCH',
   309→        headers: { 'Content-Type': 'application/json' },
   310→        body: JSON.stringify({ onboardingId, action: 'advance', status: 'pending' }),
   311→      }).then((r) => {
   312→        if (!r.ok) return r.json().then((e) => Promise.reject(new Error(e.error)));

   333→      vendor: '',

   335→      longitude: 0,
   336→      altitude: 0,
   337→      frequency: '',
   338→      bandwidth: 0,
   339→      maxCapacity: 0,
   340→    },
   341→  });
   342→
   343→  const onSubmit = (values: OnboardingFormValues) => {
   344→    createMutation.mutate(values);
   345→  };
   346→
   347→  // ── Helpers ──
   348→
   349→  const formatDate = (iso: string) => {
   350→    try {
   351→      return new Date(iso).toLocaleDateString('en-US', {
   352→        month: 'short',
   353→        day: 'numeric',
   354→        hour: '2-digit',
   355→        minute: '2-digit',
   356→      });

   358→      return '—';
   359→    }
   360→  };
   361→
   362→  const isAdvanceable = (status: OnboardingStatus) =>
   363→    status !== 'completed' && status !== 'failed';
   364→
   365→  const isRetryable = (status: OnboardingStatus) => status === 'failed';

   369→  return (
