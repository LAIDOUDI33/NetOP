   178→// ─── Component ───────────────────────────────────────────────────────
   179→
   180→export default function SonView() {
   181→  const t = useT();
   182→  const queryClient = useQueryClient();
   183→  const selectedTechnology = useAppStore((s) => s.selectedTechnology);
   184→  const [modeFilter, setModeFilter] = useState<SonModuleMode | 'ALL'>('ALL');
   185→  const [activeTab, setActiveTab] = useState('actions');
   186→
   187→  const techParam =
   188→    selectedTechnology === 'ALL' ? undefined : selectedTechnology;
   189→
   190→  // ─── Queries ──────────────────────────────────────────────────────
   191→  // Note: SON modules use compound technology (e.g. "4G,5G"), so we
   192→  // fetch all modules without technology filter.
   193→
   194→  const {
   195→    data: modulesData,
   196→    isLoading: modulesLoading,
   197→  } = useQuery<{ modules: SonModuleItem[] }>({
   198→    queryKey: ['son-modules'],
   199→    queryFn: () => fetch('/api/son').then((r) => r.json()),
   200→    refetchInterval: 15000,
   201→  });
   202→
   203→  const {
   204→    data: actionsData,
   205→    isLoading: actionsLoading,
   206→  } = useQuery<{ actions: SonActionItem[]; pagination?: any }>({
   207→    queryKey: ['son-actions'],
   208→    queryFn: () =>
   209→      fetch('/api/son/actions?limit=50').then((r) => r.json()),
   210→    refetchInterval: 15000,
   211→  });
   212→
   213→  const {
   214→    data: neighborsData,
   215→    isLoading: neighborsLoading,
   216→  } = useQuery<{ neighbors: NeighborRelationItem[] }>({
   217→    queryKey: ['son-neighbors'],
   218→    queryFn: () =>
   219→      fetch('/api/son/neighbors').then((r) => r.json()),
   220→    refetchInterval: 15000,
   221→  });
   222→
   223→  // ─── Mutations ────────────────────────────────────────────────────
   224→
   225→  const moduleMutation = useMutation({
   226→    mutationFn: (body: {
   227→      moduleId: string;
   228→      action: 'toggle' | 'execute' | 'rollback';
   229→    }) =>
   230→      fetch('/api/son', {
   231→        method: 'PATCH',
   232→        headers: { 'Content-Type': 'application/json' },
   233→        body: JSON.stringify(body),
   234→      }).then((r) => {
   235→        if (!r.ok) return r.json().then((e) => Promise.reject(e));
   236→        return r.json();
   237→      }),
   238→    onSuccess: (_data, variables) => {
   239→      const actionLabel =
   240→        variables.action === 'toggle'
   241→          ? t('son.toggled')
   242→          : variables.action === 'execute'
   243→          ? t('son.executed')
   244→          : t('son.rolledBack');
   245→      toast.success(t('son.moduleToggled', { action: actionLabel }));
   246→      queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   247→      queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   248→    },
   249→    onError: (error: any) => {
   250→      toast.error(error.error || t('son.failedToAction', { action: error.action || variables.action }));
   251→    },
   252→  });
   253→
   254→  const rollbackActionMutation = useMutation({
   255→    mutationFn: (actionId: string) =>
   256→      fetch('/api/son/actions', {
   257→        method: 'PATCH',
   258→        headers: { 'Content-Type': 'application/json' },
   259→        body: JSON.stringify({ actionId, action: 'rollback' }),
   260→      }).then((r) => {
   261→        if (!r.ok) return r.json().then((e) => Promise.reject(e));
   262→        return r.json();
   263→      }),
   264→    onSuccess: () => {
   265→      toast.success(t('son.actionRolledBack'));
   266→      queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   267→      queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   268→    },
   269→    onError: (error: any) => {
   270→      toast.error(error.error || t('son.failedToRollback'));
   271→    },
   272→  });
   273→
   274→  // ─── Derived Data ─────────────────────────────────────────────────
   275→
   276→  const modules = modulesData?.modules ?? [];
   277→
   278→  const filteredModules = useMemo(() => {
   279→    if (modeFilter === 'ALL') return modules;
   280→    return modules.filter((m) => m.mode === modeFilter);
   281→  }, [modules, modeFilter]);
   282→
   283→  const stats = useMemo(() => {
   284→    const totalModules = modules.length;
   285→    const activeModules = modules.filter((m) => m.enabled).length;
   286→    const totalActions = modules.reduce(
   287→      (sum, m) => sum + (m.actionCount ?? 0),
   288→      0
   289→    );
   290→    const allImpactScores = (actionsData?.actions ?? [])
   291→      .map((a) => a.impactScore)
   292→      .filter((s): s is number => s !== null && s !== undefined);
   293→    const avgImpact =
   294→      allImpactScores.length > 0
   295→        ? allImpactScores.reduce((a, b) => a + b, 0) / allImpactScores.length
   296→        : 0;
   297→
   298→    return { totalModules, activeModules, totalActions, avgImpact };
   299→  }, [modules, actionsData]);
   300→
   301→  const actions = actionsData?.actions ?? [];
   302→  const neighbors = neighborsData?.neighbors ?? [];
   303→
   304→  // ─── Handlers ─────────────────────────────────────────────────────
   305→
   306→  const handleToggleModule = (moduleId: string) => {
   307→    moduleMutation.mutate({ moduleId, action: 'toggle' });
   308→  };
   309→
   310→  const handleExecuteModule = (moduleId: string, displayName: string) => {
   311→    moduleMutation.mutate(
   312→      { moduleId, action: 'execute' },
   313→      {
   314→        onSuccess: () => {
   315→          toast.success(t('son.moduleExecuted', { name: displayName }));
   316→          queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   317→          queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   318→        },
   319→      }
   320→    );
   321→  };
   322→
   323→  const handleRollbackAction = (actionId: string) => {
   324→    rollbackActionMutation.mutate(actionId);
   325→  };
   326→
   327→  const handleRefresh = () => {
   328→    queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   329→    queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   330→    queryClient.invalidateQueries({ queryKey: ['son-neighbors'] });
   331→    toast.success(t('toast.dataRefreshed'));
   332→  };
   333→
   334→  const formatTime = (ts: string) => {
   335→    try {
   336→      return new Date(ts).toLocaleString(undefined, {
   337→        month: 'short',
   338→        day: 'numeric',
   339→        hour: '2-digit',
   340→        minute: '2-digit',
   341→      });
   342→    } catch {
   343→      return ts;
   344→    }
   345→  };
   346→
   347→  // ─── Render ───────────────────────────────────────────────────────
   348→
   349→  return (
   350→    <div className="space-y-6">
   351→      {/* ─── Header ───────────────────────────────────────────────── */}
   352→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   353→        <div className="flex items-center gap-3">
   354→          <div className="flex items-center justify-center h-10 w-10 rounded-lg bg-primary/10">
   355→            <Cpu className="h-5 w-5 text-primary" />
   356→          </div>
   357→          <div>
   358→            <h1 className="text-2xl font-bold tracking-tight">
   359→              {t('title.sonAutomation', { defaultValue: 'SON Automation' })}
   360→            </h1>
   361→            <p className="text-sm text-muted-foreground">
   362→              {t('son.subtitle')}
   363→            </p>
   364→          </div>
   365→        </div>
   366→        <div className="flex items-center gap-3">
   367→          <div className="flex items-center gap-2">
   368→            <Filter className="h-4 w-4 text-muted-foreground" />
   369→            <Select
   370→              value={selectedTechnology}
   371→              onValueChange={(v) =>
   372→                useAppStore.getState().setSelectedTechnology(v as Technology)
   373→              }
   374→            >
   375→              <SelectTrigger size="sm" className="w-[90px]">
   376→                <SelectValue />
   377→              </SelectTrigger>
   378→              <SelectContent>
   379→                {TECH_OPTIONS.map((t) => (
   380→                  <SelectItem key={t} value={t}>
   381→                    {t}
   382→                  </SelectItem>
   383→                ))}
   384→              </SelectContent>
   385→            </Select>
   386→          </div>
   387→          <Select
   388→            value={modeFilter}
   389→            onValueChange={(v) =>
   390→              setModeFilter(v as SonModuleMode | 'ALL')
   391→            }
   392→          >
   393→            <SelectTrigger size="sm" className="w-[170px]">
   394→              <SelectValue placeholder="Mode" />
   395→            </SelectTrigger>
   396→            <SelectContent>
   397→              {MODE_OPTIONS.map((m) => (
   398→                <SelectItem key={m} value={m}>
   399→                  {m === 'ALL'
   400→                    ? t('filter.allModes')
   401→                    : MODE_BADGE_CONFIG[m as SonModuleMode].label}
   402→                </SelectItem>
   403→              ))}
   404→            </SelectContent>
   405→          </Select>
   406→          <Button
   407→            variant="outline"
   408→            size="sm"
   409→            onClick={handleRefresh}
   410→            className="h-8"
   411→          >
   412→            <RefreshCw className="h-3.5 w-3.5 mr-1.5" />
   413→            Refresh
   414→          </Button>
   415→        </div>
   416→      </div>
   417→
   418→      {/* ─── Stats Cards ─────────────────────────────────────────── */}
   419→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   420→        <StatsCard
   421→          title={t('son.totalModules')}
   422→          value={stats.totalModules}
   423→          icon={<Layers className="h-5 w-5" />}
   424→          iconColor="text-slate-500"
   425→          iconBg="bg-slate-500/10"
   426→          loading={modulesLoading}
   427→        />
   428→        <StatsCard
   429→          title={t('son.activeModules')}
   430→          value={stats.activeModules}
   431→          subtitle={t('son.ofTotal', { n: stats.totalModules })}
   432→          icon={<Zap className="h-5 w-5" />}
   433→          iconColor="text-emerald-500"
   434→          iconBg="bg-emerald-500/10"
   435→          loading={modulesLoading}
   436→        />
   437→        <StatsCard
   438→          title={t('son.totalActions')}
   439→          value={stats.totalActions}
   440→          icon={<Activity className="h-5 w-5" />}
   441→          iconColor="text-amber-500"
   442→          iconBg="bg-amber-500/10"
   443→          loading={actionsLoading}
   444→        />
   445→        <StatsCard
   446→          title={t('son.avgImpact')}
   447→          value={stats.avgImpact}
   448→          isDecimal
   449→          icon={<Gauge className="h-5 w-5" />}
   450→          iconColor="text-cyan-500"
   451→          iconBg="bg-cyan-500/10"
   452→          loading={actionsLoading}
   453→        />
   454→      </div>
   455→
   456→      {/* ─── Module Cards Grid ────────────────────────────────────── */}
   457→      <div>
   458→        <h2 className="text-lg font-semibold mb-4">{t('son.modules')}</h2>
   459→        {modulesLoading ? (
   460→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   461→            {Array.from({ length: 6 }).map((_, i) => (
   462→              <Skeleton key={i} className="h-72 rounded-xl" />
   463→            ))}
   464→          </div>
   465→        ) : filteredModules.length === 0 ? (
   466→          <Card className="py-16">
   467→            <CardContent className="flex flex-col items-center justify-center text-center gap-2">
   468→              <Cpu className="h-10 w-10 text-muted-foreground/50" />
   469→              <p className="text-sm text-muted-foreground">
   470→                {t('son.noModules')}
   471→              </p>
   472→            </CardContent>
   473→          </Card>
   474→        ) : (
   475→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   476→            {filteredModules.map((mod) => (
   477→              <ModuleCard
   478→                key={mod.id}
   479→                module={mod}
   480→                onToggle={handleToggleModule}
   481→                onExecute={handleExecuteModule}
   482→                isMutating={moduleMutation.isPending}
   483→              />
   484→            ))}
   485→          </div>
   486→        )}
   487→      </div>
   488→
   489→      {/* ─── Tabs: Actions & Neighbors ────────────────────────────── */}
   490→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   491→        <TabsList className="grid w-full grid-cols-2 max-w-md">
   492→          <TabsTrigger value="actions" className="flex items-center gap-2">
   493→            <ArrowRightLeft className="h-3.5 w-3.5" />
   494→            {t('son.actionHistory')}
   495→            {!actionsLoading && actions.length > 0 && (
   496→              <Badge
   497→                variant="secondary"
   498→                className="ml-1 h-5 px-1.5 text-[10px] font-mono"
   499→              >
   500→                {actions.length}
   501→              </Badge>
   502→            )}
   503→          </TabsTrigger>
   504→          <TabsTrigger value="neighbors" className="flex items-center gap-2">
   505→            <Network className="h-3.5 w-3.5" />
   506→            {t('son.neighborRelations')}
   507→            {!neighborsLoading && neighbors.length > 0 && (
   508→              <Badge
   509→                variant="secondary"
   510→                className="ml-1 h-5 px-1.5 text-[10px] font-mono"
   511→              >
   512→                {neighbors.length}
   513→              </Badge>
   514→            )}
   515→          </TabsTrigger>
   516→        </TabsList>
   517→
   518→        {/* ── Action History Tab ──────────────────────────────────── */}
   519→        <TabsContent value="actions" className="mt-4">
   520→          <Card>
   521→            <CardHeader className="pb-3">
   522→              <CardTitle className="text-base font-semibold flex items-center gap-2">
   523→                <Activity className="h-4 w-4 text-muted-foreground" />
   524→                {t('son.recentActions')}
   525→              </CardTitle>
   526→            </CardHeader>
   527→            <CardContent className="p-0">
   528→              {actionsLoading ? (
   529→                <div className="p-6 space-y-3">
   530→                  {Array.from({ length: 5 }).map((_, i) => (
   531→                    <Skeleton key={i} className="h-12 w-full" />
   532→                  ))}
   533→                </div>
   534→              ) : actions.length === 0 ? (
   535→                <div className="py-12 text-center text-sm text-muted-foreground">
   536→                  {t('son.noActions')}
   537→                </div>
   538→              ) : (
   539→                <ScrollArea className="max-h-96">
   540→                  <Table>
   541→                    <TableHeader>
   542→                      <TableRow className="hover:bg-transparent">
   543→                        <TableHead className="w-[150px] text-xs">{t('son.time')}</TableHead>
   544→                        <TableHead className="text-xs">{t('son.module')}</TableHead>
   545→                        <TableHead className="text-xs">{t('son.site')}</TableHead>
   546→                        <TableHead className="text-xs">{t('son.actionType')}</TableHead>
   547→                        <TableHead className="text-xs">{t('son.parameter')}</TableHead>
   548→                        <TableHead className="text-xs">{t('son.change')}</TableHead>
   549→                        <TableHead className="text-xs">{t('th.status')}</TableHead>
   550→                        <TableHead className="text-xs text-right">{t('son.impact')}</TableHead>
   551→                        <TableHead className="text-xs text-right w-[80px]">
   552→                          {t('th.rollback')}
   553→                        </TableHead>
   554→                      </TableRow>
   555→                    </TableHeader>
   556→                    <TableBody>
   557→                      {actions.map((action) => {
   558→                        const statusCfg =
   559→                          ACTION_STATUS_BADGE[action.status];
   560→                        return (
   561→                          <TableRow key={action.id}>
   562→                            <TableCell className="text-xs text-muted-foreground py-3">
   563→                              {formatTime(action.createdAt)}
   564→                            </TableCell>
   565→                            <TableCell className="py-3">
   566→                              <div className="text-xs font-medium">
   567→                                {action.moduleDisplayName ||
