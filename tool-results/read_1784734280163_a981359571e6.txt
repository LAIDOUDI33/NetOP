   104→   104→function loadBarColor(load: number): string {
   105→   105→  if (load >= 80) return 'bg-red-500';
   106→   106→  if (load >= 60) return 'bg-amber-500';
   107→   107→  return 'bg-emerald-500';
   108→   108→}
   109→   109→
   110→   110→function loadTextColor(load: number): string {
   111→   111→  if (load >= 80) return 'text-red-600 dark:text-red-400';
   112→   112→  if (load >= 60) return 'text-amber-600 dark:text-amber-400';
   113→   113→  return 'text-emerald-600 dark:text-emerald-400';
   114→   114→}
   115→   115→
   116→   116→function loadProgressClass(load: number): string {
   117→   117→  if (load >= 80) return '[&>div]:bg-red-500';
   118→   118→  if (load >= 60) return '[&>div]:bg-amber-500';
   119→   119→  return '[&>div]:bg-emerald-500';
   120→   120→}
   121→   121→
   122→   122→function computeTypeStats(slices: NetworkSlice[], type: string) {
   123→   123→  const typed = slices.filter((s) => s.sliceType === type);
   124→   124→  if (typed.length === 0) return { count: 0, avgLoad: 0, totalUsers: 0 };
   125→   125→  const avgLoad = Number((typed.reduce((sum, s) => sum + s.currentLoad, 0) / typed.length).toFixed(1));
   126→   126→  const totalUsers = typed.reduce((sum, s) => sum + s.activeUsers, 0);
   127→   127→  return { count: typed.length, avgLoad, totalUsers };
   128→   128→}
