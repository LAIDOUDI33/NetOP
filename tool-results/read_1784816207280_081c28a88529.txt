   100→// ─── Helper Functions ──────────────────────────────────────────────────
   101→
   102→function utilizationColor(pct: number): string {
   103→  if (pct > 80) return 'text-red-600 dark:text-red-400';
   104→  if (pct > 60) return 'text-amber-600 dark:text-amber-400';
   105→  return 'text-emerald-600 dark:text-emerald-400';
   106→}
   107→
   108→function utilizationBarColor(pct: number): string {
   109→  if (pct > 80) return 'bg-red-500';
   110→  if (pct > 60) return 'bg-amber-500';
   111→  return 'bg-emerald-500';
   112→}
   113→
   114→function utilizationChartColor(pct: number): string {
   115→  if (pct > 80) return '#EF4444';
   116→  if (pct > 60) return '#F59E0B';
   117→  return '#10B981';
   118→}
   119→
   120→function formatCurrency(value: number): string {
   121→  if (value >= 1_000_000) return `$${((value ?? 0) / 1_000_000).toFixed(1)}M`;
   122→  if (value >= 1_000) return `$${((value ?? 0) / 1_000).toFixed(1)}K`;
   123→  return `$${value.toLocaleString()}`;
   124→}

   178→function ChartTooltipContent({ active, payload, label }: any) {
   179→  if (!active || !payload?.length) return null;
   180→  return (
   181→    <div className="rounded-lg border bg-background px-3 py-2 shadow-md text-xs">
   182→      <p className="font-medium mb-1">{label}</p>
   183→      {payload.map((entry: any, idx: number) => (
   184→        <div key={idx} className="flex items-center gap-2">
