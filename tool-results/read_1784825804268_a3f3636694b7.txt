   179→// ─── Component ───────────────────────────────────────────────────────
   180→
   181→export default function SonView() {
   182→  const t = useT();
   183→  const queryClient = useQueryClient();
   184→  const selectedTechnology = useAppStore((s) => s.selectedTechnology);
   185→  const [modeFilter, setModeFilter] = useState<SonModuleMode | 'ALL'>('ALL');
   186→  const [activeTab, setActiveTab] = useState('actions');
   187→
   188→
   189→  // ─── Queries ──────────────────────────────────────────────────────
   190→  // Note: SON modules use compound technology (e.g. "4G,5G"), so we
   191→  // fetch all modules without technology filter.
   192→
   193→  const {
   194→    data: modulesData,
   195→    isLoading: modulesLoading,
   196→  } = useQuery<{ modules: SonModuleItem[] }>({
   197→    queryKey: ['son-modules'],
   198→    queryFn: () => fetch('/api/son').then((r) => { if (!r.ok) throw new Error('SON API error: ' + r.status); return r.json(); }),
   199→    refetchInterval: 15000,
   200→  });
   201→
   202→  const {
   203→    data: actionsData,
   204→    isLoading: actionsLoading,
   205→  } = useQuery<{ actions: SonActionItem[]; pagination?: any }>({
   206→    queryKey: ['son-actions'],
   207→    queryFn: () =>
   208→      fetch('/api/son/actions?limit=50').then((r) => { if (!r.ok) throw new Error('SON Actions API error: ' + r.status); return r.json(); }),
   209→    refetchInterval: 15000,
   210→  });
   211→
   212→  const {
   213→    data: neighborsData,
   214→    isLoading: neighborsLoading,
   215→  } = useQuery<{ neighbors: NeighborRelationItem[] }>({
   216→    queryKey: ['son-neighbors'],
   217→    queryFn: () =>
   218→      fetch('/api/son/neighbors').then((r) => { if (!r.ok) throw new Error('SON Neighbors API error: ' + r.status); return r.json(); }),
   219→    refetchInterval: 15000,
   220→  });
   221→
   222→  // ─── Mutations ────────────────────────────────────────────────────
   223→
   224→  const moduleMutation = useMutation({
   225→    mutationFn: (body: {
   226→      moduleId: string;
   227→      action: 'toggle' | 'execute' | 'rollback';
   228→    }) =>
   229→      fetch('/api/son', {
   230→        method: 'PATCH',
   231→        headers: { 'Content-Type': 'application/json' },
   232→        body: JSON.stringify(body),
   233→      }).then((r) => {
   234→        if (!r.ok) return r.json().then((e) => Promise.reject(e));
   235→        return r.json();
   236→      }),
   237→    onSuccess: (_data, variables) => {
   238→      const actionLabel =
   239→        variables.action === 'toggle'
   240→          ? t('son.toggled')
   241→          : variables.action === 'execute'
   242→          ? t('son.executed')
   243→          : t('son.rolledBack');
   244→      toast.success(t('son.moduleToggled', { action: actionLabel }));
   245→      queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   246→      queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   247→    },
   248→    onError: (error: any, variables: any) => {
   249→      toast.error(error.error || t('son.failedToAction', { action: error.action || variables.action }));
   250→    },
   251→  });
   252→
   253→  const rollbackActionMutation = useMutation({
   254→    mutationFn: (actionId: string) =>
   255→      fetch('/api/son/actions', {
   256→        method: 'PATCH',
   257→        headers: { 'Content-Type': 'application/json' },
   258→        body: JSON.stringify({ actionId, action: 'rollback' }),
   259→      }).then((r) => {
   260→        if (!r.ok) return r.json().then((e) => Promise.reject(e));
   261→        return r.json();
   262→      }),
   263→    onSuccess: () => {
   264→      toast.success(t('son.actionRolledBack'));
   265→      queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   266→      queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   267→    },
   268→    onError: (error: any) => {
   269→      toast.error(error.error || t('son.failedToRollback'));
   270→    },
   271→  });
   272→
   273→  // ─── Derived Data ─────────────────────────────────────────────────
   274→
   275→  const modules = modulesData?.modules ?? [];
   276→
   277→  const filteredModules = useMemo(() => {
   278→    if (modeFilter === 'ALL') return modules;
   279→    return modules.filter((m) => m.mode === modeFilter);
   280→  }, [modules, modeFilter]);
   281→
   282→  const stats = useMemo(() => {
   283→    const totalModules = modules.length;
   284→    const activeModules = modules.filter((m) => m.enabled).length;
   285→    const totalActions = modules.reduce(
   286→      (sum, m) => sum + (m.actionCount ?? 0),
   287→      0
   288→    );
   289→    const allImpactScores = (actionsData?.actions ?? [])
   290→      .map((a) => a.impactScore)
   291→      .filter((s): s is number => s !== null && s !== undefined);
   292→    const avgImpact =
   293→      allImpactScores.length > 0
   294→        ? allImpactScores.reduce((a, b) => a + b, 0) / allImpactScores.length
   295→        : 0;
   296→
   297→    return { totalModules, activeModules, totalActions, avgImpact };
   298→  }, [modules, actionsData]);
   299→
   300→  const actions = actionsData?.actions ?? [];
   301→  const neighbors = neighborsData?.neighbors ?? [];
   302→
   303→  // ─── Handlers ─────────────────────────────────────────────────────
   304→
   305→  const handleToggleModule = (moduleId: string) => {
   306→    moduleMutation.mutate({ moduleId, action: 'toggle' });
   307→  };
   308→
   309→  const handleExecuteModule = (moduleId: string, displayName: string) => {
   310→    moduleMutation.mutate(
   311→      { moduleId, action: 'execute' },
   312→      {
   313→        onSuccess: () => {
   314→          toast.success(t('son.moduleExecuted', { name: displayName }));
   315→          queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   316→          queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   317→        },
   318→      }
   319→    );
   320→  };
   321→
   322→  const handleRollbackAction = (actionId: string) => {
   323→    rollbackActionMutation.mutate(actionId);
   324→  };
   325→
   326→  const handleRefresh = () => {
   327→    queryClient.invalidateQueries({ queryKey: ['son-modules'] });
   328→    queryClient.invalidateQueries({ queryKey: ['son-actions'] });
   329→    queryClient.invalidateQueries({ queryKey: ['son-neighbors'] });
   330→    toast.success(t('toast.dataRefreshed'));
   331→  };
   332→
   333→  const formatTime = (ts: string) => {
   334→    try {
   335→      return new Date(ts).toLocaleString(undefined, {
   336→        month: 'short',
   337→        day: 'numeric',
   338→        hour: '2-digit',
   339→        minute: '2-digit',
   340→      });
   341→    } catch {
   342→      return ts;
   343→    }
   344→  };
   345→
   346→  // ─── Render ───────────────────────────────────────────────────────
   347→
   348→  return (
   349→    <div className="space-y-6">
   350→      {/* ─── Header ───────────────────────────────────────────────── */}
   351→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   352→        <div className="flex items-center gap-3">
   353→          <div className="flex items-center justify-center h-10 w-10 rounded-lg bg-primary/10">
   354→            <Cpu className="h-5 w-5 text-primary" />
   355→          </div>
   356→          <div>
   357→            <h1 className="text-2xl font-bold tracking-tight">
   358→              {t('title.sonAutomation', { defaultValue: 'SON Automation' })}
   359→            </h1>
   360→            <p className="text-sm text-muted-foreground">
   361→              {t('son.subtitle')}
   362→            </p>
   363→          </div>
   364→        </div>
   365→        <div className="flex items-center gap-3">
   366→          <div className="flex items-center gap-2">
   367→            <Filter className="h-4 w-4 text-muted-foreground" />
   368→            <Select
   369→              value={selectedTechnology}
   370→              onValueChange={(v) =>
   371→                useAppStore.getState().setSelectedTechnology(v as Technology)
   372→              }
   373→            >
   374→              <SelectTrigger size="sm" className="w-[90px]">
   375→                <SelectValue />
   376→              </SelectTrigger>
   377→              <SelectContent>
   378→                {TECH_OPTIONS.map((tech) => (
   379→                  <SelectItem key={tech} value={tech}>
   380→                    {tech}
   381→                  </SelectItem>
   382→                ))}
   383→              </SelectContent>
   384→            </Select>
   385→          </div>
   386→          <Select
   387→            value={modeFilter}
   388→            onValueChange={(v) =>
   389→              setModeFilter(v as SonModuleMode | 'ALL')
   390→            }
   391→          >
   392→            <SelectTrigger size="sm" className="w-[170px]">
   393→              <SelectValue placeholder="Mode" />
   394→            </SelectTrigger>
   395→            <SelectContent>
   396→              {MODE_OPTIONS.map((m) => (
   397→                <SelectItem key={m} value={m}>
   398→                  {m === 'ALL'
   399→                    ? t('filter.allModes')
   400→                    : MODE_BADGE_CONFIG[m as SonModuleMode].label}
   401→                </SelectItem>
   402→              ))}
   403→            </SelectContent>
   404→          </Select>
   405→          <Button
   406→            variant="outline"
   407→            size="sm"
   408→            onClick={handleRefresh}
   409→            className="h-8"
   410→          >
   411→            <RefreshCw className="h-3.5 w-3.5 mr-1.5" />
   412→            Refresh
   413→          </Button>
   414→        </div>
   415→      </div>
   416→
   417→      {/* ─── Stats Cards ─────────────────────────────────────────── */}
   418→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   419→        <StatsCard
   420→          title={t('son.totalModules')}
   421→          value={stats.totalModules}
   422→          icon={<Layers className="h-5 w-5" />}
   423→          iconColor="text-slate-500"
   424→          iconBg="bg-slate-500/10"
   425→          loading={modulesLoading}
   426→        />
   427→        <StatsCard
   428→          title={t('son.activeModules')}
   429→          value={stats.activeModules}
   430→          subtitle={t('son.ofTotal', { n: stats.totalModules })}
   431→          icon={<Zap className="h-5 w-5" />}
   432→          iconColor="text-emerald-500"
   433→          iconBg="bg-emerald-500/10"
   434→          loading={modulesLoading}
   435→        />
   436→        <StatsCard
   437→          title={t('son.totalActions')}
   438→          value={stats.totalActions}
   439→          icon={<Activity className="h-5 w-5" />}
   440→          iconColor="text-amber-500"
   441→          iconBg="bg-amber-500/10"
   442→          loading={actionsLoading}
   443→        />
   444→        <StatsCard
   445→          title={t('son.avgImpact')}
   446→          value={stats.avgImpact}
   447→          isDecimal
   448→          icon={<Gauge className="h-5 w-5" />}
   449→          iconColor="text-cyan-500"
   450→          iconBg="bg-cyan-500/10"
   451→          loading={actionsLoading}
   452→        />
   453→      </div>
   454→
   455→      {/* ─── Module Cards Grid ────────────────────────────────────── */}
   456→      <div>
   457→        <h2 className="text-lg font-semibold mb-4">{t('son.modules')}</h2>
   458→        {modulesLoading ? (
   459→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   460→            {Array.from({ length: 6 }).map((_, i) => (
   461→              <Skeleton key={i} className="h-72 rounded-xl" />
   462→            ))}
   463→          </div>
   464→        ) : filteredModules.length === 0 ? (
   465→          <Card className="py-16">
   466→            <CardContent className="flex flex-col items-center justify-center text-center gap-2">
   467→              <Cpu className="h-10 w-10 text-muted-foreground/50" />
   468→              <p className="text-sm text-muted-foreground">
   469→                {t('son.noModules')}
   470→              </p>
   471→            </CardContent>
   472→          </Card>
   473→        ) : (
   474→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   475→            {filteredModules.map((mod) => (
   476→              <ModuleCard
   477→                key={mod.id}
   478→                module={mod}
   479→                onToggle={handleToggleModule}
   480→                onExecute={handleExecuteModule}
   481→                isMutating={moduleMutation.isPending}
   482→              />
   483→            ))}
   484→          </div>
   485→        )}
   486→      </div>
   487→
   488→      {/* ─── Tabs: Actions & Neighbors ────────────────────────────── */}
   489→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   490→        <TabsList className="grid w-full grid-cols-2 max-w-md">
   491→          <TabsTrigger value="actions" className="flex items-center gap-2">
   492→            <ArrowRightLeft className="h-3.5 w-3.5" />
   493→            {t('son.actionHistory')}
   494→            {!actionsLoading && actions.length > 0 && (
   495→              <Badge
   496→                variant="secondary"
   497→                className="ml-1 h-5 px-1.5 text-[10px] font-mono"
   498→              >
   499→                {actions.length}
   500→              </Badge>
   501→            )}
   502→          </TabsTrigger>
   503→          <TabsTrigger value="neighbors" className="flex items-center gap-2">
   504→            <Network className="h-3.5 w-3.5" />
   505→            {t('son.neighborRelations')}
   506→            {!neighborsLoading && neighbors.length > 0 && (
   507→              <Badge
   508→                variant="secondary"
   509→                className="ml-1 h-5 px-1.5 text-[10px] font-mono"
   510→              >
   511→                {neighbors.length}
   512→              </Badge>
   513→            )}
   514→          </TabsTrigger>
   515→        </TabsList>
   516→
   517→        {/* ── Action History Tab ──────────────────────────────────── */}
   518→        <TabsContent value="actions" className="mt-4">
   519→          <Card>
   520→            <CardHeader className="pb-3">
   521→              <CardTitle className="text-base font-semibold flex items-center gap-2">
   522→                <Activity className="h-4 w-4 text-muted-foreground" />
   523→                {t('son.recentActions')}
   524→              </CardTitle>
   525→              <ExportButton data={actions} filenamePrefix="son" columns={[{ key: 'createdAt', header: 'Time' }, { key: 'moduleName', header: 'Module' }, { key: 'siteName', header: 'Site' }, { key: 'actionType', header: 'Action Type' }, { key: 'parameter', header: 'Parameter' }, { key: 'oldValue', header: 'Old Value' }, { key: 'newValue', header: 'New Value' }, { key: 'status', header: 'Status' }]} />
   526→            </CardHeader>
   527→            <CardContent className="p-0">
   528→              {actionsLoading ? (
   529→                <div className="p-6 space-y-3">
   530→                  {Array.from({ length: 5 }).map((_, i) => (
   531→                    <Skeleton key={i} className="h-12 w-full" />
   532→                  ))}
   533→                </div>
   534→              ) : actions.length === 0 ? (
   535→                <div className="py-12 text-center text-sm text-muted-foreground">
   536→                  {t('son.noActions')}
   537→                </div>
   538→              ) : (
   539→                <ScrollArea className="max-h-96">
   540→                  <Table>
   541→                    <TableHeader>
   542→                      <TableRow className="hover:bg-transparent">
   543→                        <TableHead className="w-[150px] text-xs">{t('son.time')}</TableHead>
   544→                        <TableHead className="text-xs">{t('son.module')}</TableHead>
   545→                        <TableHead className="text-xs">{t('son.site')}</TableHead>
   546→                        <TableHead className="text-xs">{t('son.actionType')}</TableHead>
   547→                        <TableHead className="text-xs">{t('son.parameter')}</TableHead>
   548→                        <TableHead className="text-xs">{t('son.change')}</TableHead>
   549→                        <TableHead className="text-xs">{t('th.status')}</TableHead>
   550→                        <TableHead className="text-xs text-right">{t('son.impact')}</TableHead>
   551→                        <TableHead className="text-xs text-right w-[80px]">
   552→                          {t('th.rollback')}
   553→                        </TableHead>
   554→                      </TableRow>
